<script>
(function () {
  'use strict';

  const $ = (id) => document.getElementById(id);
  const conv = $('conversation');
  const emptyHint = $('empty-hint');
  const promptEl = $('prompt');
  const composer = $('composer');
  const sendBtn = $('send');
  const stopBtn = $('stop');
  const regenBtn = $('regenerate');
  const statusEl = $('status');

  // Conversation model. Each entry: { role, content, note?, noteKind?, wrap?, body?, noteEl? }
  const history = [];

  // The stream currently in progress, or null.
  let run = null;

  // ---------- Helpers ----------

  function setStatus(kind, text) {
    statusEl.className = 'status ' + kind;
    statusEl.textContent = text;
  }

  function describeError(err) {
    if (err && typeof err === 'object' && err.message) return String(err.message);
    if (err) return String(err);
    return 'Unknown error';
  }

  function scrollToEnd() {
    conv.scrollTop = conv.scrollHeight;
  }

  function updateEmptyHint() {
    emptyHint.hidden = history.length > 0;
  }

  // Disable a button without losing keyboard focus. If the button currently has
  // focus, move focus to the message box (which is never disabled) first, so
  // focus does not fall back to the document body.
  function setDisabled(btn, value) {
    if (value && document.activeElement === btn) {
      promptEl.focus();
    }
    btn.disabled = value;
  }

  function appendEntry(entry) {
    const wrap = document.createElement('div');
    wrap.className = 'msg ' + entry.role;

    const label = document.createElement('div');
    label.className = 'label';
    label.textContent = entry.role === 'user' ? 'You' : 'Assistant';

    const body = document.createElement('div');
    body.className = 'body';
    body.textContent = entry.content;

    wrap.append(label, body);
    entry.wrap = wrap;
    entry.body = body;
    entry.noteEl = null;

    conv.appendChild(wrap);

    if (entry.note) setNote(entry, entry.note, entry.noteKind);
  }

  function setNote(entry, text, kind) {
    entry.note = text;
    entry.noteKind = kind;
    if (!entry.noteEl) {
      entry.noteEl = document.createElement('div');
      entry.wrap.appendChild(entry.noteEl);
    }
    entry.noteEl.className = 'note ' + kind;
    entry.noteEl.textContent = text;
  }

  function renderAll() {
    conv.textContent = '';
    conv.appendChild(emptyHint);
    for (const entry of history) appendEntry(entry);
    updateEmptyHint();
    scrollToEnd();
  }

  function updateControls() {
    const streaming = run !== null;
    const hasUser = history.some((m) => m.role === 'user');
    setDisabled(sendBtn, streaming);
    setDisabled(stopBtn, !streaming);
    setDisabled(regenBtn, streaming || !hasUser);
  }

  // Build the payload sent to the SDK: only role/content, skipping empty assistant
  // placeholders so the server never receives an empty assistant turn.
  function buildPayload(entries) {
    return entries
      .filter((m) => !(m.role === 'assistant' && m.content === ''))
      .map((m) => ({ role: m.role, content: m.content }));
  }

  // ---------- Streaming ----------

  function startStream() {
    if (run) return;

    if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
      const entry = { role: 'assistant', content: '' };
      history.push(entry);
      appendEntry(entry);
      setNote(entry, 'The chat service is not available.', 'error');
      setStatus('error', 'Error: chat service unavailable');
      updateControls();
      updateEmptyHint();
      scrollToEnd();
      return;
    }

    const entry = { role: 'assistant', content: '' };
    history.push(entry);

    const current = { entry: entry, handle: null, done: false };
    run = current;

    appendEntry(entry);
    entry.wrap.classList.add('pending');
    updateEmptyHint();
    setStatus('generating', 'Generating…');
    updateControls();
    scrollToEnd();

    // Everything except the new (empty) assistant entry is the request context.
    const messages = buildPayload(history.slice(0, -1));

    const callbacks = {
      onToken(text) {
        if (run !== current || current.done) return; // ignore late chunks after stop/finish
        entry.content += String(text);
        entry.wrap.classList.remove('pending');
        entry.body.textContent = entry.content;
        scrollToEnd();
      },
      onDone() {
        if (run !== current || current.done) return;
        finish(current, 'finished');
      },
      onError(err) {
        if (run !== current || current.done) return;
        finish(current, 'error', err);
      }
    };

    try {
      current.handle = window.ChatSDK.streamChat(messages, callbacks);
    } catch (err) {
      if (run === current && !current.done) finish(current, 'error', err);
    }
  }

  function finish(current, state, err) {
    if (current.done) return;
    current.done = true;
    if (run === current) run = null;

    const entry = current.entry;
    entry.wrap.classList.remove('pending');
    entry.body.textContent = entry.content;

    if (state === 'finished') {
      setNote(entry, 'Finished', 'finished');
      setStatus('finished', 'Finished');
    } else if (state === 'stopped') {
      setNote(entry, 'Stopped', 'stopped');
      setStatus('stopped', 'Stopped');
    } else {
      const msg = describeError(err);
      setNote(entry, 'Error: ' + msg, 'error');
      setStatus('error', 'Error: ' + msg);
    }

    updateControls();
    scrollToEnd();
  }

  function stop() {
    if (!run) return;
    const current = run;
    // Mark done first so late onToken/onDone/onError calls are ignored.
    current.done = true;
    run = null;
    try {
      if (current.handle && typeof current.handle.abort === 'function') {
        current.handle.abort();
      }
    } catch (e) {
      // Abort failures should not block the UI; the partial answer is kept.
    }
    const entry = current.entry;
    entry.wrap.classList.remove('pending');
    entry.body.textContent = entry.content;
    setNote(entry, 'Stopped', 'stopped');
    setStatus('stopped', 'Stopped');
    updateControls();
    scrollToEnd();
  }

  // ---------- Actions ----------

  function send() {
    if (run) return;
    const text = promptEl.value.trim();
    if (!text) return;

    history.push({ role: 'user', content: text });
    promptEl.value = '';

    const userEntry = history[history.length - 1];
    appendEntry(userEntry);
    updateEmptyHint();
    startStream();
  }

  function regenerate() {
    if (run) return;

    let lastUser = -1;
    for (let i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') { lastUser = i; break; }
    }
    if (lastUser < 0) return;

    // Drop everything after the last user message (the previous answer, if any).
    history.length = lastUser + 1;
    renderAll();
    startStream();
  }

  // ---------- Wiring ----------

  composer.addEventListener('submit', (e) => {
    e.preventDefault();
    send();
  });

  promptEl.addEventListener('keydown', (e) => {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });

  stopBtn.addEventListener('click', stop);
  regenBtn.addEventListener('click', regenerate);

  updateControls();
  updateEmptyHint();
})();
</script>
</body>
</html>
