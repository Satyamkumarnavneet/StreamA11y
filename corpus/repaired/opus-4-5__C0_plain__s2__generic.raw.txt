    <script>
        (function() {
            const messagesContainer = document.getElementById('messages');
            const emptyState = document.getElementById('empty-state');
            const promptInput = document.getElementById('prompt');
            const sendBtn = document.getElementById('send');
            const stopBtn = document.getElementById('stop');
            const regenerateBtn = document.getElementById('regenerate');
            const statusIndicator = document.getElementById('status');
            const statusText = document.getElementById('status-text');
            const typingDots = document.getElementById('typing-dots');

            let conversationHistory = [];
            let currentHandle = null;
            let isGenerating = false;
            let currentAssistantMessageEl = null;

            function setStatus(type, message) {
                statusIndicator.className = 'status-indicator';
                typingDots.style.display = 'none';
                
                if (type) {
                    statusIndicator.classList.add('visible', type);
                    statusText.textContent = message;
                    if (type === 'generating') {
                        typingDots.style.display = 'flex';
                    }
                }
            }

            function clearStatus() {
                statusIndicator.className = 'status-indicator';
                statusText.textContent = '';
            }

            function updateButtonStates() {
                const sendDisabled = isGenerating || !promptInput.value.trim();
                const stopDisabled = !isGenerating;
                const regenerateDisabled = isGenerating || conversationHistory.length === 0 || 
                    conversationHistory[conversationHistory.length - 1].role !== 'assistant';

                sendBtn.disabled = sendDisabled;
                sendBtn.setAttribute('aria-disabled', sendDisabled);
                
                stopBtn.disabled = stopDisabled;
                stopBtn.setAttribute('aria-disabled', stopDisabled);
                
                regenerateBtn.disabled = regenerateDisabled;
                regenerateBtn.setAttribute('aria-disabled', regenerateDisabled);
            }

            function scrollToBottom() {
                messagesContainer.scrollTop = messagesContainer.scrollHeight;
            }

            function createMessageElement(role, content) {
                const messageDiv = document.createElement('article');
                messageDiv.className = `message ${role}`;
                messageDiv.setAttribute('aria-label', role === 'user' ? 'Your message' : 'Assistant response');
                
                const labelDiv = document.createElement('div');
                labelDiv.className = 'message-label';
                labelDiv.setAttribute('aria-hidden', 'true');
                labelDiv.textContent = role === 'user' ? 'You' : 'Assistant';
                
                const contentDiv = document.createElement('div');
                contentDiv.className = 'message-content';
                contentDiv.textContent = content;
                
                messageDiv.appendChild(labelDiv);
                messageDiv.appendChild(contentDiv);
                
                return messageDiv;
            }

            function addMessageToUI(role, content) {
                if (emptyState) {
                    emptyState.style.display = 'none';
                }
                const messageEl = createMessageElement(role, content);
                messagesContainer.appendChild(messageEl);
                scrollToBottom();
                return messageEl;
            }

            function startStream(messages) {
                isGenerating = true;
                updateButtonStates();
                setStatus('generating', 'Generating response');
                
                currentAssistantMessageEl = addMessageToUI('assistant', '');
                const contentEl = currentAssistantMessageEl.querySelector('.message-content');
                let accumulatedContent = '';

                currentHandle = window.ChatSDK.streamChat(messages, {
                    onToken: function(text) {
                        accumulatedContent += text;
                        contentEl.textContent = accumulatedContent;
                        scrollToBottom();
                    },
                    onDone: function() {
                        conversationHistory.push({ role: 'assistant', content: accumulatedContent });
                        isGenerating = false;
                        currentHandle = null;
                        currentAssistantMessageEl = null;
                        setStatus('finished', 'Response complete');
                        setTimeout(clearStatus, 2000);
                        updateButtonStates();
                        promptInput.focus();
                    },
                    onError: function(error) {
                        isGenerating = false;
                        currentHandle = null;
                        if (accumulatedContent) {
                            conversationHistory.push({ role: 'assistant', content: accumulatedContent });
                        } else if (currentAssistantMessageEl) {
                            currentAssistantMessageEl.remove();
                        }
                        currentAssistantMessageEl = null;
                        setStatus('error', 'Error: ' + (error.message || 'Something went wrong'));
                        updateButtonStates();
                        promptInput.focus();
                    }
                });
            }

            function sendMessage() {
                const content = promptInput.value.trim();
                if (!content || isGenerating) return;

                conversationHistory.push({ role: 'user', content: content });
                addMessageToUI('user', content);
                promptInput.value = '';
                updateButtonStates();

                startStream([...conversationHistory]);
            }

            function stopGeneration() {
                if (currentHandle && isGenerating) {
                    currentHandle.abort();
                    
                    if (currentAssistantMessageEl) {
                        const contentEl = currentAssistantMessageEl.querySelector('.message-content');
                        const partialContent = contentEl.textContent;
                        if (partialContent) {
                            conversationHistory.push({ role: 'assistant', content: partialContent });
                        } else {
                            currentAssistantMessageEl.remove();
                        }
                    }
                    
                    isGenerating = false;
                    currentHandle = null;
                    currentAssistantMessageEl = null;
                    setStatus('stopped', 'Response stopped');
                    setTimeout(clearStatus, 2000);
                    updateButtonStates();
                    promptInput.focus();
                }
            }

            function regenerate() {
                if (isGenerating || conversationHistory.length === 0) return;

                if (conversationHistory[conversationHistory.length - 1].role === 'assistant') {
                    conversationHistory.pop();
                    const assistantMessages = messagesContainer.querySelectorAll('.message.assistant');
                    const lastAssistantMessage = assistantMessages[assistantMessages.length - 1];
                    if (lastAssistantMessage) {
                        lastAssistantMessage.remove();
                    }
                }

                startStream([...conversationHistory]);
            }

            sendBtn.addEventListener('click', sendMessage);
            
            promptInput.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    sendMessage();
                }
            });

            promptInput.addEventListener('input', updateButtonStates);

            stopBtn.addEventListener('click', stopGeneration);
            
            regenerateBtn.addEventListener('click', regenerate);

            updateButtonStates();
        })();
    </script>
</body>
