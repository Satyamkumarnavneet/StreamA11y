    <title>Chat Assistant</title>
    <style>
        * {
            margin: 0;
            padding: 0;
            box-sizing: border-box;
        }

        body {
            font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, 'Helvetica Neue', Arial, sans-serif;
            display: flex;
            flex-direction: column;
            height: 100vh;
            background: #f5f5f5;
        }

        #chat-container {
            flex: 1;
            overflow-y: auto;
            padding: 20px;
            display: flex;
            flex-direction: column;
            gap: 12px;
        }

        .message {
            display: flex;
            gap: 12px;
            animation: slideIn 0.3s ease-out;
        }

        @keyframes slideIn {
            from {
                opacity: 0;
                transform: translateY(10px);
            }
            to {
                opacity: 1;
                transform: translateY(0);
            }
        }

        .message.user {
            justify-content: flex-end;
        }

        .message.assistant {
            justify-content: flex-start;
        }

        .message-content {
            max-width: 70%;
            padding: 12px 16px;
            border-radius: 12px;
            word-wrap: break-word;
            white-space: pre-wrap;
            line-height: 1.5;
        }

        .message.user .message-content {
            background: #007bff;
            color: white;
        }

        .message.assistant .message-content {
            background: #e9ecef;
            color: #333;
        }

        .status-indicator {
            font-size: 12px;
            color: #666;
            padding: 0 16px;

        .status-dot {

            flex-shrink: 0;
        }

        .status-dot.generating {
            background: #ffc107;

        }

        .status-dot.done {
            background: #28a745;
        }

        .status-dot.error {
            background: #dc3545;
        }

        .status-dot.stopped {
            background: #6c757d;

        @keyframes pulse {
            0%, 100% { opacity: 0.6; }

        #input-area {

<body>
    <div aria-live="polite" aria-atomic="true" class="sr-only" id="sr-status"></div>
    
    <div id="chat-container" role="log" aria-label="Chat conversation">
        <div class="empty-state">Start a conversation</div>
    </div>

    <div id="input-area">
        <textarea id="prompt" placeholder="Type your message..." aria-label="Message input"></textarea>
        <button id="send" aria-label="Send message">Send</button>
        <div class="control-group">
            <button id="stop" disabled aria-label="Stop generating response">Stop</button>
            <button id="regenerate" disabled aria-label="Regenerate response to last message">Regenerate</button>
        </div>
    </div>

    <script>
        const chatContainer = document.getElementById('chat-container');
        const promptInput = document.getElementById('prompt');
        const sendBtn = document.getElementById('send');
        const stopBtn = document.getElementById('stop');
        const regenerateBtn = document.getElementById('regenerate');
        const inputArea = document.getElementById('input-area');
        const srStatus = document.getElementById('sr-status');

        let messages = [];
        let currentStream = null;
        let isStreaming = false;
        let lastUserMessage = null;
        let currentAssistantMessageEl = null;
        let statusEl = null;

        function announceToScreenReader(text) {
            srStatus.textContent = text;
        }

        function clearEmptyState() {
            const emptyState = chatContainer.querySelector('.empty-state');
            if (emptyState) {
                emptyState.remove();
            }
        }

        function createStatusElement() {
            const status = document.createElement('div');
            status.className = 'status-indicator';
            status.setAttribute('role', 'status');
            status.setAttribute('aria-live', 'polite');
            return status;
        }

        function updateStatus(text, dotClass) {
            if (!statusEl) {
                statusEl = createStatusElement();
                chatContainer.appendChild(statusEl);
            }
            statusEl.innerHTML = `<span class="status-dot ${dotClass}"></span>${text}`;
        }

        function clearStatus() {
            if (statusEl) {
                statusEl.remove();
                statusEl = null;
            }
        }

        function addMessage(role, content) {
            clearEmptyState();

            const messageEl = document.createElement('div');
            messageEl.className = `message ${role}`;

            const contentEl = document.createElement('div');
            contentEl.className = 'message-content';
            contentEl.textContent = content;

            messageEl.appendChild(contentEl);
            chatContainer.appendChild(messageEl);
            chatContainer.scrollTop = chatContainer.scrollHeight;

            return messageEl;
        }

        function updateLastAssistantMessage(content) {
            if (currentAssistantMessageEl) {
                const contentEl = currentAssistantMessageEl.querySelector('.message-content');
                contentEl.textContent = content;
                chatContainer.scrollTop = chatContainer.scrollHeight;
            }
        }

        function createNewAssistantMessage() {
            clearEmptyState();
            const messageEl = document.createElement('div');
            messageEl.className = 'message assistant';

            const contentEl = document.createElement('div');
            contentEl.className = 'message-content';
            contentEl.textContent = '';

            messageEl.appendChild(contentEl);
            chatContainer.appendChild(messageEl);
            chatContainer.scrollTop = chatContainer.scrollHeight;

            return messageEl;
        }

        function updateUIState() {
            const canSend = !isStreaming && promptInput.value.trim().length > 0;
            sendBtn.disabled = !canSend;

            stopBtn.disabled = !isStreaming;
            regenerateBtn.disabled = isStreaming || messages.length === 0 || messages[messages.length - 1].role !== 'assistant';

            if (isStreaming) {
                inputArea.classList.add('with-controls');
            }
        }

        function restoreFocus(targetBtn) {
            setTimeout(() => {
                if (targetBtn && !targetBtn.disabled) {
                    targetBtn.focus();
                } else if (sendBtn && !sendBtn.disabled) {
                    sendBtn.focus();
                } else if (promptInput) {
                    promptInput.focus();
                }
            }, 0);
        }

        function handleSendMessage() {
            const content = promptInput.value.trim();
            if (!content || isStreaming) return;

            lastUserMessage = content;
            messages.push({ role: 'user', content });
            addMessage('user', content);

            promptInput.value = '';
            promptInput.style.height = 'auto';
            promptInput.minHeight = '40px';

            isStreaming = true;
            updateUIState();
            updateStatus('Generating...', 'generating');
            announceToScreenReader('Assistant is generating a response');

            currentAssistantMessageEl = createNewAssistantMessage();
            let assistantContent = '';

            currentStream = window.ChatSDK.streamChat(messages, {
                onToken(text) {
                    assistantContent += text;
                    updateLastAssistantMessage(assistantContent);
                },
                onDone() {
                    messages.push({ role: 'assistant', content: assistantContent });
                    isStreaming = false;
                    currentStream = null;
                    clearStatus();
                    updateStatus('Done', 'done');
                    announceToScreenReader('Response complete');
                    updateUIState();
                    restoreFocus(sendBtn);
                    setTimeout(clearStatus, 2000);
                },
                onError(error) {
                    isStreaming = false;
                    currentStream = null;
                    clearStatus();
                    updateStatus(`Error: ${error.message}`, 'error');
                    announceToScreenReader(`Response failed: ${error.message}`);
                    updateUIState();
                    restoreFocus(sendBtn);
                }
            });
        }

        function handleStop() {
            if (currentStream && isStreaming) {
                currentStream.abort();
                isStreaming = false;
                currentStream = null;

                const contentEl = currentAssistantMessageEl.querySelector('.message-content');
                const assistantContent = contentEl.textContent;
                messages.push({ role: 'assistant', content: assistantContent });

                clearStatus();
                updateStatus('Stopped', 'stopped');
                announceToScreenReader('Response generation stopped');
                updateUIState();
                restoreFocus(sendBtn);
                setTimeout(clearStatus, 2000);
            }
        }

        function handleRegenerate() {
            if (isStreaming || messages.length === 0 || messages[messages.length - 1].role !== 'assistant') {
                return;
            }

            const lastUserIdx = messages.length - 2;
            if (lastUserIdx >= 0 && messages[lastUserIdx].role === 'user') {
                messages = messages.slice(0, lastUserIdx + 1);

                chatContainer.querySelectorAll('.message').forEach(el => el.remove());
                chatContainer.querySelectorAll('.status-indicator').forEach(el => el.remove());
                statusEl = null;

                messages.forEach(msg => {
                    addMessage(msg.role, msg.content);
                });

                isStreaming = true;
                updateUIState();
                updateStatus('Generating...', 'generating');
                announceToScreenReader('Regenerating response to previous message');

                currentAssistantMessageEl = createNewAssistantMessage();
                let assistantContent = '';

                currentStream = window.ChatSDK.streamChat(messages, {
                    onToken(text) {
                        assistantContent += text;
                        updateLastAssistantMessage(assistantContent);
                    },
                    onDone() {
                        messages.push({ role: 'assistant', content: assistantContent });
                        isStreaming = false;
                        currentStream = null;
                        clearStatus();
                        updateStatus('Done', 'done');
                        announceToScreenReader('Regenerated response complete');
                        updateUIState();
                        restoreFocus(sendBtn);
                        setTimeout(clearStatus, 2000);
                    },
                    onError(error) {
                        isStreaming = false;
                        currentStream = null;
                        clearStatus();
                        updateStatus(`Error: ${error.message}`, 'error');
                        announceToScreenReader(`Regeneration failed: ${error.message}`);
                        updateUIState();
                        restoreFocus(sendBtn);
                    }
                });
            }
        }

        promptInput.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' && !e.shiftKey) {
                e.preventDefault();
                handleSendMessage();
            }
        });

        promptInput.addEventListener('input', () => {
            promptInput.style.height = 'auto';
            promptInput.style.height = Math.min(promptInput.scrollHeight, 100) + 'px';
            updateUIState();
        });

        sendBtn.addEventListener('click', handleSendMessage);
        stopBtn.addEventListener('click', handleStop);
        regenerateBtn.addEventListener('click', handleRegenerate);

        updateUIState();
    </script>
</body>
</html>
