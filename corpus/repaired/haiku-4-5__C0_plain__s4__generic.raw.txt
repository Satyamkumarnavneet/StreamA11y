    <script>
        const messagesContainer = document.getElementById('messages');
        const promptInput = document.getElementById('prompt');
        const sendButton = document.getElementById('send');
        const stopButton = document.getElementById('stop');
        const regenerateButton = document.getElementById('regenerate');
        const statusAnnouncement = document.getElementById('status-announcement');

        let messages = [];
        let isStreaming = false;
        let currentStreamHandle = null;

        // Auto-resize textarea
        promptInput.addEventListener('input', () => {
            promptInput.style.height = 'auto';
            promptInput.style.height = Math.min(promptInput.scrollHeight, 120) + 'px';
        });

        // Handle Enter key
        promptInput.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' && !e.shiftKey) {
                e.preventDefault();
                sendMessage();
            }
        });

        sendButton.addEventListener('click', sendMessage);
        stopButton.addEventListener('click', stopStreaming);
        regenerateButton.addEventListener('click', regenerateLastMessage);

        function announceToScreenReader(text) {
            statusAnnouncement.textContent = text;
        }

        function clearEmptyState() {
            if (messagesContainer.innerHTML.includes('empty-state')) {
                messagesContainer.innerHTML = '';
            }
        }

        function addMessage(role, content) {
            clearEmptyState();
            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${role}`;
            
            const contentDiv = document.createElement('div');
            contentDiv.className = 'message-content';
            contentDiv.textContent = content;
            
            messageDiv.appendChild(contentDiv);
            messagesContainer.appendChild(messageDiv);
            messagesContainer.scrollTop = messagesContainer.scrollHeight;

            return contentDiv;
        }

        function addStatusIndicator(text, isError = false, isStopped = false) {
            clearEmptyState();
            const statusDiv = document.createElement('div');
            statusDiv.className = 'status-indicator';
            if (isError) statusDiv.classList.add('error');
            if (isStopped) statusDiv.classList.add('stopped');
            statusDiv.textContent = text;
            messagesContainer.appendChild(statusDiv);
            messagesContainer.scrollTop = messagesContainer.scrollHeight;
            return statusDiv;
        }

        function sendMessage() {
            const content = promptInput.value.trim();
            if (!content || isStreaming) return;

            // Add user message
            addMessage('user', content);
            announceToScreenReader(`You: ${content}`);
            messages.push({ role: 'user', content });

            // Clear input
            promptInput.value = '';
            promptInput.style.height = 'auto';

            // Update UI
            sendButton.disabled = true;
            stopButton.style.display = 'inline-block';
            regenerateButton.style.display = 'none';

            // Stream response
            streamResponse();
        }

        function streamResponse() {
            isStreaming = true;
            let assistantContent = '';
            let contentElement = null;

            announceToScreenReader('Assistant is generating a response');
            addStatusIndicator('Generating...');

            currentStreamHandle = window.ChatSDK.streamChat(messages, {
                onToken: (text) => {
                    if (!contentElement) {
                        // Remove status indicator and create message
                        const lastChild = messagesContainer.lastChild;
                        if (lastChild && lastChild.classList.contains('status-indicator')) {
                            lastChild.remove();
                        }
                        contentElement = addMessage('assistant', '');
                    }
                    assistantContent += text;
                    contentElement.textContent = assistantContent;
                    messagesContainer.scrollTop = messagesContainer.scrollHeight;
                },
                onDone: () => {
                    isStreaming = false;
                    currentStreamHandle = null;

                    // Remove status indicator if still there
                    const lastChild = messagesContainer.lastChild;
                    if (lastChild && lastChild.classList.contains('status-indicator')) {
                        lastChild.remove();
                    }

                    if (assistantContent) {
                        messages.push({ role: 'assistant', content: assistantContent });
                        addStatusIndicator('✓ Done');
                        announceToScreenReader('Response complete');
                    }

                    stopButton.style.display = 'none';
                    regenerateButton.style.display = 'inline-block';
                    sendButton.disabled = false;
                    promptInput.focus();
                },
                onError: (error) => {
                    isStreaming = false;
                    currentStreamHandle = null;

                    // Remove status indicator
                    const lastChild = messagesContainer.lastChild;
                    if (lastChild && lastChild.classList.contains('status-indicator')) {
                        lastChild.remove();
                    }

                    const errorMsg = error?.message || 'An error occurred';
                    addStatusIndicator(`Error: ${errorMsg}`, true);
                    announceToScreenReader(`Error: ${errorMsg}`);

                    stopButton.style.display = 'none';
                    regenerateButton.style.display = 'inline-block';
                    sendButton.disabled = false;
                    promptInput.focus();
                }
            });
        }

        function stopStreaming() {
            if (currentStreamHandle) {
                currentStreamHandle.abort();
                currentStreamHandle = null;
            }

            isStreaming = false;

            // Remove status indicator
            const lastChild = messagesContainer.lastChild;
            if (lastChild && lastChild.classList.contains('status-indicator')) {
                lastChild.remove();
            }

            addStatusIndicator('⊘ Stopped', false, true);
            announceToScreenReader('Response generation stopped');

            stopButton.style.display = 'none';
            regenerateButton.style.display = 'inline-block';
            sendButton.disabled = false;
            promptInput.focus();
        }

        function regenerateLastMessage() {
            if (messages.length === 0 || messages[messages.length - 1].role !== 'user') {
                return;
            }

            // Remove last assistant message if exists
            if (messages.length > 1 && messages[messages.length - 1].role === 'assistant') {
                messages.pop();

                // Remove last message div from UI
                let messageCount = 0;
                for (let i = messagesContainer.children.length - 1; i >= 0; i--) {
                    const child = messagesContainer.children[i];
                    if (child.classList.contains('message') || child.classList.contains('status-indicator')) {
                        child.remove();
                        if (child.classList.contains('message')) {
                            messageCount++;
                            if (messageCount === 1) break;
                        }
                    }
                }
            }

            announceToScreenReader('Regenerating response');

            // Update UI
            sendButton.disabled = true;
            stopButton.style.display = 'inline-block';
            regenerateButton.style.display = 'none';

            // Stream new response
            streamResponse();
        }

        // Focus input on load
        promptInput.focus();
    </script>
</body>
</html>

## Accessibility Improvements Made

**Keyboard Navigation:**
- Added `focus-visible` outlines on all interactive elements (2px solid border)
- Ensured all buttons are keyboard-accessible with proper focus states
- Textarea has visible focus styling with 3px shadow

**Screen Reader Support:**
- Added `role="log"` and `aria-live="polite"` to messages container for real-time updates
- Added semantic `aria-label` attributes to all buttons describing their actions
- Created hidden `#status-announcement` element with `role="status"` and `aria-live="polite"` for status updates
- Added `aria-label` to textarea with visible `<label>` (screen-reader-only)
- Screen reader announcements for user messages, generation status, completion, errors, and stops

**Color & Contrast:**
- Increased border thickness from 1px to 2px on inputs and buttons for better visibility
- Improved error/stopped indicator text colors with higher contrast ratios (WCAG AA compliant)
- Added `@media (prefers-contrast: more)` support with borders on all interactive elements
- Font sizes meet minimum 14px baseline

**Motion & Animation:**
- Added `@media (prefers-reduced-motion: reduce)` to disable animations for users with vestibular sensitivities
- Reduced animation durations to 0.01ms and iteration counts to 1 when reduced motion is preferred

**Semantic HTML & Labeling:**
- Proper label-input associations with `<label>` and `for` attributes
- Used `.sr-only` class for screen-reader-only content
- Clear button labels with `aria-label` for icon/minimal-text buttons
