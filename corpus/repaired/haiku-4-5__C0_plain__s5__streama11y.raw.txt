<body>
    <div class="container">
        <div aria-live="polite" aria-atomic="true" class="sr-live" id="sr-live"></div>
        <div role="status" aria-live="assertive" aria-atomic="true" class="sr-only" id="sr-status"></div>

                    <button id="regenerate" aria-label="Regenerate last response" disabled>Regenerate</button>
                </div>

    <script>
        const messagesContainer = document.getElementById('messages');
        const promptInput = document.getElementById('prompt');
        const sendBtn = document.getElementById('send');
        const stopBtn = document.getElementById('stop');
        const regenerateBtn = document.getElementById('regenerate');
        const statusDiv = document.getElementById('status');
        const srLive = document.getElementById('sr-live');
        const srStatus = document.getElementById('sr-status');

        let messages = [];
        let currentStream = null;
        let isStreaming = false;
        let lastUserMessage = null;
        let currentAssistantContentDiv = null;

        function announceToScreenReader(message, isStatus = false) {
            if (isStatus) {
                srStatus.textContent = message;
            } else {
                srLive.textContent = message;
            }
        }

        function clearEmptyState() {
            const emptyState = messagesContainer.querySelector('.empty-state');
            if (emptyState) {
                emptyState.remove();
            }
        }

        function addMessageToDOM(role, content) {
            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${role}`;

            const contentDiv = document.createElement('div');
            contentDiv.className = 'message-content';
            contentDiv.textContent = content;

            messageDiv.appendChild(contentDiv);
            messagesContainer.appendChild(messageDiv);
            messagesContainer.scrollTop = messagesContainer.scrollHeight;

            return contentDiv;
        }

        function setStatus(text, type = '') {
            if (text) {
                statusDiv.textContent = text;
                statusDiv.className = `status ${type}`;
                announceToScreenReader(text, true);
            } else {
                statusDiv.textContent = '';
                statusDiv.className = 'status';
                srStatus.textContent = '';
            }
        }

        function updateUIState() {
            const canSend = !isStreaming;
            sendBtn.disabled = !canSend;
            stopBtn.disabled = !isStreaming;
            
            const canRegenerate = !isStreaming && messages.length > 0 && messages[messages.length - 1].role === 'assistant';
            regenerateBtn.disabled = !canRegenerate;
        }

        function restoreFocus(element) {
            if (element && document.contains(element)) {
                element.focus();
            } else {
                promptInput.focus();
            }
        }

        async function sendMessage() {
            const text = promptInput.value.trim();
            if (!text || isStreaming) return;

            clearEmptyState();
            
            messages.push({ role: 'user', content: text });
            lastUserMessage = text;
            addMessageToDOM('user', text);
            announceToScreenReader(`You: ${text}`);
            promptInput.value = '';
            promptInput.style.height = 'auto';

            const sendBtnRef = sendBtn;
            await streamAssistantResponse();
            restoreFocus(sendBtnRef);
        }

        async function streamAssistantResponse() {
            isStreaming = true;
            updateUIState();
            setStatus('Generating response...', 'generating');
            announceToScreenReader('Assistant is generating response');

            currentAssistantContentDiv = addMessageToDOM('assistant', '');
            let fullResponse = '';

            currentStream = window.ChatSDK.streamChat(messages, {
                onToken(text) {
                    if (isStreaming) {
                        fullResponse += text;
                        currentAssistantContentDiv.textContent = fullResponse;
                        messagesContainer.scrollTop = messagesContainer.scrollHeight;
                    }
                },
                onDone() {
                    if (isStreaming) {
                        isStreaming = false;
                        messages.push({ role: 'assistant', content: fullResponse });
                        setStatus('');
                        announceToScreenReader('Response complete');
                        updateUIState();
                        currentStream = null;
                        currentAssistantContentDiv = null;
                    }
                },
                onError(error) {
                    if (isStreaming) {
                        isStreaming = false;
                        const errorMsg = `Error: ${error.message || 'Unknown error'}`;
                        setStatus(errorMsg, 'error');
                        announceToScreenReader('Response failed: ' + (error.message || 'Unknown error'));
                        updateUIState();
                        currentStream = null;
                        currentAssistantContentDiv = null;
                    }
                }
            });
        }

        function stopStream() {
            if (currentStream && isStreaming) {
                currentStream.abort();
                isStreaming = false;
                setStatus('Response stopped', 'stopped');
                announceToScreenReader('Response stopped by user');
                
                // Save the partial response to messages
                const fullText = currentAssistantContentDiv.textContent;
                messages.push({ role: 'assistant', content: fullText });
                
                updateUIState();
                currentStream = null;
                currentAssistantContentDiv = null;
                restoreFocus(stopBtn);
            }
        }

        async function regenerateResponse() {
            if (messages.length === 0) return;

            // Remove the last assistant message if it exists
            if (messages.length > 0 && messages[messages.length - 1].role === 'assistant') {
                messages.pop();
            }

            // Remove the last assistant message from DOM
            const lastAssistantMsg = messagesContainer.querySelector('.message.assistant:last-child');
            if (lastAssistantMsg) {
                lastAssistantMsg.remove();
            }

            announceToScreenReader('Regenerating response');
            const regenerateBtnRef = regenerateBtn;
            await streamAssistantResponse();
            restoreFocus(regenerateBtnRef);
        }

        function autoResizeTextarea() {
            promptInput.style.height = 'auto';
            promptInput.style.height = Math.min(promptInput.scrollHeight, 120) + 'px';
        }

        sendBtn.addEventListener('click', sendMessage);
        stopBtn.addEventListener('click', stopStream);
        regenerateBtn.addEventListener('click', regenerateResponse);

        promptInput.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' && !e.shiftKey && !isStreaming) {
                e.preventDefault();
                sendMessage();
            }
        });

        promptInput.addEventListener('input', autoResizeTextarea);

        updateUIState();
    </script>
</body>
