<script>
(function () {
  'use strict';

  var logEl = document.getElementById('log');
  var promptEl = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var statusEl = document.getElementById('status');
  var statusText = statusEl.querySelector('.text');
  var bannerEl = document.getElementById('banner');

  // Conversation: { role, content, status, el, bubble, tagEl }
  // status (assistant only): 'streaming' | 'finished' | 'stopped' | 'error'
  var messages = [];
  // The currently active stream session (null when idle). Each stream gets its own
  // session object, so late callbacks from an old/aborted stream can be ignored.
  var current = null;

  function isStreaming() { return current !== null; }

  function setStatus(state, text) {
    statusEl.setAttribute('data-state', state);
    statusText.textContent = text;
  }

  function showBanner(text) { bannerEl.textContent = text; }
  function hideBanner() { bannerEl.textContent = ''; }

  function lastUserIndex() {
    for (var i = messages.length - 1; i >= 0; i--) {
      if (messages[i].role === 'user') return i;
    }
    return -1;
  }

  function updateControls() {
    var hadFocus = document.activeElement;
    var streaming = isStreaming();
    sendBtn.disabled = streaming;
    stopBtn.disabled = !streaming;
    regenBtn.disabled = streaming || lastUserIndex() === -1;
    logEl.setAttribute('aria-busy', streaming ? 'true' : 'false');
    // If the focused button just became disabled, keep keyboard focus somewhere useful.
    if (hadFocus && hadFocus.tagName === 'BUTTON' && hadFocus.disabled) {
      promptEl.focus();
    }
  }

  function nearBottom() {
    return logEl.scrollHeight - logEl.scrollTop - logEl.clientHeight < 80;
  }
  function scrollToBottom() { logEl.scrollTop = logEl.scrollHeight; }

  function renderEmptyHint() {
    var existing = logEl.querySelector('.empty-hint');
    if (messages.length === 0 && !existing) {
      var h = document.createElement('div');
      h.className = 'empty-hint';
      h.textContent = 'Start the conversation by sending a message.';
      logEl.appendChild(h);
    } else if (messages.length > 0 && existing) {
      existing.remove();
    }
  }

  function createMessage(role, content, status) {
    var m = { role: role, content: content, status: status || null };
    var el = document.createElement('div');
    el.className = 'msg ' + role;
    var roleEl = document.createElement('div');
    roleEl.className = 'role';
    roleEl.textContent = role === 'user' ? 'You' : 'Assistant';
    var bubble = document.createElement('div');
    bubble.className = 'bubble';
    el.appendChild(roleEl);
    el.appendChild(bubble);
    var tag = document.createElement('div');
    tag.className = 'tag';
    tag.hidden = true;
    el.appendChild(tag);
    m.el = el; m.bubble = bubble; m.tagEl = tag;
    messages.push(m);
    renderEmptyHint();
    logEl.appendChild(el);
    renderMessage(m);
    return m;
  }

  function setTag(m, cls, icon, text) {
    var tag = m.tagEl;
    tag.className = 'tag ' + cls;
    tag.hidden = false;
    tag.textContent = '';
    var i = document.createElement('span');
    i.setAttribute('aria-hidden', 'true');
    i.textContent = icon + ' ';
    tag.appendChild(i);
    tag.appendChild(document.createTextNode(text));
  }

  function renderMessage(m) {
    if (m.role === 'user') {
      m.bubble.textContent = m.content;
      return;
    }
    m.el.classList.toggle('streaming', m.status === 'streaming');
    m.bubble.textContent = '';
    if (m.content) {
      m.bubble.textContent = m.content;
    } else if (m.status === 'streaming') {
      var t = document.createElement('span');
      t.className = 'typing';
      t.setAttribute('aria-hidden', 'true');
      t.innerHTML = '<i></i><i></i><i></i>';
      m.bubble.appendChild(t);
    } else {
      var p = document.createElement('span');
      p.className = 'placeholder';
      p.textContent = m.status === 'error' ? '(no answer received)'
                    : m.status === 'stopped' ? '(stopped before any text arrived)'
                    : '(empty answer)';
      m.bubble.appendChild(p);
    }
    if (m.status === 'stopped') {
      setTag(m, 'stopped', '■', 'Stopped — partial answer kept');
    } else if (m.status === 'error') {
      setTag(m, 'error', '⚠', (m.errorText || 'Error') + (m.content ? ' — partial answer kept' : ''));
    } else if (m.status === 'finished') {
      setTag(m, 'finished', '✓', 'Finished');
    } else {
      m.tagEl.className = 'tag';
      m.tagEl.hidden = true;
      m.tagEl.textContent = '';
    }
  }

  function errorMessage(err) {
    if (!err) return 'Unknown error';
    if (typeof err === 'string') return err;
    if (err.message) return String(err.message);
    try { return String(err); } catch (e) { return 'Unknown error'; }
  }

  // Messages to send to the API: everything up to `endIndex` (inclusive),
  // skipping assistant messages that have no text.
  function buildApiMessages(endIndex) {
    var out = [];
    for (var i = 0; i <= endIndex; i++) {
      var m = messages[i];
      if (m.role === 'assistant' && !m.content) continue;
      out.push({ role: m.role, content: m.content });
    }
    return out;
  }

  function startStream(apiMessages) {
    hideBanner();
    var assistant = createMessage('assistant', '', 'streaming');
    var session = { msg: assistant, stopped: false, handle: null, finished: false };
    current = session;
    setStatus('generating', 'Generating answer…');
    updateControls();
    scrollToBottom();

    function active() { return current === session && !session.stopped && !session.finished; }

    function finishSession() {
      session.finished = true;
      if (current === session) current = null;
    }

    var callbacks = {
      onToken: function (text) {
        if (!active()) return; // ignore late tokens after stop / done / error
        if (text == null || text === '') return;
        var stick = nearBottom();
        var wasEmpty = !assistant.content;
        assistant.content += String(text);
        if (wasEmpty) {
          renderMessage(assistant);
        } else {
          assistant.bubble.textContent = assistant.content;
        }
        if (stick) scrollToBottom();
      },
      onDone: function () {
        if (!active()) return;
        assistant.status = 'finished';
        finishSession();
        renderMessage(assistant);
        hideBanner();
        setStatus('finished', 'Finished — answer is ready');
        updateControls();
        scrollToBottom();
      },
      onError: function (err) {
        if (!active()) return;
        var text = errorMessage(err);
        assistant.status = 'error';
        assistant.errorText = 'Error: ' + text;
        finishSession();
        renderMessage(assistant);
        setStatus('error', 'Error — answer failed');
        showBanner('Something went wrong: ' + text + '. You can press Regenerate to try again.');
        updateControls();
        scrollToBottom();
      }
    };

    try {
      if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
        throw new Error('Chat service is not available');
      }
      session.handle = window.ChatSDK.streamChat(apiMessages, callbacks);
    } catch (e) {
      callbacks.onError(e);
    }
  }

  function send() {
    if (isStreaming()) {
      showBanner('Please wait for the current answer to finish, or press Stop.');
      return;
    }
    var text = promptEl.value.trim();
    if (!text) {
      promptEl.setAttribute('aria-invalid', 'true');
      showBanner('Please type a message before sending.');
      promptEl.focus();
      return;
    }
    promptEl.removeAttribute('aria-invalid');
    promptEl.value = '';
    autoSize();
    createMessage('user', text);
    startStream(buildApiMessages(messages.length - 1));
  }

  function stop() {
    var session = current;
    if (!session) return;
    session.stopped = true;   // from now on, ignore any callback for this stream
    current = null;
    try {
      if (session.handle && typeof session.handle.abort === 'function') session.handle.abort();
    } catch (e) { /* aborting is best-effort */ }
    session.msg.status = 'stopped';
    renderMessage(session.msg);
    hideBanner();
    setStatus('stopped', 'Stopped — partial answer kept');
    updateControls();
  }

  function regenerate() {
    if (isStreaming()) return;
    var idx = lastUserIndex();
    if (idx === -1) return;
    // Drop every message after the last user message (the old answer).
    while (messages.length > idx + 1) {
      var removed = messages.pop();
      if (removed.el && removed.el.parentNode) removed.el.parentNode.removeChild(removed.el);
    }
    startStream(buildApiMessages(idx));
  }

  function autoSize() {
    var max = parseFloat(getComputedStyle(promptEl).maxHeight) || 160;
    promptEl.style.height = 'auto';
    promptEl.style.height = Math.min(promptEl.scrollHeight + 2, max) + 'px';
  }

  sendBtn.addEventListener('click', send);
  stopBtn.addEventListener('click', stop);
  regenBtn.addEventListener('click', regenerate);
  promptEl.addEventListener('input', function () {
    promptEl.removeAttribute('aria-invalid');
    autoSize();
  });
  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing && e.keyCode !== 229) {
      e.preventDefault();
      send();
    }
  });
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && isStreaming()) {
      e.preventDefault();
      stop();
    }
  });
  window.addEventListener('beforeunload', function () {
    if (current && current.handle && typeof current.handle.abort === 'function') {
      try { current.handle.abort(); } catch (e) {}
    }
  });

  renderEmptyHint();
  updateControls();
  setStatus('idle', 'Ready');
  promptEl.focus();
})();
</script>
</body>
</html>
