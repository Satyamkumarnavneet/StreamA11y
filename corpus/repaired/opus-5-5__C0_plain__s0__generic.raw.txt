<script>
(function () {
  'use strict';

  var promptEl = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var historyEl = document.getElementById('history');
  var emptyEl = document.getElementById('empty');
  var formEl = document.getElementById('composer');
  var statusEl = document.getElementById('status');
  var statusTextEl = document.getElementById('status-text');
  var announcerEl = document.getElementById('announcer');
  var alerterEl = document.getElementById('alerter');

  // Conversation model. Each entry: { role, content, status, el, textEl, note }
  // status for assistant: 'streaming' | 'done' | 'stopped' | 'error'
  var conversation = [];
  var msgCounter = 0;

  // The generation currently allowed to modify the UI (or null).
  var current = null;

  var announceTimers = new Map();
  function speak(region, text) {
    // Clear, then set after a short delay so repeated identical messages are re-announced.
    clearTimeout(announceTimers.get(region));
    region.textContent = '';
    announceTimers.set(region, setTimeout(function () { region.textContent = text; }, 60));
  }
  function announce(text) { speak(announcerEl, text); }
  function alertUser(text) { speak(alerterEl, text); }

  function setStatus(state, text) {
    statusEl.setAttribute('data-state', state);
    statusTextEl.textContent = text;
  }

  function isNearBottom() {
    return historyEl.scrollHeight - historyEl.scrollTop - historyEl.clientHeight < 60;
  }

  function scrollToBottom() {
    historyEl.scrollTop = historyEl.scrollHeight;
  }

  function hasUserMessage() {
    for (var i = conversation.length - 1; i >= 0; i--) {
      if (conversation[i].role === 'user') return true;
    }
    return false;
  }

  function updateControls() {
    var busy = current !== null;
    var focused = document.activeElement;
    sendBtn.disabled = busy || promptEl.value.trim() === '';
    stopBtn.disabled = !busy;
    regenBtn.disabled = busy || !hasUserMessage();
    emptyEl.style.display = conversation.length ? 'none' : '';
    // If the focused control just became disabled, keep focus somewhere useful
    // instead of letting it fall back to the document body.
    if (focused && focused.tagName === 'BUTTON' && focused.disabled) {
      if (busy && !stopBtn.disabled) stopBtn.focus();
      else promptEl.focus();
    }
  }

  function renderMessage(msg) {
    var id = 'msg-' + (++msgCounter);

    var wrap = document.createElement('article');
    wrap.className = 'msg ' + msg.role;
    wrap.setAttribute('aria-labelledby', id + '-role');

    // Headings let screen reader users jump between messages.
    var role = document.createElement('h3');
    role.className = 'role';
    role.id = id + '-role';
    role.textContent = msg.role === 'user' ? 'You' : 'Assistant';

    var bubble = document.createElement('div');
    bubble.className = 'bubble';

    var textEl = document.createElement('span');
    textEl.className = 'text';
    textEl.textContent = msg.content;
    bubble.appendChild(textEl);

    if (msg.role === 'assistant') {
      var thinking = document.createElement('span');
      thinking.className = 'thinking';
      thinking.setAttribute('aria-hidden', 'true');
      thinking.textContent = 'Thinking…';
      bubble.appendChild(thinking);

      var cursor = document.createElement('span');
      cursor.className = 'cursor';
      cursor.setAttribute('aria-hidden', 'true');
      cursor.textContent = '▍';
      bubble.appendChild(cursor);
    }

    var note = document.createElement('p');
    note.className = 'note';
    note.style.margin = '0';

    wrap.appendChild(role);
    wrap.appendChild(bubble);
    wrap.appendChild(note);

    msg.el = wrap;
    msg.textEl = textEl;
    msg.note = note;
    historyEl.appendChild(wrap);
  }

  function addMessage(role, content, status) {
    var msg = { role: role, content: content, status: status || 'done' };
    conversation.push(msg);
    renderMessage(msg);
    applyMessageStatus(msg);
    return msg;
  }

  function applyMessageStatus(msg, errorText) {
    if (msg.role !== 'assistant') return;
    var streaming = msg.status === 'streaming';
    msg.el.classList.toggle('streaming', streaming);
    msg.el.classList.toggle('empty-answer', msg.content === '');
    if (streaming) msg.el.setAttribute('aria-busy', 'true');
    else msg.el.removeAttribute('aria-busy');
    msg.note.className = 'note';
    msg.note.textContent = '';
    if (msg.status === 'streaming') {
      msg.note.textContent = '';
    } else if (msg.status === 'stopped') {
      msg.note.classList.add('stopped');
      msg.note.textContent = msg.content ? 'Stopped — partial answer' : 'Stopped before any text was received';
    } else if (msg.status === 'error') {
      msg.note.classList.add('error');
      msg.note.textContent = 'Error: ' + (errorText || 'the answer could not be completed');
    } else if (msg.status === 'done' && msg.content === '') {
      msg.note.classList.add('done');
      msg.note.textContent = '(empty answer)';
    }
  }

  function removeMessagesAfter(index) {
    while (conversation.length > index + 1) {
      var m = conversation.pop();
      if (m.el && m.el.parentNode) m.el.parentNode.removeChild(m.el);
    }
  }

  // Failed or empty assistant answers are not sent back as context.
  function buildApiMessages() {
    var out = [];
    for (var i = 0; i < conversation.length; i++) {
      var m = conversation[i];
      if (m.role === 'assistant' && (m.status === 'error' || m.status === 'streaming' || m.content === '')) continue;
      out.push({ role: m.role, content: m.content });
    }
    return out;
  }

  function describeError(err) {
    if (!err) return 'Unknown error';
    if (typeof err === 'string') return err;
    if (err.message) return String(err.message);
    try { return String(err); } catch (e) { return 'Unknown error'; }
  }

  function finishGeneration(gen, status, errorText) {
    if (current !== gen || gen.ended) return;
    gen.ended = true;
    current = null;
    gen.msg.status = status;
    applyMessageStatus(gen.msg, errorText);
    if (status === 'done') {
      setStatus('finished', 'Finished');
      announce(gen.msg.content
        ? 'Assistant finished. ' + gen.msg.content
        : 'Assistant finished with an empty answer.');
    } else if (status === 'stopped') {
      setStatus('stopped', 'Stopped');
      announce(gen.msg.content
        ? 'Stopped. The partial answer was kept in the conversation.'
        : 'Stopped before any text was received.');
    } else {
      setStatus('error', 'Error: ' + errorText);
      alertUser('Error: ' + errorText);
    }
    updateControls();
  }

  function startGeneration() {
    var apiMessages = buildApiMessages();
    var msg = addMessage('assistant', '', 'streaming');
    var gen = { msg: msg, handle: null, ended: false };
    current = gen;
    setStatus('generating', 'Generating…');
    announce('Assistant is generating an answer. Press Escape or the Stop button to stop.');
    updateControls();
    scrollToBottom();

    var sdk = window.ChatSDK;
    if (!sdk || typeof sdk.streamChat !== 'function') {
      finishGeneration(gen, 'error', 'Chat service is not available');
      return;
    }

    try {
      var handle = sdk.streamChat(apiMessages, {
        onToken: function (text) {
          if (current !== gen || gen.ended) return; // ignore late/in-flight tokens after stop
          if (text === undefined || text === null) return;
          var stick = isNearBottom();
          msg.content += String(text);
          msg.textEl.textContent = msg.content;
          msg.el.classList.toggle('empty-answer', msg.content === '');
          if (stick) scrollToBottom();
        },
        onDone: function () {
          finishGeneration(gen, 'done');
        },
        onError: function (error) {
          finishGeneration(gen, 'error', describeError(error));
        }
      });
      gen.handle = handle || null;
    } catch (e) {
      finishGeneration(gen, 'error', describeError(e));
    }
  }

  function send() {
    if (current) return;
    var text = promptEl.value;
    if (text.trim() === '') return;
    promptEl.value = '';
    addMessage('user', text, 'done');
    startGeneration();
    promptEl.focus();
  }

  function stop() {
    var gen = current;
    if (!gen) return;
    // Detach first so any in-flight token arriving after abort is ignored.
    finishGeneration(gen, 'stopped');
    try {
      if (gen.handle && typeof gen.handle.abort === 'function') gen.handle.abort();
    } catch (e) {
      // The stream is already detached from the UI; nothing else to do.
    }
    promptEl.focus();
  }

  function regenerate() {
    if (current) return;
    var lastUser = -1;
    for (var i = conversation.length - 1; i >= 0; i--) {
      if (conversation[i].role === 'user') { lastUser = i; break; }
    }
    if (lastUser === -1) return;
    removeMessagesAfter(lastUser);
    startGeneration();
  }

  formEl.addEventListener('submit', function (e) {
    e.preventDefault();
    send();
  });
  stopBtn.addEventListener('click', stop);
  regenBtn.addEventListener('click', regenerate);
  promptEl.addEventListener('input', updateControls);
  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });
  // Escape stops the current answer from anywhere on the page.
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && current && !e.isComposing) {
      e.preventDefault();
      stop();
    }
  });

  updateControls();
  promptEl.focus();
})();
</script>
</body>
</html>
