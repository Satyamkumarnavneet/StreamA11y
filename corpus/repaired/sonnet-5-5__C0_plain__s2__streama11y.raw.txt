<script>
(function () {
  'use strict';

  var logEl = document.getElementById('log');
  var promptEl = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var statusEl = document.getElementById('status');
  var statusText = document.getElementById('status-text');
  var hintEl = document.getElementById('hint');

  // history items: { role, content, status, error, el, bubble, tag }
  // status (assistant only): 'streaming' | 'done' | 'stopped' | 'error'
  var history = [];
  var current = null; // the active request, or null when nothing is streaming

  function setStatus(state, text) {
    statusEl.setAttribute('data-state', state);
    statusText.textContent = text;
  }

  // Updates button enabled states. If the control that currently has keyboard
  // focus is about to be disabled, focus is moved to the message box first so
  // it never falls back to <body>.
  function updateControls() {
    var busy = current !== null;
    var hasUser = history.some(function (m) { return m.role === 'user'; });
    var want = [
      [sendBtn, busy],
      [stopBtn, !busy],
      [regenBtn, !hasUser]
    ];

    // Enable first (so a focus target is always valid)
    want.forEach(function (w) { if (!w[1]) w[0].disabled = false; });

    var active = document.activeElement;
    var losing = want.some(function (w) { return w[1] && w[0] === active; });
    if (losing) {
      promptEl.focus();
    }

    want.forEach(function (w) { if (w[1]) w[0].disabled = true; });

    // Safety net: if focus ended up on body anyway, restore it to the prompt
    if (document.activeElement === document.body || document.activeElement === null) {
      if (losing) promptEl.focus();
    }

    hintEl.textContent = busy ? 'Generating…' : '';
  }

  function isNearBottom() {
    return logEl.scrollHeight - logEl.scrollTop - logEl.clientHeight < 60;
  }
  function scrollToBottom() { logEl.scrollTop = logEl.scrollHeight; }

  function buildMessageEl(msg) {
    var wrap = document.createElement('div');
    wrap.className = 'msg ' + msg.role;
    var role = document.createElement('div');
    role.className = 'role';
    role.textContent = msg.role === 'user' ? 'You' : 'Assistant';
    var bubble = document.createElement('div');
    bubble.className = 'bubble';
    bubble.textContent = msg.content;
    wrap.appendChild(role);
    wrap.appendChild(bubble);
    var tag = null;
    if (msg.role === 'assistant') {
      tag = document.createElement('div');
      wrap.appendChild(tag);
    }
    msg.el = wrap;
    msg.bubble = bubble;
    msg.tag = tag;
    refreshMessage(msg);
    return wrap;
  }

  function refreshMessage(msg) {
    if (msg.role !== 'assistant') return;
    var cls = 'msg assistant';
    if (msg.status === 'streaming') cls += ' streaming';
    if (msg.status === 'error') cls += ' error';
    msg.el.className = cls;
    var tag = msg.tag;
    tag.className = 'tag';
    tag.textContent = '';
    if (msg.status === 'stopped') {
      tag.classList.add('stopped');
      tag.textContent = msg.content ? '■ Stopped — partial answer kept' : '■ Stopped before any text arrived';
    } else if (msg.status === 'error') {
      tag.classList.add('error');
      tag.textContent = '⚠ Error: ' + (msg.error || 'Something went wrong') +
        (msg.content ? ' (partial answer shown)' : '');
    } else if (msg.status === 'done') {
      tag.classList.add('done');
      tag.textContent = msg.content ? '✓ Finished' : '✓ Finished (empty answer)';
    }
  }

  function renderAll() {
    logEl.textContent = '';
    if (history.length === 0) {
      var empty = document.createElement('div');
      empty.className = 'empty';
      empty.textContent = 'Start the conversation by sending a message.';
      logEl.appendChild(empty);
      return;
    }
    history.forEach(function (m) { logEl.appendChild(buildMessageEl(m)); });
    scrollToBottom();
  }

  function addMessage(role, content, status) {
    if (history.length === 0) logEl.textContent = ''; // remove placeholder
    var msg = { role: role, content: content, status: status || null, error: null };
    history.push(msg);
    logEl.appendChild(buildMessageEl(msg));
    scrollToBottom();
    return msg;
  }

  function errText(e) {
    if (!e) return 'Unknown error';
    if (typeof e === 'string') return e;
    return e.message || String(e);
  }

  function endRequest(req, status, errorMessage) {
    if (current !== req) return;
    req.closed = true;
    current = null;
    req.msg.status = status;
    req.msg.error = errorMessage || null;
    refreshMessage(req.msg);
    if (status === 'done') setStatus('done', 'Finished');
    else if (status === 'error') setStatus('error', 'Error: ' + errorMessage);
    updateControls();
  }

  // Send the current history (ending with a user message) and stream a reply.
  function startStream() {
    var apiMessages = history
      .filter(function (m) { return m.content.trim() !== ''; })
      .map(function (m) { return { role: m.role, content: m.content }; });

    var msg = addMessage('assistant', '', 'streaming');
    var req = { msg: msg, handle: null, closed: false };
    current = req;
    setStatus('generating', 'Generating answer…');
    updateControls();

    try {
      if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
        throw new Error('Chat service is not available');
      }
      req.handle = window.ChatSDK.streamChat(apiMessages, {
        onToken: function (text) {
          // Ignore late chunks from aborted / superseded streams.
          if (req.closed || current !== req) return;
          var stick = isNearBottom();
          msg.content += text;
          msg.bubble.textContent = msg.content;
          if (stick) scrollToBottom();
        },
        onDone: function () {
          if (req.closed || current !== req) return;
          endRequest(req, 'done');
        },
        onError: function (e) {
          if (req.closed || current !== req) return;
          endRequest(req, 'error', errText(e));
        }
      });
    } catch (e) {
      endRequest(req, 'error', errText(e));
    }
  }

  // Abort the active request (if any) without touching the UI state.
  function abortCurrent() {
    var req = current;
    if (!req) return null;
    req.closed = true; // from here on, late callbacks are ignored
    current = null;
    try { if (req.handle && typeof req.handle.abort === 'function') req.handle.abort(); } catch (e) {}
    return req;
  }

  function send() {
    if (current) return;
    var text = promptEl.value.trim();
    if (!text) { promptEl.focus(); return; }
    promptEl.value = '';
    autosize();
    addMessage('user', text);
    startStream();
  }

  function stop() {
    var req = abortCurrent();
    if (!req) return;
    req.msg.status = 'stopped';
    refreshMessage(req.msg);
    setStatus('stopped', req.msg.content ? 'Stopped — partial answer kept' : 'Stopped');
    updateControls();
  }

  function regenerate() {
    var lastUser = -1;
    for (var i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') { lastUser = i; break; }
    }
    if (lastUser < 0) return;
    abortCurrent(); // if still streaming, discard that answer
    history.length = lastUser + 1; // drop the previous answer(s) to the last user message
    renderAll();
    startStream();
  }

  function autosize() {
    promptEl.style.height = 'auto';
    promptEl.style.height = Math.min(promptEl.scrollHeight, 160) + 'px';
  }

  sendBtn.addEventListener('click', send);
  stopBtn.addEventListener('click', stop);
  regenBtn.addEventListener('click', regenerate);
  promptEl.addEventListener('input', autosize);
  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });

  renderAll();
  setStatus('idle', 'Ready');
  updateControls();
  promptEl.focus();
})();
</script>
</body>
</html>
