<script>
(function () {
  "use strict";

  var conversationEl = document.getElementById("conversation");
  var statusEl = document.getElementById("status");
  var promptEl = document.getElementById("prompt");
  var formEl = document.getElementById("composer");
  var sendBtn = document.getElementById("send");
  var stopBtn = document.getElementById("stop");
  var regenBtn = document.getElementById("regenerate");

  // Each entry: { role: 'user'|'assistant', content: string, state: 'complete'|'streaming'|'stopped'|'error', error?: string }
  var conversation = [];
  // The currently streaming run, or null when idle.
  var activeRun = null;

  function setStatus(kind, text) {
    statusEl.className = kind;
    statusEl.textContent = text;
  }

  function hasUserMessage() {
    return conversation.some(function (m) { return m.role === "user"; });
  }

  // Controls are never natively disabled, so a focused control never drops focus to <body>.
  // Unavailable controls are marked with aria-disabled and ignore activation.
  function setAvailable(btn, ok) {
    btn.setAttribute("aria-disabled", ok ? "false" : "true");
    btn.classList.toggle("unavailable", !ok);
  }

  function isAvailable(btn) {
    return btn.getAttribute("aria-disabled") !== "true";
  }

  function updateControls() {
    var focused = document.activeElement;
    var busy = activeRun !== null;

    setAvailable(sendBtn, !busy);
    setAvailable(stopBtn, busy);
    setAvailable(regenBtn, !busy && hasUserMessage());

    // If the user's focus is on a control that just became unavailable, move it to the
    // prompt so keyboard users keep a predictable place in the page.
    if ((focused === sendBtn || focused === stopBtn || focused === regenBtn) && !isAvailable(focused)) {
      promptEl.focus();
    }
  }

  function errorMessage(err) {
    if (typeof err === "string" && err) return err;
    if (err && typeof err.message === "string" && err.message) return err.message;
    return "Unknown error";
  }

  function render() {
    conversationEl.innerHTML = "";
    if (conversation.length === 0) {
      var empty = document.createElement("div");
      empty.className = "empty";
      empty.textContent = "Send a message to start the conversation.";
      conversationEl.appendChild(empty);
      return;
    }

    conversation.forEach(function (m) {
      var wrap = document.createElement("div");
      wrap.className = "msg " + m.role;

      var role = document.createElement("div");
      role.className = "role";
      role.textContent = m.role === "user" ? "You" : "Assistant";
      wrap.appendChild(role);

      var body = document.createElement("div");
      body.className = "body";
      if (m.state === "streaming") body.classList.add("cursor");
      if (m.content) {
        body.textContent = m.content;
      } else if (m.state === "streaming") {
        body.textContent = "Generating…";
        body.style.color = "var(--muted)";
      } else if (m.role === "assistant") {
        body.textContent = m.state === "stopped" ? "(stopped before any text was shown)" : "(no response)";
        body.style.color = "var(--muted)";
      }
      wrap.appendChild(body);

      if (m.role === "assistant" && m.state === "stopped") {
        var stoppedNote = document.createElement("div");
        stoppedNote.className = "note stopped";
        stoppedNote.textContent = "Stopped";
        wrap.appendChild(stoppedNote);
      } else if (m.role === "assistant" && m.state === "error") {
        var errNote = document.createElement("div");
        errNote.className = "note error";
        errNote.textContent = "Error: " + (m.error || "request failed");
        wrap.appendChild(errNote);
      }

      conversationEl.appendChild(wrap);
    });

    conversationEl.scrollTop = conversationEl.scrollHeight;
  }

  // Convert conversation entries into the payload shape the SDK expects.
  // Empty assistant entries are dropped so the server never sees blank turns.
  function toPayload(list) {
    var out = [];
    list.forEach(function (m) {
      if (m.role === "user") {
        out.push({ role: "user", content: m.content });
      } else if (m.role === "assistant" && m.content) {
        out.push({ role: "assistant", content: m.content });
      }
    });
    return out;
  }

  function finishRun(run, kind, text) {
    run.finished = true;
    run.target.state = kind === "finished" ? "complete" : "error";
    if (kind !== "finished") run.target.error = text;
    activeRun = null;
    setStatus(kind, kind === "finished" ? "Finished" : "Error: " + text);
    updateControls();
    render();
  }

  function startRun(payload, target) {
    var run = { aborted: false, finished: false, handle: null, target: target };
    activeRun = run;
    target.state = "streaming";
    setStatus("generating", "Generating…");
    updateControls();
    render();

    if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== "function") {
      finishRun(run, "error", "Chat API is not available on this page.");
      return;
    }

    var callbacks = {
      onToken: function (text) {
        // Ignore late chunks that were in flight when the user pressed Stop.
        if (run.aborted || run.finished) return;
        target.content += String(text == null ? "" : text);
        render();
      },
      onDone: function () {
        if (run.aborted || run.finished) return;
        finishRun(run, "finished");
      },
      onError: function (err) {
        if (run.aborted || run.finished) return;
        finishRun(run, "error", errorMessage(err));
      }
    };

    try {
      run.handle = window.ChatSDK.streamChat(payload, callbacks);
    } catch (e) {
      if (!run.aborted && !run.finished) {
        finishRun(run, "error", errorMessage(e));
      }
    }
  }

  function sendMessage() {
    var text = promptEl.value.trim();
    if (!text || activeRun !== null) return;

    conversation.push({ role: "user", content: text, state: "complete" });
    promptEl.value = "";

    var target = { role: "assistant", content: "", state: "streaming" };
    conversation.push(target);
    startRun(toPayload(conversation.slice(0, -1)), target);
  }

  function stopStream() {
    var run = activeRun;
    if (!run) return;

    run.aborted = true;
    try {
      if (run.handle && typeof run.handle.abort === "function") run.handle.abort();
    } catch (e) {
      // Ignore abort failures; the local state is already marked stopped.
    }

    run.target.state = "stopped";
    activeRun = null;
    setStatus("stopped", "Stopped");
    updateControls();
    render();
  }

  function regenerate() {
    if (activeRun !== null) return;

    var lastUserIdx = -1;
    for (var i = conversation.length - 1; i >= 0; i--) {
      if (conversation[i].role === "user") { lastUserIdx = i; break; }
    }
    if (lastUserIdx === -1) return;

    // Drop everything after the last user message (the old answer and any trailing entries).
    conversation = conversation.slice(0, lastUserIdx + 1);

    var target = { role: "assistant", content: "", state: "streaming" };
    conversation.push(target);
    startRun(toPayload(conversation.slice(0, -1)), target);
  }

  formEl.addEventListener("submit", function (e) {
    e.preventDefault();
    sendMessage();
  });

  promptEl.addEventListener("keydown", function (e) {
    if (e.key === "Enter" && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      sendMessage();
    }
  });

  stopBtn.addEventListener("click", function () {
    if (!isAvailable(stopBtn)) return;
    stopStream();
  });

  regenBtn.addEventListener("click", function () {
    if (!isAvailable(regenBtn)) return;
    regenerate();
  });

  setStatus("ready", "Ready");
  updateControls();
  render();
})();
</script>
</body>
</html>
