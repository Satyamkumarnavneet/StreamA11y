    <div class="chat-container">
        <header class="chat-header">
            <h1>Chat Assistant</h1>
            <div class="status-indicator" role="status" aria-live="polite" aria-atomic="true">
                <div class="status-dot" id="statusDot" aria-hidden="true"></div>
                <span id="statusText">Ready</span>
            </div>

    <script>
        (function() {
            const messagesContainer = document.getElementById('messagesContainer');
            const emptyState = document.getElementById('emptyState');
            const promptInput = document.getElementById('prompt');
            const sendButton = document.getElementById('send');
            const stopButton = document.getElementById('stop');
            const regenerateButton = document.getElementById('regenerate');
            const statusDot = document.getElementById('statusDot');
            const statusText = document.getElementById('statusText');
            const liveAnnouncements = document.getElementById('liveAnnouncements');

            let conversationHistory = [];
            let currentHandle = null;
            let isGenerating = false;
            let currentAssistantElement = null;

            function announce(message) {
                liveAnnouncements.textContent = '';
                setTimeout(function() {
                    liveAnnouncements.textContent = message;
                }, 100);
            }

            function setStatus(status, text) {
                statusDot.className = 'status-dot';
                if (status !== 'ready') {
                    statusDot.classList.add(status);
                }
                statusText.textContent = text;
            }

            function updateButtonStates() {
                const canSend = !isGenerating && promptInput.value.trim() !== '';
                const canStop = isGenerating;
                const canRegenerate = !isGenerating && conversationHistory.length > 0 && conversationHistory[conversationHistory.length - 1].role === 'assistant';

                sendButton.disabled = !canSend;
                sendButton.setAttribute('aria-disabled', !canSend);
                
                stopButton.disabled = !canStop;
                stopButton.setAttribute('aria-disabled', !canStop);
                
                regenerateButton.disabled = !canRegenerate;
                regenerateButton.setAttribute('aria-disabled', !canRegenerate);
            }

            function scrollToBottom() {
                messagesContainer.scrollTop = messagesContainer.scrollHeight;
            }

            function createMessageElement(role, content, status) {
                const messageDiv = document.createElement('article');
                messageDiv.className = `message ${role}`;
                messageDiv.setAttribute('aria-label', role === 'user' ? 'Your message' : 'Assistant response');
                if (status) {
                    messageDiv.classList.add(status);
                }

                const labelDiv = document.createElement('div');
                labelDiv.className = 'message-label';
                labelDiv.setAttribute('aria-hidden', 'true');
                labelDiv.textContent = role === 'user' ? 'You' : 'Assistant';

                const contentDiv = document.createElement('div');
                contentDiv.className = 'message-content';
                contentDiv.textContent = content;

                messageDiv.appendChild(labelDiv);
                messageDiv.appendChild(contentDiv);

                return messageDiv;
            }

            function addUserMessage(content) {
                if (emptyState.parentNode) {
                    emptyState.remove();
                }
                const messageElement = createMessageElement('user', content);
                messagesContainer.appendChild(messageElement);
                scrollToBottom();
            }

            function startAssistantMessage() {
                if (emptyState.parentNode) {
                    emptyState.remove();
                }
                const messageDiv = document.createElement('article');
                messageDiv.className = 'message assistant generating';
                messageDiv.setAttribute('aria-label', 'Assistant response, generating');
                messageDiv.setAttribute('aria-busy', 'true');

                const labelDiv = document.createElement('div');
                labelDiv.className = 'message-label';
                labelDiv.setAttribute('aria-hidden', 'true');
                labelDiv.textContent = 'Assistant';

                const contentDiv = document.createElement('div');
                contentDiv.className = 'message-content';
                
                const cursor = document.createElement('span');
                cursor.className = 'cursor';
                cursor.setAttribute('aria-hidden', 'true');
                contentDiv.appendChild(cursor);

                messageDiv.appendChild(labelDiv);
                messageDiv.appendChild(contentDiv);

                messagesContainer.appendChild(messageDiv);
                currentAssistantElement = messageDiv;
                scrollToBottom();

                return messageDiv;
            }

            function updateAssistantMessage(text) {
                if (currentAssistantElement) {
                    const contentDiv = currentAssistantElement.querySelector('.message-content');
                    const cursor = contentDiv.querySelector('.cursor');
                    
                    if (cursor) {
                        contentDiv.insertBefore(document.createTextNode(text), cursor);
                    } else {
                        contentDiv.textContent += text;
                    }
                    scrollToBottom();
                }
            }

            function finishAssistantMessage(status, statusMessage) {
                if (currentAssistantElement) {
                    currentAssistantElement.classList.remove('generating');
                    currentAssistantElement.setAttribute('aria-busy', 'false');
                    
                    const contentDiv = currentAssistantElement.querySelector('.message-content');
                    const cursor = contentDiv.querySelector('.cursor');
                    if (cursor) {
                        cursor.remove();
                    }

                    if (status) {
                        currentAssistantElement.classList.add(status);
                        currentAssistantElement.setAttribute('aria-label', 'Assistant response, ' + status);
                        
                        const statusDiv = document.createElement('div');
                        statusDiv.className = 'message-status';
                        statusDiv.setAttribute('role', 'status');
                        statusDiv.textContent = statusMessage;
                        currentAssistantElement.appendChild(statusDiv);
                    } else {
                        currentAssistantElement.setAttribute('aria-label', 'Assistant response');
                    }
                }
                currentAssistantElement = null;
            }

            function getAssistantContent() {
                if (currentAssistantElement) {
                    const contentDiv = currentAssistantElement.querySelector('.message-content');
                    let text = contentDiv.textContent;
                    return text;
                }
                return '';
            }

            function removeLastAssistantMessage() {
                const messages = messagesContainer.querySelectorAll('.message.assistant');
                if (messages.length > 0) {
                    messages[messages.length - 1].remove();
                }
                if (conversationHistory.length > 0 && conversationHistory[conversationHistory.length - 1].role === 'assistant') {
                    conversationHistory.pop();
                }
            }

            function streamChat(messages) {
                isGenerating = true;
                updateButtonStates();
                setStatus('generating', 'Generating...');
                announce('Generating response');

                startAssistantMessage();

                currentHandle = window.ChatSDK.streamChat(messages, {
                    onToken: function(text) {
                        updateAssistantMessage(text);
                    },
                    onDone: function() {
                        const content = getAssistantContent();
                        conversationHistory.push({ role: 'assistant', content: content });
                        
                        finishAssistantMessage();
                        setStatus('ready', 'Ready');
                        announce('Response complete');
                        isGenerating = false;
                        currentHandle = null;
                        updateButtonStates();
                        promptInput.focus();
                    },
                    onError: function(error) {
                        const content = getAssistantContent();
                        if (content) {
                            conversationHistory.push({ role: 'assistant', content: content });
                        }
                        
                        const errorMessage = error.message || 'Something went wrong';
                        finishAssistantMessage('error', 'Error: ' + errorMessage);
                        setStatus('error', 'Error');
                        announce('Error: ' + errorMessage);
                        isGenerating = false;
                        currentHandle = null;
                        updateButtonStates();
                        promptInput.focus();
                    }
                });
            }

            function sendMessage() {
                const content = promptInput.value.trim();
                if (!content || isGenerating) return;

                addUserMessage(content);
                conversationHistory.push({ role: 'user', content: content });
                promptInput.value = '';
                autoResizeTextarea();
                announce('Message sent');

                streamChat([...conversationHistory]);
                updateButtonStates();
            }

            function stopGeneration() {
                if (currentHandle && isGenerating) {
                    currentHandle.abort();
                    
                    const content = getAssistantContent();
                    if (content) {
                        conversationHistory.push({ role: 'assistant', content: content });
                    }
                    
                    finishAssistantMessage('stopped', 'Stopped by user');
                    setStatus('stopped', 'Stopped');
                    announce('Response stopped');
                    isGenerating = false;
                    currentHandle = null;
                    updateButtonStates();
                    promptInput.focus();
                }
            }

            function regenerate() {
                if (isGenerating || conversationHistory.length === 0) return;

                removeLastAssistantMessage();
                announce('Regenerating response');

                const messagesToSend = [...conversationHistory];
                streamChat(messagesToSend);
            }

            function autoResizeTextarea() {
                promptInput.style.height = 'auto';
                promptInput.style.height = Math.min(promptInput.scrollHeight, 120) + 'px';
            }

            sendButton.addEventListener('click', sendMessage);
            stopButton.addEventListener('click', stopGeneration);
            regenerateButton.addEventListener('click', regenerate);

            promptInput.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    sendMessage();
                }
            });

            promptInput.addEventListener('input', function() {
                autoResizeTextarea();
                updateButtonStates();
            });

            updateButtonStates();
        })();
    </script>
</body>
</html>
