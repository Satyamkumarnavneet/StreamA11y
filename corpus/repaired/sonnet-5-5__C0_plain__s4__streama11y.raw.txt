<script>
(function () {
  'use strict';

  var logEl = document.getElementById('log');
  var promptEl = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var statusEl = document.getElementById('status');
  var statusText = document.getElementById('statusText');

  // history items: { role, content, status?: 'streaming'|'done'|'stopped'|'error', error?: string, bodyEl?, tagEl? }
  var history = [];
  var current = null; // the active streaming session, or null

  function setStatus(state, text) {
    statusEl.setAttribute('data-state', state);
    statusText.textContent = text;
  }

  function isOurButton(el) {
    return el === sendBtn || el === stopBtn || el === regenBtn;
  }

  // Update enabled/disabled states. If the control that currently has focus
  // becomes disabled, move focus to a sensible enabled control so keyboard
  // focus is never dropped to <body>.
  function updateControls() {
    var before = document.activeElement;
    var streaming = current !== null;
    var hasUser = history.some(function (m) { return m.role === 'user'; });
    sendBtn.disabled = streaming;
    stopBtn.disabled = !streaming;
    regenBtn.disabled = streaming || !hasUser;

    if (isOurButton(before) && before.disabled) {
      var target;
      if (streaming) {
        target = stopBtn;            // Stop is the natural control while generating
      } else if (before === stopBtn && !regenBtn.disabled && false) {
        target = regenBtn;
      } else {
        target = promptEl;           // ready for the next message
      }
      target.focus();
    }
  }

  function nearBottom() {
    return logEl.scrollHeight - logEl.scrollTop - logEl.clientHeight < 80;
  }
  function scrollToBottom() { logEl.scrollTop = logEl.scrollHeight; }

  function tagFor(msg) {
    if (msg.status === 'stopped') return { cls: 'stopped', text: 'Stopped — partial answer kept' };
    if (msg.status === 'error') return { cls: 'error', text: 'Error: ' + (msg.error || 'The response failed.') };
    return null;
  }

  function applyTag(msg) {
    var t = tagFor(msg);
    if (!msg.tagEl) return;
    if (t) {
      msg.tagEl.className = 'tag ' + t.cls;
      msg.tagEl.textContent = t.text;
      msg.tagEl.hidden = false;
    } else {
      msg.tagEl.hidden = true;
      msg.tagEl.textContent = '';
    }
  }

  function applyMsgClasses(msg) {
    if (!msg.wrapEl) return;
    msg.wrapEl.className = 'msg ' + msg.role +
      (msg.status === 'streaming' ? ' streaming' : '') +
      (msg.status === 'error' ? ' error' : '');
  }

  function render() {
    logEl.textContent = '';
    if (!history.length) {
      var empty = document.createElement('div');
      empty.className = 'empty';
      empty.textContent = 'Send a message to start the conversation.';
      logEl.appendChild(empty);
      return;
    }
    history.forEach(function (msg) {
      var wrap = document.createElement('div');
      var who = document.createElement('div');
      who.className = 'who';
      who.textContent = msg.role === 'user' ? 'You' : 'Assistant';
      var bubble = document.createElement('div');
      bubble.className = 'bubble';
      bubble.textContent = msg.content;
      var tag = document.createElement('div');
      tag.hidden = true;
      wrap.appendChild(who);
      wrap.appendChild(bubble);
      wrap.appendChild(tag);
      logEl.appendChild(wrap);
      msg.wrapEl = wrap;
      msg.bodyEl = bubble;
      msg.tagEl = tag;
      applyMsgClasses(msg);
      applyTag(msg);
    });
    scrollToBottom();
  }

  function buildPayload() {
    return history
      .filter(function (m) { return m.content && m.content.length > 0; })
      .map(function (m) { return { role: m.role, content: m.content }; });
  }

  function finishSession(session, status, statusState, statusMsg, errText) {
    if (!session.active) return;
    session.active = false;
    if (current === session) current = null;
    session.msg.status = status;
    if (errText) session.msg.error = errText;
    applyMsgClasses(session.msg);
    applyTag(session.msg);
    setStatus(statusState, statusMsg);
    updateControls();
    if (nearBottom()) scrollToBottom();
  }

  function startStream() {
    var payload = buildPayload();
    var msg = { role: 'assistant', content: '', status: 'streaming' };
    history.push(msg);
    var session = { msg: msg, active: true, handle: null };
    current = session;
    render();
    setStatus('generating', 'Assistant is generating…');
    updateControls();

    function onToken(text) {
      // Ignore late chunks that arrive after stop/done/error or from an older stream.
      if (!session.active || current !== session) return;
      var chunk = text == null ? '' : String(text);
      if (!chunk) return;
      var stick = nearBottom();
      msg.content += chunk;
      msg.bodyEl.textContent = msg.content;
      if (stick) scrollToBottom();
    }
    function onDone() {
      finishSession(session, 'done', 'finished', 'Finished');
    }
    function onError(err) {
      var text = (err && err.message) ? err.message : (err ? String(err) : 'Unknown error');
      finishSession(session, 'error', 'error', 'Error: ' + text, text);
    }

    try {
      if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
        throw new Error('Chat service is not available.');
      }
      session.handle = window.ChatSDK.streamChat(payload, {
        onToken: onToken,
        onDone: onDone,
        onError: onError
      });
    } catch (e) {
      onError(e);
    }
  }

  function send() {
    if (current) return;
    var text = promptEl.value.trim();
    if (!text) { promptEl.focus(); return; }
    history.push({ role: 'user', content: text });
    promptEl.value = '';
    autosize();
    startStream();
  }

  function stop() {
    var session = current;
    if (!session) return;
    // Mark stopped first so any in-flight chunk is ignored.
    session.active = false;
    current = null;
    try {
      if (session.handle && typeof session.handle.abort === 'function') session.handle.abort();
    } catch (e) { /* abort failures should not break the UI */ }
    session.msg.status = 'stopped';
    applyMsgClasses(session.msg);
    applyTag(session.msg);
    setStatus('stopped', 'Stopped');
    updateControls();
  }

  function regenerate() {
    if (current) return;
    var idx = -1;
    for (var i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') { idx = i; break; }
    }
    if (idx < 0) return;
    history.length = idx + 1; // drop the previous answer (if any) to the last user message
    startStream();
  }

  function autosize() {
    promptEl.style.height = 'auto';
    promptEl.style.height = Math.min(promptEl.scrollHeight, 160) + 'px';
  }

  sendBtn.addEventListener('click', send);
  stopBtn.addEventListener('click', stop);
  regenBtn.addEventListener('click', regenerate);
  promptEl.addEventListener('input', autosize);
  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });

  render();
  updateControls();
  promptEl.focus();
})();
</script>
</body>
</html>
