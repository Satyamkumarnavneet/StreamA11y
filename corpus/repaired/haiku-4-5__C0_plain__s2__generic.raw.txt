    <script>
        const messagesContainer = document.getElementById('messages');
        const promptInput = document.getElementById('prompt');
        const sendButton = document.getElementById('send');
        const stopButton = document.getElementById('stop');
        const regenerateButton = document.getElementById('regenerate');
        const statusDiv = document.getElementById('status');

        let messages = [];
        let currentStream = null;
        let isStreaming = false;
        let lastUserMessage = null;

        // Auto-resize textarea
        promptInput.addEventListener('input', () => {
            promptInput.style.height = 'auto';
            promptInput.style.height = Math.min(promptInput.scrollHeight, 100) + 'px';
        });

        // Send message on Enter (Shift+Enter for new line)
        promptInput.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' && !e.shiftKey) {
                e.preventDefault();
                sendMessage();
            }
        });

        sendButton.addEventListener('click', sendMessage);
        stopButton.addEventListener('click', stopStream);
        regenerateButton.addEventListener('click', regenerateMessage);

        function sendMessage() {
            const content = promptInput.value.trim();
            if (!content || isStreaming) return;

            // Add user message
            messages.push({ role: 'user', content });
            lastUserMessage = content;
            addMessageToUI('user', content);

            promptInput.value = '';
            promptInput.style.height = 'auto';
            regenerateButton.style.display = 'none';
            sendButton.disabled = true;

            streamResponse();
        }

        function streamResponse() {
            isStreaming = true;
            stopButton.style.display = 'inline-block';
            setStatus('Generating<span class="spinner"></span><span class="spinner"></span><span class="spinner"></span>', 'generating');

            let assistantContent = '';
            const messageElement = addMessageToUI('assistant', '');

            currentStream = window.ChatSDK.streamChat(messages, {
                onToken: (text) => {
                    if (!isStreaming) return;
                    assistantContent += text;
                    messageElement.querySelector('.message-content').textContent = assistantContent;
                    messagesContainer.scrollTop = messagesContainer.scrollHeight;
                },
                onDone: () => {
                    messages.push({ role: 'assistant', content: assistantContent });
                    isStreaming = false;
                    currentStream = null;
                    stopButton.style.display = 'none';
                    regenerateButton.style.display = 'inline-block';
                    sendButton.disabled = false;
                    setStatus('Response completed', '');
                    promptInput.focus();
                },
                onError: (error) => {
                    isStreaming = false;
                    currentStream = null;
                    stopButton.style.display = 'none';
                    regenerateButton.style.display = 'inline-block';
                    sendButton.disabled = false;
                    setStatus('Error: ' + (error?.message || 'Unknown error'), 'error');
                    promptInput.focus();
                }
            });
        }

        function stopStream() {
            if (currentStream && isStreaming) {
                currentStream.abort();
                isStreaming = false;
                stopButton.style.display = 'none';
                regenerateButton.style.display = 'inline-block';
                sendButton.disabled = false;
                setStatus('Response stopped', 'stopped');
                promptInput.focus();
            }
        }

        function regenerateMessage() {
            if (!lastUserMessage || isStreaming) return;

            // Remove the last assistant message if it exists
            if (messages.length > 0 && messages[messages.length - 1].role === 'assistant') {
                messages.pop();
                const assistantMessages = document.querySelectorAll('.message.assistant');
                if (assistantMessages.length > 0) {
                    assistantMessages[assistantMessages.length - 1].remove();
                }
            }

            regenerateButton.style.display = 'none';
            sendButton.disabled = true;
            streamResponse();
        }

        function addMessageToUI(role, content) {
            // Remove empty state if this is the first message
            const emptyState = messagesContainer.querySelector('.empty-state');
            if (emptyState) {
                emptyState.remove();
            }

            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${role}`;

            const contentDiv = document.createElement('div');
            contentDiv.className = 'message-content';
            contentDiv.textContent = content;

            messageDiv.appendChild(contentDiv);
            messagesContainer.appendChild(messageDiv);
            messagesContainer.scrollTop = messagesContainer.scrollHeight;

            return messageDiv;
        }

        function setStatus(text, className) {
            statusDiv.innerHTML = text;
            statusDiv.className = `status ${className}`.trim();
        }
    </script>
</body>
</html>

## Accessibility Fixes Applied

1. **Semantic HTML & ARIA Labels**: Added `aria-label` attributes to all buttons and the textarea; added `role="log"` and `aria-live="polite"` to the messages container to announce new messages to screen readers.

2. **Status Region**: Wrapped the status div with `role="status"` and `aria-live="assertive"` so status changes (generating, stopped, error) are immediately announced.

3. **Visual Focus Indicators**: Enhanced all interactive elements with visible `:focus` states using a 3px colored box-shadow instead of relying solely on browser defaults.

4. **Border Contrast**: Changed input and button borders from 1px to 2px for better visibility; status text color contrast meets AA standards.

5. **Screen Reader Only Content**: Added `.sr-only` utility class and an associated `<label>` for the textarea (hidden visually but available to screen readers).

6. **Keyboard Navigation**: Ensured all controls are keyboard accessible (tab order, Enter to send, focus management after actions).

7. **Reduced Motion Support**: Added `@media (prefers-reduced-motion: reduce)` to respect user preferences and disable animations.

8. **Responsive Button Layout**: On mobile, buttons stack vertically for easier touch access and better visibility.

9. **Status Messages**: Changed generic labels to descriptive phrases ("Response completed" instead of "Done", "Response stopped" instead of "Stopped") for clarity.

10. **Focus Management**: After streaming completes or an error occurs, focus returns to the prompt input for continuous keyboard-based workflow.
