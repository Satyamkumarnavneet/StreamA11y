<script>
(function () {
  'use strict';

  var formEl = document.getElementById('chat-form');
  var promptEl = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var convEl = document.getElementById('conversation');
  var emptyEl = document.getElementById('empty');
  var statusEl = document.getElementById('status');
  var statusTextEl = document.getElementById('status-text');
  var announcerEl = document.getElementById('announcer');
  var alerterEl = document.getElementById('alerter');

  // Conversation state. Each entry: { role, content, status, el, bubble, noteEl }
  // status for assistant: 'streaming' | 'finished' | 'stopped' | 'error'
  var history = [];

  // The currently active stream (or null). Each stream gets a unique id so
  // callbacks from stale or aborted streams are ignored.
  var active = null;
  var streamCounter = 0;

  var announceTimer = null;
  var alertTimer = null;

  // Clear then set, so repeated identical messages are re-announced.
  function announce(text) {
    clearTimeout(announceTimer);
    announcerEl.textContent = '';
    announceTimer = setTimeout(function () { announcerEl.textContent = text; }, 100);
  }

  function announceError(text) {
    clearTimeout(alertTimer);
    alerterEl.textContent = '';
    alertTimer = setTimeout(function () { alerterEl.textContent = text; }, 100);
  }

  function setStatus(state, text) {
    statusEl.setAttribute('data-state', state);
    statusTextEl.textContent = text;
  }

  function isNearBottom() {
    return convEl.scrollHeight - convEl.scrollTop - convEl.clientHeight < 60;
  }

  function scrollToBottom() {
    convEl.scrollTop = convEl.scrollHeight;
  }

  // aria-disabled (instead of the disabled attribute) keeps buttons focusable,
  // so focus is never lost when a button becomes unavailable while focused.
  function setDisabled(btn, disabled) {
    btn.setAttribute('aria-disabled', disabled ? 'true' : 'false');
  }

  function isDisabled(btn) {
    return btn.getAttribute('aria-disabled') === 'true';
  }

  function updateControls() {
    var streaming = active !== null;
    var hasText = promptEl.value.trim().length > 0;
    setDisabled(sendBtn, streaming || !hasText);
    setDisabled(stopBtn, !streaming);
    setDisabled(regenBtn, streaming || lastUserIndex() === -1);
  }

  function lastUserIndex() {
    for (var i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') return i;
    }
    return -1;
  }

  function renderMessage(entry) {
    if (emptyEl.parentNode) emptyEl.parentNode.removeChild(emptyEl);

    var wrap = document.createElement('article');
    wrap.className = 'msg ' + entry.role;

    var role = document.createElement('h3');
    role.className = 'role';
    role.textContent = entry.role === 'user' ? 'You' : 'Assistant';

    var bubble = document.createElement('div');
    bubble.className = 'bubble';
    bubble.textContent = entry.content; // textContent: never interpret as HTML

    var note = document.createElement('div');
    note.className = 'note';

    wrap.appendChild(role);
    wrap.appendChild(bubble);
    wrap.appendChild(note);
    convEl.appendChild(wrap);

    entry.el = wrap;
    entry.bubble = bubble;
    entry.noteEl = note;
    scrollToBottom();
  }

  function setAssistantStatus(entry, status, noteText) {
    entry.status = status;
    entry.el.classList.toggle('streaming', status === 'streaming');
    entry.el.setAttribute('aria-busy', status === 'streaming' ? 'true' : 'false');
    entry.noteEl.className = 'note ' + status;
    entry.noteEl.textContent = noteText || '';
  }

  // Messages to send to the API: user messages plus completed/stopped
  // assistant answers that have content. Errored or empty answers are skipped.
  function buildPayload() {
    var out = [];
    for (var i = 0; i < history.length; i++) {
      var m = history[i];
      if (m.role === 'user') {
        out.push({ role: 'user', content: m.content });
      } else if (m.status !== 'error' && m.status !== 'streaming' && m.content.length > 0) {
        out.push({ role: 'assistant', content: m.content });
      }
    }
    return out;
  }

  function errorMessage(err) {
    return (err && err.message) ? err.message : String(err || 'Unknown error');
  }

  function startStream(isRegenerate) {
    var payload = buildPayload();

    var entry = { role: 'assistant', content: '', status: 'streaming' };
    history.push(entry);
    renderMessage(entry);
    setAssistantStatus(entry, 'streaming', 'Generating…');

    var id = ++streamCounter;
    var stream = { id: id, entry: entry, handle: null };
    active = stream;
    setStatus('generating', 'Assistant is generating…');
    announce(isRegenerate ? 'Regenerating answer. Assistant is generating…' : 'Assistant is generating…');
    updateControls();

    function isCurrent() {
      return active !== null && active.id === id;
    }

    function finish() {
      if (active && active.id === id) active = null;
      updateControls();
    }

    function fail(err) {
      var msg = errorMessage(err);
      setAssistantStatus(entry, 'error', 'Error: ' + msg);
      setStatus('error', 'Error: ' + msg);
      finish();
      announceError('Error: ' + msg + (entry.content ? '. Partial answer: ' + entry.content : ''));
    }

    try {
      stream.handle = window.ChatSDK.streamChat(payload, {
        onToken: function (text) {
          // Ignore tokens that arrive after abort() or from a stale stream.
          if (!isCurrent()) return;
          if (typeof text !== 'string') text = text == null ? '' : String(text);
          var stick = isNearBottom();
          entry.content += text;
          entry.bubble.textContent = entry.content;
          if (stick) scrollToBottom();
        },
        onDone: function () {
          if (!isCurrent()) return;
          setAssistantStatus(entry, 'finished', 'Finished');
          setStatus('finished', 'Answer finished');
          finish();
          announce('Answer finished. Assistant said: ' + (entry.content || '(empty answer)'));
        },
        onError: function (error) {
          if (!isCurrent()) return;
          fail(error);
        }
      });
    } catch (err) {
      if (isCurrent()) fail(err);
    }
  }

  function stopStream() {
    if (!active) return;
    var stream = active;
    // Mark inactive FIRST so any in-flight onToken arriving after abort is ignored.
    active = null;
    try {
      if (stream.handle && typeof stream.handle.abort === 'function') {
        stream.handle.abort();
      }
    } catch (e) { /* ignore abort failures; UI is already stopped */ }
    var hasText = stream.entry.content.length > 0;
    setAssistantStatus(stream.entry, 'stopped',
      hasText ? 'Stopped — partial answer kept' : 'Stopped before any text was generated');
    setStatus('stopped', 'Generation stopped');
    updateControls();
    announce(hasText
      ? 'Generation stopped. Partial answer kept: ' + stream.entry.content
      : 'Generation stopped before any text was generated.');
  }

  function send() {
    if (active) return;
    var text = promptEl.value.trim();
    if (!text) return;

    var userEntry = { role: 'user', content: text, status: 'sent' };
    history.push(userEntry);
    renderMessage(userEntry);

    promptEl.value = '';
    autoResize();
    startStream(false);
  }

  function regenerate() {
    if (active) return;
    var idx = lastUserIndex();
    if (idx === -1) return;
    // Remove every message after the last user message (its previous answer(s)).
    var removed = history.splice(idx + 1);
    for (var i = 0; i < removed.length; i++) {
      if (removed[i].el && removed[i].el.parentNode) {
        removed[i].el.parentNode.removeChild(removed[i].el);
      }
    }
    startStream(true);
  }

  function autoResize() {
    promptEl.style.height = 'auto';
    promptEl.style.height = Math.min(promptEl.scrollHeight, 200) + 'px';
  }

  formEl.addEventListener('submit', function (e) {
    e.preventDefault();
    if (isDisabled(sendBtn)) return;
    send();
  });

  stopBtn.addEventListener('click', function () {
    if (isDisabled(stopBtn)) return;
    stopStream();
  });

  regenBtn.addEventListener('click', function () {
    if (isDisabled(regenBtn)) return;
    regenerate();
  });

  promptEl.addEventListener('input', function () {
    autoResize();
    updateControls();
  });

  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      if (active) {
        announce('Please wait: the assistant is still generating. Press Escape to stop.');
        return;
      }
      send();
    }
  });

  // Escape stops generation from anywhere in the page.
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && active && !e.isComposing) {
      e.preventDefault();
      stopStream();
    }
  });

  if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
    setStatus('error', 'Error: chat service is unavailable');
    announceError('Error: chat service is unavailable');
  }

  updateControls();
  promptEl.focus();
})();
</script>
</body>
</html>
