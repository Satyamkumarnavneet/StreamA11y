<script>
(function () {
  const conversationEl = document.getElementById('conversation');
  const scrollEl = document.getElementById('chat-scroll');
  const statusEl = document.getElementById('status');
  const promptEl = document.getElementById('prompt');
  const sendBtn = document.getElementById('send');
  const stopBtn = document.getElementById('stop');
  const regenBtn = document.getElementById('regenerate');

  // Conversation history: [{role: 'user'|'assistant', content: string}]
  const history = [];

  // The stream currently in progress, or null.
  let current = null;
  let runCounter = 0;

  function scrollToBottom(force) {
    const nearBottom = scrollEl.scrollHeight - scrollEl.scrollTop - scrollEl.clientHeight < 80;
    if (force || nearBottom) {
      scrollEl.scrollTop = scrollEl.scrollHeight;
    }
  }

  function removeEmptyState() {
    const empty = document.getElementById('empty-state');
    if (empty) empty.remove();
  }

  function createBubble(role, content) {
    const wrap = document.createElement('div');
    wrap.className = 'message ' + role;
    wrap.setAttribute('role', 'group');
    wrap.setAttribute('aria-label', role === 'user' ? 'Your message' : 'Assistant answer');

    const label = document.createElement('div');
    label.className = 'role';
    label.setAttribute('aria-hidden', 'true');
    label.textContent = role === 'user' ? 'You' : 'Assistant';

    const bubble = document.createElement('div');
    bubble.className = 'bubble';
    bubble.textContent = content; // textContent: never interpret text as HTML

    wrap.appendChild(label);
    wrap.appendChild(bubble);
    return { wrap, bubble };
  }

  function appendMessage(role, content) {
    removeEmptyState();
    const { wrap, bubble } = createBubble(role, content);
    conversationEl.appendChild(wrap);
    return { wrap, bubble };
  }

  function renderAll() {
    conversationEl.innerHTML = '';
    if (history.length === 0) {
      const empty = document.createElement('div');
      empty.className = 'empty';
      empty.id = 'empty-state';
      empty.textContent = 'Ask a question to get started.';
      conversationEl.appendChild(empty);
      return;
    }
    for (const m of history) {
      appendMessage(m.role, m.content);
    }
    scrollToBottom(true);
  }

  // kind: 'generating' | 'stopped' | 'finished' | 'error' | '' (idle)
  function setStatus(text, kind) {
    statusEl.textContent = text;
    statusEl.className = kind || '';
    if (kind === 'generating') {
      statusEl.classList.add('generating');
    }
  }

  function hasUserMessage() {
    return history.some(m => m.role === 'user');
  }

  // Use aria-disabled instead of the disabled attribute so focused controls
  // do not drop focus when they become unavailable. Handlers also guard.
  function setDisabled(btn, isDisabled) {
    btn.setAttribute('aria-disabled', isDisabled ? 'true' : 'false');
  }

  function isDisabled(btn) {
    return btn.getAttribute('aria-disabled') === 'true';
  }

  function updateButtons() {
    const busy = current !== null;
    setDisabled(sendBtn, busy);
    setDisabled(stopBtn, !busy);
    setDisabled(regenBtn, busy || !hasUserMessage());
    conversationEl.setAttribute('aria-busy', busy ? 'true' : 'false');
  }

  // Finish the given run: record the answer (if any), update UI and status.
  // Focus is intentionally not moved here, so the user's current control keeps focus.
  function finishRun(run, statusText, kind) {
    run.active = false;
    if (current === run) current = null;

    if (run.text.length > 0) {
      history.push({ role: 'assistant', content: run.text });
    } else {
      // Nothing was shown; remove the empty placeholder bubble.
      run.wrap.remove();
    }

    run.wrap.classList.remove('streaming');
    setStatus(statusText, kind);
    updateButtons();
  }

  function startStream() {
    const run = {
      id: ++runCounter,
      text: '',
      active: true,
      handle: null,
      wrap: null,
      bubble: null
    };

    const placeholder = appendMessage('assistant', '');
    run.wrap = placeholder.wrap;
    run.bubble = placeholder.bubble;
    run.wrap.classList.add('streaming');
    scrollToBottom(true);

    current = run;
    setStatus('Generating…', 'generating');
    updateButtons();

    // Send a snapshot of the history so later edits cannot affect this request.
    const messages = history.map(m => ({ role: m.role, content: m.content }));

    try {
      if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
        throw new Error('Chat service is not available.');
      }

      run.handle = window.ChatSDK.streamChat(messages, {
        onToken(text) {
          // Ignore tokens after the user stopped (in-flight chunks may still arrive).
          if (!run.active) return;
          run.text += String(text);
          run.bubble.textContent = run.text;
          scrollToBottom(false);
        },
        onDone() {
          if (!run.active) return;
          finishRun(run, 'Finished', 'finished');
        },
        onError(error) {
          if (!run.active) return;
          const msg = (error && error.message) ? error.message : String(error || 'Unknown error');
          finishRun(run, 'Error: ' + msg, 'error');
        }
      });
    } catch (err) {
      if (run.active) {
        const msg = (err && err.message) ? err.message : String(err);
        finishRun(run, 'Error: ' + msg, 'error');
      }
    }
  }

  function sendMessage() {
    if (current !== null) return;
    const text = promptEl.value.trim();
    if (!text) return;

    history.push({ role: 'user', content: text });
    appendMessage('user', text);
    promptEl.value = '';
    scrollToBottom(true);
    startStream();
  }

  function stopStream() {
    const run = current;
    if (!run) return;

    // Mark inactive first so no further tokens are applied, then ask the server to stop.
    run.active = false;
    try {
      if (run.handle && typeof run.handle.abort === 'function') {
        run.handle.abort();
      }
    } catch (e) {
      // Ignore abort failures; the local state is already stopped.
    }

    // Keep the partial answer that was already shown.
    finishRun(run, 'Stopped', 'stopped');
  }

  function regenerate() {
    if (current !== null || !hasUserMessage()) return;

    // Drop the last assistant answer, if any, so we answer the last user message again.
    if (history.length > 0 && history[history.length - 1].role === 'assistant') {
      history.pop();
      renderAll();
    }

    // The last message must now be the user message to answer.
    if (history.length === 0 || history[history.length - 1].role !== 'user') {
      setStatus('Idle');
      updateButtons();
      return;
    }

    startStream();
  }

  sendBtn.addEventListener('click', function () {
    if (isDisabled(sendBtn)) return;
    sendMessage();
  });
  stopBtn.addEventListener('click', function () {
    if (isDisabled(stopBtn)) return;
    stopStream();
  });
  regenBtn.addEventListener('click', function () {
    if (isDisabled(regenBtn)) return;
    regenerate();
  });

  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      sendMessage();
    }
  });

  renderAll();
  setStatus('Idle');
  updateButtons();
})();
</script>
</body>
</html>
