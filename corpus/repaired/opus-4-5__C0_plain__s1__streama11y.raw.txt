<body>
    <div class="chat-container">
        <div class="chat-header">Chat Assistant</div>
        <div class="chat-messages" id="messages"></div>
        <div class="status-indicator" id="status"></div>
        <div aria-live="polite" aria-atomic="true" class="sr-only" id="sr-announcer"></div>

    <script>
        (function() {
            const messagesContainer = document.getElementById('messages');
            const promptInput = document.getElementById('prompt');
            const sendButton = document.getElementById('send');
            const stopButton = document.getElementById('stop');
            const regenerateButton = document.getElementById('regenerate');
            const statusIndicator = document.getElementById('status');
            const srAnnouncer = document.getElementById('sr-announcer');

            let conversationHistory = [];
            let currentHandle = null;
            let isGenerating = false;
            let currentAssistantMessageEl = null;
            let currentAssistantContent = '';
            let currentStreamId = 0;

            function announce(message) {
                srAnnouncer.textContent = '';
                void srAnnouncer.offsetWidth;
                srAnnouncer.textContent = message;
            }

            function setStatus(text, type) {
                statusIndicator.textContent = text;
                statusIndicator.className = 'status-indicator';
                if (type) {
                    statusIndicator.classList.add(type);
                }
            }

            function clearStatus() {
                statusIndicator.textContent = '';
                statusIndicator.className = 'status-indicator';
            }

            function updateButtonStates() {
                const hasUserMessage = conversationHistory.some(m => m.role === 'user');
                sendButton.disabled = isGenerating;
                stopButton.disabled = !isGenerating;
                regenerateButton.disabled = isGenerating || !hasUserMessage;
            }

            function manageFocus() {
                if (stopButton.disabled === false) {
                    stopButton.focus();
                } else {
                    promptInput.focus();
                }
            }

            function scrollToBottom() {
                messagesContainer.scrollTop = messagesContainer.scrollHeight;
            }

            function addMessageElement(role, content, isStreaming = false) {
                const messageEl = document.createElement('div');
                messageEl.className = `message ${role}`;
                messageEl.textContent = content;
                if (isStreaming && role === 'assistant') {
                    const cursor = document.createElement('span');
                    cursor.className = 'typing-indicator';
                    messageEl.appendChild(cursor);
                }
                messagesContainer.appendChild(messageEl);
                scrollToBottom();
                return messageEl;
            }

            function updateAssistantMessage(content, isStreaming = false) {
                if (currentAssistantMessageEl) {
                    currentAssistantMessageEl.textContent = content;
                    if (isStreaming) {
                        const cursor = document.createElement('span');
                        cursor.className = 'typing-indicator';
                        currentAssistantMessageEl.appendChild(cursor);
                    }
                    scrollToBottom();
                }
            }

            function renderConversation() {
                messagesContainer.innerHTML = '';
                for (const msg of conversationHistory) {
                    addMessageElement(msg.role, msg.content);
                }
            }

            function startStreaming(messages) {
                isGenerating = true;
                currentStreamId++;
                const thisStreamId = currentStreamId;
                currentAssistantContent = '';
                updateButtonStates();
                manageFocus();
                setStatus('Generating...', 'generating');
                announce('Generating response');

                currentAssistantMessageEl = addMessageElement('assistant', '', true);

                currentHandle = window.ChatSDK.streamChat(messages, {
                    onToken: function(text) {
                        if (thisStreamId !== currentStreamId) {
                            return;
                        }
                        currentAssistantContent += text;
                        updateAssistantMessage(currentAssistantContent, true);
                    },
                    onDone: function() {
                        if (thisStreamId !== currentStreamId) {
                            return;
                        }
                        isGenerating = false;
                        currentHandle = null;
                        updateAssistantMessage(currentAssistantContent, false);
                        conversationHistory.push({
                            role: 'assistant',
                            content: currentAssistantContent
                        });
                        currentAssistantMessageEl = null;
                        updateButtonStates();
                        manageFocus();
                        setStatus('Response complete', 'finished');
                        announce('Response finished');
                        setTimeout(clearStatus, 2000);
                    },
                    onError: function(error) {
                        if (thisStreamId !== currentStreamId) {
                            return;
                        }
                        isGenerating = false;
                        currentHandle = null;
                        
                        if (currentAssistantContent) {
                            updateAssistantMessage(currentAssistantContent, false);
                            conversationHistory.push({
                                role: 'assistant',
                                content: currentAssistantContent
                            });
                        } else if (currentAssistantMessageEl) {
                            currentAssistantMessageEl.remove();
                        }
                        
                        const errorMsg = error && error.message ? error.message : 'An error occurred';
                        const errorEl = addMessageElement('assistant', 'Error: ' + errorMsg);
                        errorEl.classList.add('error');
                        
                        currentAssistantMessageEl = null;
                        updateButtonStates();
                        manageFocus();
                        setStatus('Error: ' + errorMsg, 'error');
                        announce('Response failed: ' + errorMsg);
                    }
                });
            }

            function sendMessage() {
                const content = promptInput.value.trim();
                if (!content || isGenerating) return;

                promptInput.value = '';
                promptInput.style.height = 'auto';

                conversationHistory.push({
                    role: 'user',
                    content: content
                });

                addMessageElement('user', content);

                const messagesToSend = conversationHistory.map(m => ({
                    role: m.role,
                    content: m.content
                }));

                startStreaming(messagesToSend);
            }

            function stopGeneration() {
                if (currentHandle && isGenerating) {
                    currentStreamId++;
                    currentHandle.abort();
                    currentHandle = null;
                    isGenerating = false;

                    updateAssistantMessage(currentAssistantContent, false);
                    
                    if (currentAssistantContent) {
                        conversationHistory.push({
                            role: 'assistant',
                            content: currentAssistantContent
                        });
                    } else if (currentAssistantMessageEl) {
                        currentAssistantMessageEl.remove();
                    }

                    currentAssistantMessageEl = null;
                    updateButtonStates();
                    manageFocus();
                    setStatus('Generation stopped', 'stopped');
                    announce('Response stopped');
                    setTimeout(clearStatus, 2000);
                }
            }

            function regenerate() {
                if (isGenerating) return;

                let lastUserIndex = -1;
                for (let i = conversationHistory.length - 1; i >= 0; i--) {
                    if (conversationHistory[i].role === 'user') {
                        lastUserIndex = i;
                        break;
                    }
                }

                if (lastUserIndex === -1) return;

                conversationHistory = conversationHistory.slice(0, lastUserIndex + 1);

                renderConversation();

                const messagesToSend = conversationHistory.map(m => ({
                    role: m.role,
                    content: m.content
                }));

                startStreaming(messagesToSend);
            }

            sendButton.addEventListener('click', sendMessage);

            promptInput.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    sendMessage();
                }
            });

            promptInput.addEventListener('input', function() {
                this.style.height = 'auto';
                this.style.height = Math.min(this.scrollHeight, 120) + 'px';
            });

            stopButton.addEventListener('click', stopGeneration);

            regenerateButton.addEventListener('click', regenerate);

            updateButtonStates();
        })();
    </script>
</body>
</html>
