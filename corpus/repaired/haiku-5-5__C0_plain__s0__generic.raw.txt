<script>
(function () {
  const logEl = document.getElementById('log');
  const statusEl = document.getElementById('status');
  const promptEl = document.getElementById('prompt');
  const sendBtn = document.getElementById('send');
  const stopBtn = document.getElementById('stop');
  const regenBtn = document.getElementById('regenerate');

  const hasSDK = !!(window.ChatSDK && typeof window.ChatSDK.streamChat === 'function');

  // Conversation history: [{ role: 'user' | 'assistant', content: string }]
  let history = [];
  // Whether an answer is currently streaming
  let busy = false;
  // Handle returned by ChatSDK.streamChat for the active stream
  let activeHandle = null;
  // Incremented whenever a stream is stopped or replaced, so late callbacks are ignored
  let streamId = 0;
  // The assistant entry currently being streamed into
  let pending = null;
  // DOM node showing the text of the pending assistant entry
  let streamingBody = null;
  // DOM node of the pending assistant message (for aria-busy)
  let streamingMsg = null;

  function setStatus(text, kind) {
    statusEl.textContent = text;
    statusEl.className = kind || '';
  }

  function describeError(err) {
    if (!err) return 'unknown error';
    if (typeof err === 'string') return err;
    return err.message || String(err);
  }

  function lastUserIndex() {
    for (let i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') return i;
    }
    return -1;
  }

  function updateControls() {
    sendBtn.disabled = busy || !hasSDK;
    stopBtn.disabled = !busy;
    regenBtn.disabled = busy || !hasSDK || lastUserIndex() < 0;

    // Disabling a focused button drops keyboard focus to <body>.
    // Move focus to the message box so keyboard users stay in place.
    const active = document.activeElement;
    if (active && active.tagName === 'BUTTON' && active.disabled) {
      promptEl.focus();
    }
  }

  function renderAll() {
    logEl.textContent = '';
    streamingBody = null;
    streamingMsg = null;
    if (history.length === 0) {
      const empty = document.createElement('p');
      empty.className = 'empty';
      empty.textContent = 'Send a message to start the conversation.';
      logEl.append(empty);
      return;
    }
    for (const entry of history) {
      const el = document.createElement('article');
      el.className = 'msg ' + entry.role;
      el.setAttribute('aria-label', entry.role === 'user' ? 'Your message' : 'Assistant answer');

      const label = document.createElement('div');
      label.className = 'role';
      label.setAttribute('aria-hidden', 'true');
      label.textContent = entry.role === 'user' ? 'You' : 'Assistant';

      const body = document.createElement('div');
      body.className = 'body';
      body.textContent = entry.content;

      el.append(label, body);
      logEl.append(el);
    }
    logEl.scrollTop = logEl.scrollHeight;
  }

  function addErrorNotice(message) {
    const note = document.createElement('div');
    note.className = 'notice';
    note.setAttribute('role', 'alert');
    note.textContent = message;
    logEl.append(note);
    logEl.scrollTop = logEl.scrollHeight;
  }

  // Ends the current streaming turn and updates the UI.
  // kind: 'done' | 'stopped' | 'error'
  function endStream(kind, message) {
    busy = false;
    activeHandle = null;
    const entry = pending;
    pending = null;

    // Drop an empty placeholder if nothing was produced (stop/error).
    if (entry && !entry.content && kind !== 'done') {
      const idx = history.indexOf(entry);
      if (idx >= 0) history.splice(idx, 1);
    }

    if (streamingMsg) streamingMsg.setAttribute('aria-busy', 'false');
    renderAll();
    if (kind === 'error') addErrorNotice(message);
    setStatus(message, kind);
    updateControls();
  }

  function startStream() {
    const id = ++streamId;

    // Snapshot the history to send (before adding the placeholder).
    const apiMessages = history.map(m => ({ role: m.role, content: m.content }));

    const entry = { role: 'assistant', content: '' };
    history.push(entry);
    pending = entry;
    busy = true;

    renderAll();
    streamingMsg = logEl.querySelector('.msg:last-child');
    streamingBody = logEl.querySelector('.msg:last-child .body');
    if (streamingMsg) streamingMsg.setAttribute('aria-busy', 'true');
    if (streamingBody) streamingBody.textContent = '…';
    setStatus('Assistant is generating…', 'generating');
    updateControls();

    let handle = null;
    try {
      handle = window.ChatSDK.streamChat(apiMessages, {
        onToken(text) {
          // Ignore tokens from stopped/replaced streams (including late in-flight chunks).
          if (id !== streamId || !busy) return;
          const atBottom = logEl.scrollHeight - logEl.scrollTop - logEl.clientHeight < 60;
          entry.content += String(text);
          if (streamingBody) streamingBody.textContent = entry.content;
          if (atBottom) logEl.scrollTop = logEl.scrollHeight;
        },
        onDone() {
          if (id !== streamId || !busy) return;
          endStream('done', 'Finished.');
        },
        onError(err) {
          if (id !== streamId || !busy) return;
          endStream('error', 'Error: ' + describeError(err));
        }
      });
    } catch (err) {
      if (id === streamId && busy) {
        endStream('error', 'Error: ' + describeError(err));
      }
      return;
    }

    // Only keep the handle if the stream is still the active one.
    if (id === streamId && busy) activeHandle = handle;
  }

  function send() {
    if (busy || !hasSDK) return;
    const text = promptEl.value.trim();
    if (!text) return;
    history.push({ role: 'user', content: text });
    promptEl.value = '';
    startStream();
  }

  function stop() {
    if (!busy) return;
    // Invalidate the current stream so any in-flight tokens are ignored.
    streamId++;
    const handle = activeHandle;
    activeHandle = null;
    try {
      if (handle && typeof handle.abort === 'function') handle.abort();
    } catch (e) {
      // Ignore abort failures; the stream is considered stopped locally.
    }
    endStream('stopped', 'Stopped. Partial answer kept.');
  }

  function regenerate() {
    if (busy) return;
    const u = lastUserIndex();
    if (u < 0) return;
    // Remove the previous answer(s) that follow the last user message.
    history.length = u + 1;
    startStream();
  }

  sendBtn.addEventListener('click', send);
  stopBtn.addEventListener('click', stop);
  regenBtn.addEventListener('click', regenerate);

  promptEl.addEventListener('keydown', (e) => {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });

  renderAll();
  if (hasSDK) {
    setStatus('Ready.');
  } else {
    setStatus('Chat service unavailable: window.ChatSDK.streamChat is not defined.', 'error');
  }
  updateControls();
})();
</script>
</body>
</html>
