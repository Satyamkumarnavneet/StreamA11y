<script>
(function () {
  'use strict';

  var promptEl = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var form = document.getElementById('composer');
  var convEl = document.getElementById('conversation');
  var emptyEl = document.getElementById('empty');
  var statusEl = document.getElementById('status');
  var announcerEl = document.getElementById('announcer');

  // Assistant messages carry a status: 'streaming' | 'done' | 'stopped' | 'error'
  var messages = [];

  // The currently active generation (or null). Callbacks from stale
  // generations (late tokens after abort, superseded streams) are ignored.
  var active = null;
  var announceTimer = null;

  function setStatus(state, text) {
    statusEl.setAttribute('data-state', state);
    statusEl.textContent = text;
  }

  // Announce text once to screen readers (clear first so repeats are read).
  function announce(text) {
    if (announceTimer) clearTimeout(announceTimer);
    announcerEl.textContent = '';
    announceTimer = setTimeout(function () {
      announceTimer = null;
      announcerEl.textContent = text;
    }, 50);
  }

  function errorText(err) {
    if (err == null) return 'Unknown error';
    if (typeof err === 'string') return err;
    if (err.message) return String(err.message);
    try { return String(err); } catch (e) { return 'Unknown error'; }
  }

  function isNearBottom() {
    return convEl.scrollHeight - convEl.scrollTop - convEl.clientHeight < 80;
  }

  function scrollToBottom() {
    convEl.scrollTop = convEl.scrollHeight;
  }

  function createMessageElement(msg) {
    var wrap = document.createElement('div');
    wrap.className = 'msg ' + msg.role;

    var role = document.createElement('div');
    role.className = 'role';
    role.textContent = msg.role === 'user' ? 'You' : 'Assistant';

    var content = document.createElement('div');
    content.className = 'content';

    wrap.appendChild(role);
    wrap.appendChild(content);

    if (msg.role === 'assistant') {
      var meta = document.createElement('div');
      meta.className = 'meta';
      wrap.appendChild(meta);
    }
    msg.el = wrap;
    updateMessageElement(msg);
    return wrap;
  }

  function updateMessageElement(msg) {
    var el = msg.el;
    if (!el) return;
    var content = el.querySelector('.content');
    content.textContent = msg.content; // never interpret model output as HTML

    if (msg.role === 'assistant') {
      el.setAttribute('data-status', msg.status);
      el.setAttribute('aria-busy', msg.status === 'streaming' ? 'true' : 'false');
      if (msg.status === 'streaming') {
        var cursor = document.createElement('span');
        cursor.className = 'cursor';
        cursor.setAttribute('aria-hidden', 'true');
        content.appendChild(cursor);
      }
      var meta = el.querySelector('.meta');
      if (msg.status === 'streaming') meta.textContent = 'Generating…';
      else if (msg.status === 'stopped') meta.textContent = msg.content ? 'Stopped — partial answer' : 'Stopped before any text was received';
      else if (msg.status === 'error') meta.textContent = 'Error: ' + (msg.error || 'Unknown error');
      else meta.textContent = '';
    }
  }

  function renderAll() {
    convEl.textContent = '';
    if (messages.length === 0) {
      convEl.appendChild(emptyEl);
    } else {
      for (var i = 0; i < messages.length; i++) {
        convEl.appendChild(createMessageElement(messages[i]));
      }
    }
    scrollToBottom();
  }

  function appendMessage(msg) {
    if (emptyEl.parentNode) emptyEl.parentNode.removeChild(emptyEl);
    convEl.appendChild(createMessageElement(msg));
    scrollToBottom();
  }

  function lastUserIndex() {
    for (var i = messages.length - 1; i >= 0; i--) {
      if (messages[i].role === 'user') return i;
    }
    return -1;
  }

  // Disable a button without dropping keyboard focus to <body>:
  // if it currently has focus, move focus to the message box first.
  function setDisabled(btn, disabled) {
    if (disabled && document.activeElement === btn) {
      promptEl.focus();
    }
    btn.disabled = disabled;
  }

  function updateControls() {
    var streaming = active !== null;
    setDisabled(sendBtn, streaming || promptEl.value.trim() === '');
    setDisabled(stopBtn, !streaming);
    setDisabled(regenBtn, streaming || lastUserIndex() === -1);
  }

  // Build API payload from messages[0..endExclusive). Empty assistant
  // messages are omitted; partial stopped answers are kept.
  function buildHistory(endExclusive) {
    var out = [];
    for (var i = 0; i < endExclusive; i++) {
      var m = messages[i];
      if (m.role === 'assistant' && !m.content) continue;
      out.push({ role: m.role, content: m.content });
    }
    return out;
  }

  function finish(gen, status, err) {
    if (active !== gen) return;
    active = null;
    gen.msg.status = status;
    if (status === 'error') gen.msg.error = errorText(err);
    updateMessageElement(gen.msg);
    if (status === 'done') {
      setStatus('done', 'Finished');
      announce('Assistant: ' + (gen.msg.content || '(empty answer)'));
    } else if (status === 'error') {
      setStatus('error', 'Error: ' + gen.msg.error);
    }
    updateControls();
  }

  function startGeneration() {
    if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
      var failMsg = { role: 'assistant', content: '', status: 'error', error: 'Chat service is unavailable.' };
      messages.push(failMsg);
      appendMessage(failMsg);
      setStatus('error', 'Error: Chat service is unavailable.');
      updateControls();
      return;
    }

    var history = buildHistory(messages.length);
    var msg = { role: 'assistant', content: '', status: 'streaming' };
    messages.push(msg);
    appendMessage(msg);

    var gen = { msg: msg, handle: null };
    active = gen;
    setStatus('generating', 'Generating…');
    updateControls();

    try {
      var handle = window.ChatSDK.streamChat(history, {
        onToken: function (text) {
          if (active !== gen) return; // ignore late tokens after stop / superseded streams
          if (text == null) return;
          var stick = isNearBottom();
          msg.content += String(text);
          updateMessageElement(msg);
          if (stick) scrollToBottom();
        },
        onDone: function () {
          finish(gen, 'done');
        },
        onError: function (error) {
          finish(gen, 'error', error);
        }
      });
      gen.handle = handle || null;
    } catch (e) {
      finish(gen, 'error', e);
    }
  }

  // markStopped=false is used when the message is about to be discarded.
  function stopGeneration(markStopped) {
    var gen = active;
    if (!gen) return;
    active = null; // detach first so any in-flight onToken is ignored
    try {
      if (gen.handle && typeof gen.handle.abort === 'function') gen.handle.abort();
    } catch (e) { /* ignore */ }
    if (markStopped) {
      gen.msg.status = 'stopped';
      updateMessageElement(gen.msg);
      setStatus('stopped', 'Stopped');
      if (gen.msg.content) announce('Partial answer: ' + gen.msg.content);
    }
    updateControls();
  }

  function send() {
    var text = promptEl.value;
    if (active !== null || text.trim() === '') return;
    var userMsg = { role: 'user', content: text };
    messages.push(userMsg);
    appendMessage(userMsg);
    promptEl.value = '';
    autoResize();
    promptEl.focus();
    startGeneration();
  }

  function regenerate() {
    var idx = lastUserIndex();
    if (idx === -1) return;
    if (active !== null) stopGeneration(false);
    messages.length = idx + 1;
    renderAll();
    startGeneration();
  }

  function autoResize() {
    promptEl.style.height = 'auto';
    promptEl.style.height = Math.min(promptEl.scrollHeight, 200) + 'px';
  }

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    send();
  });

  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });

  promptEl.addEventListener('input', function () {
    autoResize();
    updateControls();
  });

  stopBtn.addEventListener('click', function () {
    stopGeneration(true);
  });

  regenBtn.addEventListener('click', function () {
    regenerate();
  });

  updateControls();
  promptEl.focus();
})();
</script>
</body>
</html>
