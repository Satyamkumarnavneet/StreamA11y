    <script>
        (function() {
            const messagesContainer = document.getElementById('messages');
            const promptInput = document.getElementById('prompt');
            const sendButton = document.getElementById('send');
            const stopButton = document.getElementById('stop');
            const regenerateButton = document.getElementById('regenerate');
            const statusIndicator = document.getElementById('status');
            const srAnnouncements = document.getElementById('sr-announcements');

            let conversationHistory = [];
            let currentHandle = null;
            let isStreaming = false;
            let currentAssistantMessageEl = null;
            let currentAssistantContent = '';

            function announce(message) {
                srAnnouncements.textContent = '';
                setTimeout(() => {
                    srAnnouncements.textContent = message;
                }, 100);
            }

            function setStatus(status, text) {
                statusIndicator.className = 'status-indicator';
                if (status) {
                    statusIndicator.classList.add(status);
                }
                statusIndicator.textContent = text;
            }

            function updateButtonStates() {
                const hasMessages = conversationHistory.length > 0;
                const lastMessageIsAssistant = hasMessages && conversationHistory[conversationHistory.length - 1].role === 'assistant';

                sendButton.disabled = isStreaming;
                sendButton.setAttribute('aria-disabled', isStreaming);
                
                stopButton.disabled = !isStreaming;
                stopButton.setAttribute('aria-disabled', !isStreaming);
                
                regenerateButton.disabled = isStreaming || !lastMessageIsAssistant;
                regenerateButton.setAttribute('aria-disabled', isStreaming || !lastMessageIsAssistant);
            }

            function clearEmptyState() {
                const emptyState = messagesContainer.querySelector('.empty-state');
                if (emptyState) {
                    emptyState.remove();
                }
            }

            function createMessageElement(role, content) {
                const messageDiv = document.createElement('article');
                messageDiv.className = `message ${role}`;
                messageDiv.setAttribute('aria-label', role === 'user' ? 'You said' : 'Assistant said');

                const labelDiv = document.createElement('div');
                labelDiv.className = 'message-label';
                labelDiv.setAttribute('aria-hidden', 'true');
                labelDiv.textContent = role === 'user' ? 'You' : 'Assistant';

                const contentDiv = document.createElement('div');
                contentDiv.className = 'message-content';
                contentDiv.textContent = content;

                messageDiv.appendChild(labelDiv);
                messageDiv.appendChild(contentDiv);

                return messageDiv;
            }

            function addUserMessage(content) {
                clearEmptyState();
                const messageEl = createMessageElement('user', content);
                messagesContainer.appendChild(messageEl);
                conversationHistory.push({ role: 'user', content: content });
                scrollToBottom();
            }

            function startAssistantMessage() {
                clearEmptyState();
                currentAssistantContent = '';
                currentAssistantMessageEl = createMessageElement('assistant', '');
                currentAssistantMessageEl.classList.add('streaming');
                messagesContainer.appendChild(currentAssistantMessageEl);
                scrollToBottom();
            }

            function appendToAssistantMessage(text) {
                currentAssistantContent += text;
                if (currentAssistantMessageEl) {
                    const contentEl = currentAssistantMessageEl.querySelector('.message-content');
                    contentEl.textContent = currentAssistantContent;
                    scrollToBottom();
                }
            }

            function finalizeAssistantMessage(addToHistory = true) {
                if (currentAssistantMessageEl) {
                    currentAssistantMessageEl.classList.remove('streaming');
                    if (addToHistory && currentAssistantContent) {
                        conversationHistory.push({ role: 'assistant', content: currentAssistantContent });
                    }
                }
                currentAssistantMessageEl = null;
            }

            function scrollToBottom() {
                messagesContainer.scrollTop = messagesContainer.scrollHeight;
            }

            function startStreaming(messages) {
                isStreaming = true;
                updateButtonStates();
                setStatus('generating', 'Generating...');
                announce('Assistant is generating a response');
                startAssistantMessage();

                currentHandle = window.ChatSDK.streamChat(messages, {
                    onToken: function(text) {
                        appendToAssistantMessage(text);
                    },
                    onDone: function() {
                        isStreaming = false;
                        finalizeAssistantMessage(true);
                        currentHandle = null;
                        setStatus('finished', 'Finished');
                        announce('Assistant finished responding');
                        updateButtonStates();
                        promptInput.focus();
                    },
                    onError: function(error) {
                        isStreaming = false;
                        finalizeAssistantMessage(true);
                        currentHandle = null;
                        const errorMsg = error.message || 'Unknown error';
                        setStatus('error', 'Error: ' + errorMsg);
                        announce('Error occurred: ' + errorMsg);
                        updateButtonStates();
                        promptInput.focus();
                    }
                });
            }

            function sendMessage() {
                const content = promptInput.value.trim();
                if (!content || isStreaming) return;

                addUserMessage(content);
                promptInput.value = '';
                promptInput.style.height = 'auto';

                startStreaming([...conversationHistory]);
            }

            function stopStreaming() {
                if (currentHandle && isStreaming) {
                    currentHandle.abort();
                    isStreaming = false;
                    finalizeAssistantMessage(true);
                    currentHandle = null;
                    setStatus('stopped', 'Stopped');
                    announce('Response generation stopped');
                    updateButtonStates();
                    promptInput.focus();
                }
            }

            function regenerate() {
                if (isStreaming || conversationHistory.length === 0) return;

                // Remove the last assistant message from history and UI
                if (conversationHistory[conversationHistory.length - 1].role === 'assistant') {
                    conversationHistory.pop();
                    const lastMessageEl = messagesContainer.querySelector('.message.assistant:last-of-type');
                    if (lastMessageEl) {
                        lastMessageEl.remove();
                    }
                }

                announce('Regenerating response');
                // Start streaming with the updated history
                startStreaming([...conversationHistory]);
            }

            // Auto-resize textarea
            promptInput.addEventListener('input', function() {
                this.style.height = 'auto';
                this.style.height = Math.min(this.scrollHeight, 120) + 'px';
            });

            // Send on Enter (Shift+Enter for new line)
            promptInput.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    sendMessage();
                }
            });

            sendButton.addEventListener('click', sendMessage);
            stopButton.addEventListener('click', stopStreaming);
            regenerateButton.addEventListener('click', regenerate);

            // Initial button states
            updateButtonStates();
        })();
    </script>
</body>
