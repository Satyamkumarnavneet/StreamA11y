<script>
(function () {
  'use strict';

  var $ = function (id) { return document.getElementById(id); };
  var transcript = $('transcript');
  var list = $('messages');
  var emptyNote = $('empty');
  var promptEl = $('prompt');
  var sendBtn = $('send');
  var stopBtn = $('stop');
  var regenBtn = $('regenerate');
  var statusEl = $('status');
  var statusRow = $('statusRow');
  var form = $('composer');

  // history entries: {role, content, state, li, textNode, stateEl, errorMsg}
  var messages = [];
  var current = null;   // the one active request, or null
  var seq = 0;
  var statusFlip = false;

  function setStatus(text, isError) {
    // Buttons use aria-disabled (never `disabled`) so focus is never dropped.
    statusFlip = !statusFlip;
    statusEl.textContent = text + (statusFlip ? '' : '\u00A0'); // makes repeated text a real change
    statusEl.classList.toggle('is-error', !!isError);
  }

  function setAriaDisabled(btn, v) { btn.setAttribute('aria-disabled', v ? 'true' : 'false'); }

  function hasUser() {
    for (var i = 0; i < messages.length; i++) if (messages[i].role === 'user') return true;
    return false;
  }

  function updateControls() {
    var streaming = !!current;
    setAriaDisabled(sendBtn, streaming);
    setAriaDisabled(stopBtn, !streaming);
    setAriaDisabled(regenBtn, !hasUser());
    statusRow.classList.toggle('generating', streaming);
  }

  function nearBottom() {
    return transcript.scrollHeight - transcript.scrollTop - transcript.clientHeight < 80;
  }
  function scrollDown() { transcript.scrollTop = transcript.scrollHeight; }

  function renderState(entry) {
    var el = entry.stateEl;
    if (!el) return;
    var text = '', isErr = false;
    if (entry.state === 'streaming') {
      text = 'Generating\u2026';
    } else if (entry.state === 'stopped') {
      text = entry.content ? 'Stopped. Partial answer kept.' : 'Stopped before any text arrived.';
    } else if (entry.state === 'error') {
      isErr = true;
      text = 'Error: ' + (entry.errorMsg || 'the answer could not be completed.') +
             (entry.content ? ' Partial answer kept.' : '');
    }
    el.textContent = text;
    el.hidden = !text;
    el.classList.toggle('error', isErr);
    entry.li.classList.toggle('is-error', isErr);
  }

  function addMessage(role, content, state) {
    var li = document.createElement('li');
    li.className = 'msg ' + role;
    var who = document.createElement('span');
    who.className = 'who';
    who.textContent = role === 'user' ? 'You' : 'Assistant';
    var body = document.createElement('div');
    body.className = 'body';
    var tn = document.createTextNode(content || '');
    body.appendChild(tn);
    var st = document.createElement('p');
    st.className = 'state';
    st.hidden = true;
    li.appendChild(who);
    li.appendChild(body);
    li.appendChild(st);

    var stick = nearBottom();
    list.appendChild(li);
    list.hidden = false;
    emptyNote.hidden = true;

    var entry = { role: role, content: content || '', state: state || 'done', li: li, textNode: tn, stateEl: st, errorMsg: '' };
    messages.push(entry);
    renderState(entry);
    if (stick || role === 'user') scrollDown();
    return entry;
  }

  function removeEntry(entry) {
    var i = messages.indexOf(entry);
    if (i >= 0) messages.splice(i, 1);
    if (entry.li.parentNode) entry.li.parentNode.removeChild(entry.li);
    if (!messages.length) { list.hidden = true; emptyNote.hidden = false; }
  }

  function abortRequest(req) {
    req.aborted = true;
    if (req.handle && typeof req.handle.abort === 'function') {
      try { req.handle.abort(); } catch (e) { /* ignore */ }
    }
  }

  function startAssistant() {
    var history = [];
    for (var i = 0; i < messages.length; i++) {
      var m = messages[i];
      if (m.content) history.push({ role: m.role, content: m.content });
    }
    var entry = addMessage('assistant', '', 'streaming');
    var req = { id: ++seq, entry: entry, handle: null, aborted: false };
    current = req;
    updateControls();
    setStatus('Generating answer\u2026');

    function live() { return current === req && !req.aborted; }

    var callbacks = {
      onToken: function (text) {
        if (!live()) return; // late chunk from a stopped/replaced answer: ignore
        var t = (text === null || text === undefined) ? '' : String(text);
        if (!t) return;
        var stick = nearBottom();
        entry.content += t;
        entry.textNode.appendData(t);
        if (stick) scrollDown();
      },
      onDone: function () {
        if (!live()) return;
        current = null;
        entry.state = 'done';
        renderState(entry);
        updateControls();
        setStatus('Answer finished.');
      },
      onError: function (err) {
        if (!live()) return;
        current = null;
        var msg = '';
        if (err) msg = typeof err === 'string' ? err : (err.message || '');
        entry.state = 'error';
        entry.errorMsg = msg ? String(msg).slice(0, 200) : 'the answer could not be completed.';
        renderState(entry);
        updateControls();
        setStatus('Error: ' + (msg ? String(msg).slice(0, 200) : 'the answer could not be completed.') +
                  (entry.content ? ' Partial answer kept.' : ''), true);
      }
    };

    try {
      var h = window.ChatSDK.streamChat(history, callbacks);
      req.handle = h;
      if (req.aborted && h && typeof h.abort === 'function') {
        try { h.abort(); } catch (e) { /* ignore */ }
      }
    } catch (e) {
      if (current === req) {
        current = null;
        entry.state = 'error';
        entry.errorMsg = (e && e.message) ? String(e.message).slice(0, 200) : 'could not start the answer.';
        renderState(entry);
        updateControls();
        setStatus('Error: ' + entry.errorMsg, true);
      }
    }
  }

  function send() {
    if (current) {
      setStatus('An answer is still being generated. Press Stop, or wait until it finishes.');
      return;
    }
    var text = promptEl.value.trim();
    if (!text) {
      setStatus('Nothing to send. Type a message first.');
      return;
    }
    addMessage('user', text, 'done');
    promptEl.value = '';
    startAssistant();
  }

  function stop() {
    var req = current;
    if (!req) {
      setStatus('There is no answer being generated.');
      return;
    }
    current = null;          // from now on, chunks of this request are ignored
    abortRequest(req);
    req.entry.state = 'stopped';
    renderState(req.entry);  // keeps the partial text already shown
    updateControls();
    setStatus(req.entry.content ? 'Generation stopped. Partial answer kept.' : 'Generation stopped before any text arrived.');
  }

  function regenerate() {
    if (!hasUser()) {
      setStatus('There is no message to regenerate an answer for.');
      return;
    }
    if (current) {
      var req = current;
      current = null;
      abortRequest(req);
    }
    // drop everything after the last user message (the previous answer)
    while (messages.length && messages[messages.length - 1].role !== 'user') {
      removeEntry(messages[messages.length - 1]);
    }
    startAssistant();
    setStatus('Generating a new answer\u2026');
  }

  form.addEventListener('submit', function (e) { e.preventDefault(); send(); });
  stopBtn.addEventListener('click', function (e) { e.preventDefault(); stop(); });
  regenBtn.addEventListener('click', function (e) { e.preventDefault(); regenerate(); });
  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing && !e.ctrlKey && !e.altKey && !e.metaKey) {
      e.preventDefault();
      send();
    }
  });

  updateControls();
})();
</script>
</body>
</html>
