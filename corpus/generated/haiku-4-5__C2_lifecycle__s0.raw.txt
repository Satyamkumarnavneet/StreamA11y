<body>
    <div class="chat-container">

    <!-- Screen reader announcements -->
    <div id="sr-announcements" class="sr-only" role="status" aria-live="assertive" aria-atomic="true"></div>

    <script>
        // State management
        const state = {
            messages: [],
            isGenerating: false,
            currentStreamHandle: null,
            lastUserMessageIndex: -1
        };

        // DOM elements
        const promptInput = document.getElementById('prompt');
        const sendButton = document.getElementById('send');
        const stopButton = document.getElementById('stop');
        const regenerateButton = document.getElementById('regenerate');
        const chatMessages = document.getElementById('chat-messages');
        const statusMessage = document.getElementById('status-message');
        const srAnnouncements = document.getElementById('sr-announcements');

        /**
         * Announce a message to screen readers
         */
        function announceToScreenReader(text) {
            srAnnouncements.textContent = text;
        }

        /**
         * Clear the empty state if needed
         */
        function clearEmptyState() {
            const emptyState = chatMessages.querySelector('.empty-state');
            if (emptyState) {
                emptyState.remove();
            }
        }

        /**
         * Create a message element
         */
        function createMessageElement(role, content, isStreaming = false) {
            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${role}`;
            messageDiv.setAttribute('role', 'article');

            const label = document.createElement('div');
            label.className = 'message-label';
            label.textContent = role === 'user' ? 'You' : 'Assistant';

            const contentDiv = document.createElement('div');
            contentDiv.className = 'message-content';
            contentDiv.textContent = content;

            if (isStreaming) {
                contentDiv.setAttribute('aria-live', 'polite');
                contentDiv.setAttribute('aria-label', 'Assistant is typing');
                contentDiv.id = `assistant-response-${Date.now()}`;
            }

            messageDiv.appendChild(label);
            messageDiv.appendChild(contentDiv);

            return messageDiv;
        }

        /**
         * Add a message to the conversation
         */
        function addMessage(role, content, isStreaming = false) {
            clearEmptyState();
            const messageElement = createMessageElement(role, content, isStreaming);
            chatMessages.appendChild(messageElement);
            scrollToBottom();
            return messageElement;
        }

        /**
         * Update the last assistant message with new content
         */
        function updateLastAssistantMessage(content) {
            const lastMessage = chatMessages.querySelector('.message.assistant:last-of-type .message-content');
            if (lastMessage) {
                lastMessage.textContent = content;
            }
        }

        /**
         * Scroll to bottom of chat
         */
        function scrollToBottom() {
            chatMessages.scrollTop = chatMessages.scrollHeight;
        }

        /**
         * Update status message
         */
        function updateStatus(message, type = '') {
            statusMessage.textContent = message;
            statusMessage.className = `status-message ${type}`.trim();
        }

        /**
         * Clear status message
         */
        function clearStatus() {
            statusMessage.textContent = '';
            statusMessage.className = 'status-message';
        }

        /**
         * Update button visibility
         */
        function updateButtonStates() {
            if (state.isGenerating) {
                sendButton.disabled = true;
                stopButton.style.display = 'block';
                regenerateButton.style.display = 'none';
            } else {
                sendButton.disabled = false;
                stopButton.style.display = 'none';
                regenerateButton.style.display = state.messages.length > 0 ? 'block' : 'none';
            }
        }

        /**
         * Send a message
         */
        async function sendMessage() {
            const message = promptInput.value.trim();
            if (!message) return;

            // Clear input
            promptInput.value = '';
            promptInput.style.height = 'auto';

            // Add user message to state and UI
            state.messages.push({ role: 'user', content: message });
            state.lastUserMessageIndex = state.messages.length - 1;
            addMessage('user', message);

            // Start generating
            generateResponse();
        }

        /**
         * Generate a response to the last user message
         */
        async function generateResponse() {
            if (state.isGenerating) return;
            if (state.lastUserMessageIndex === -1) return;

            state.isGenerating = true;
            updateButtonStates();
            updateStatus('Generating...', 'generating');
            announceToScreenReader('Assistant is generating a response');

            // Add empty assistant message
            const assistantMessageElement = addMessage('assistant', '');
            let accumulatedContent = '';

            // Prepare messages for API
            const messagesForAPI = state.messages.map(msg => ({
                role: msg.role,
                content: msg.content
            }));

            // Stream the response
            const onToken = (text) => {
                // Only update if we're still the current stream
                if (!state.isGenerating) return;

                accumulatedContent += text;
                updateLastAssistantMessage(accumulatedContent);
                scrollToBottom();
            };

            const onDone = () => {
                if (!state.isGenerating) return;

                state.isGenerating = false;
                state.currentStreamHandle = null;

                // Update state with complete message
                if (state.messages[state.messages.length - 1].role === 'assistant') {
                    state.messages[state.messages.length - 1].content = accumulatedContent;
                } else {
                    state.messages.push({ role: 'assistant', content: accumulatedContent });
                }

                updateButtonStates();
                clearStatus();
                announceToScreenReader('Assistant finished generating');

                // Return focus to input if user didn't click something else
                if (document.activeElement === document.body || document.activeElement === stopButton) {
                    promptInput.focus();
                }
            };

            const onError = (error) => {
                if (!state.isGenerating) return;

                state.isGenerating = false;
                state.currentStreamHandle = null;

                const errorMessage = error?.message || 'An error occurred while generating the response';
                updateStatus(`Error: ${errorMessage}`, 'error');
                announceToScreenReader(`Error: ${errorMessage}`);

                updateButtonStates();

                // Return focus to input
                if (document.activeElement === stopButton) {
                    promptInput.focus();
                }
            };

            // Start streaming
            state.currentStreamHandle = window.ChatSDK.streamChat(messagesForAPI, {
                onToken,
                onDone,
                onError
            });
        }

        /**
         * Stop the current generation
         */
        function stopGeneration() {
            if (!state.isGenerating || !state.currentStreamHandle) return;

            state.isGenerating = false;
            state.currentStreamHandle.abort();
            state.currentStreamHandle = null;

            updateButtonStates();
            updateStatus('Stopped', 'success');
            announceToScreenReader('Generation stopped');

            // Return focus to input
            promptInput.focus();
        }

        /**
         * Regenerate the last response
         */
        function regenerateResponse() {
            if (state.isGenerating) return;
            if (state.lastUserMessageIndex === -1) return;

            // Remove last assistant message if it exists
            const lastMessage = chatMessages.querySelector('.message.assistant:last-of-type');
            if (lastMessage && state.messages[state.messages.length - 1].role === 'assistant') {
                lastMessage.remove();
                state.messages.pop();
            }

            generateResponse();
        }

        /**
         * Handle textarea auto-resize
         */
        function autoResizeTextarea() {
            promptInput.style.height = 'auto';
            promptInput.style.height = Math.min(promptInput.scrollHeight, 120) + 'px';
        }

        // Event listeners
        sendButton.addEventListener('click', sendMessage);
        stopButton.addEventListener('click', stopGeneration);
        regenerateButton.addEventListener('click', regenerateResponse);

        promptInput.addEventListener('input', autoResizeTextarea);
        promptInput.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' && !e.shiftKey && !state.isGenerating) {
                e.preventDefault();
                sendMessage();
            }
        });

        // Initialize button states
        updateButtonStates();
    </script>
</body>
</html>
