<script>
(function () {
  'use strict';

  var promptEl = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var form = document.getElementById('composer');
  var transcript = document.getElementById('transcript');
  var list = document.getElementById('messages');
  var emptyNote = document.getElementById('empty');
  var statusEl = document.getElementById('status');

  // Conversation state. Each entry: { role, content, status, el, textNode, stateEl, errorEl }
  var history = [];
  // The single active generation, or null. Callbacks from any other generation are ignored.
  var active = null;
  var genCounter = 0;

  /* ---------- Status announcements ---------- */
  var announceTimer = null;
  function announce(message, kind) {
    statusEl.setAttribute('data-kind', kind || 'idle');
    // Clear then set so repeated identical messages are still announced.
    if (announceTimer) clearTimeout(announceTimer);
    statusEl.textContent = '';
    announceTimer = setTimeout(function () {
      statusEl.textContent = message;
      announceTimer = null;
    }, 60);
  }

  /* ---------- Rendering ---------- */
  function isNearBottom() {
    return transcript.scrollHeight - transcript.scrollTop - transcript.clientHeight < 40;
  }
  function scrollToBottomIf(wasNear) {
    // Scrolling does not move keyboard focus.
    if (wasNear) transcript.scrollTop = transcript.scrollHeight;
  }

  function createMessageElement(entry) {
    var li = document.createElement('li');
    li.className = 'msg ' + entry.role;

    var meta = document.createElement('p');
    meta.className = 'meta';
    var who = document.createElement('span');
    who.textContent = entry.role === 'user' ? 'You:' : 'Assistant:';
    var stateEl = document.createElement('span');
    stateEl.className = 'state';
    meta.appendChild(who);
    meta.appendChild(document.createTextNode(' '));
    meta.appendChild(stateEl);

    var content = document.createElement('div');
    content.className = 'content';
    var textNode = document.createTextNode(entry.content);
    content.appendChild(textNode);

    var errorEl = document.createElement('p');
    errorEl.className = 'error-text';

    li.appendChild(meta);
    li.appendChild(content);
    li.appendChild(errorEl);

    entry.el = li;
    entry.textNode = textNode;
    entry.stateEl = stateEl;
    entry.errorEl = errorEl;
    return li;
  }

  function addEntry(entry) {
    var near = isNearBottom();
    history.push(entry);
    list.appendChild(createMessageElement(entry));
    emptyNote.hidden = history.length > 0;
    scrollToBottomIf(near);
  }

  function removeEntriesAfter(index) {
    var removed = history.splice(index + 1);
    removed.forEach(function (e) {
      if (e.el && e.el.parentNode) e.el.parentNode.removeChild(e.el);
    });
    emptyNote.hidden = history.length > 0;
  }

  function setEntryState(entry, status, errorMessage) {
    entry.status = status;
    var label = '';
    entry.stateEl.classList.remove('dots');
    if (status === 'streaming') {
      label = 'Generating';
      entry.stateEl.classList.add('dots');
    } else if (status === 'stopped') {
      label = 'Stopped';
    } else if (status === 'error') {
      label = 'Error';
    }
    entry.stateEl.textContent = label;
    entry.el.classList.toggle('error', status === 'error');
    entry.errorEl.textContent = status === 'error'
      ? 'The answer could not be completed: ' + errorMessage
      : '';
  }

  /* ---------- Controls ---------- */
  function lastUserIndex() {
    for (var i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') return i;
    }
    return -1;
  }

  // We use aria-disabled instead of the disabled attribute so a focused
  // button never loses focus when its state changes.
  function updateControls() {
    var streaming = active !== null;
    sendBtn.setAttribute('aria-disabled', streaming ? 'true' : 'false');
    stopBtn.setAttribute('aria-disabled', streaming ? 'false' : 'true');
    regenBtn.setAttribute('aria-disabled', (streaming || lastUserIndex() === -1) ? 'true' : 'false');
  }

  /* ---------- Streaming ---------- */
  function apiMessages() {
    var out = [];
    history.forEach(function (e) {
      if ((e.role === 'user' || e.role === 'assistant') && e.content.length > 0) {
        out.push({ role: e.role, content: e.content });
      }
    });
    return out;
  }

  function errorText(err) {
    if (!err) return 'Unknown error.';
    if (typeof err === 'string') return err;
    if (err.message) return String(err.message);
    try { return String(err); } catch (e) { return 'Unknown error.'; }
  }

  function endGeneration(gen, outcome, err) {
    if (gen.ended) return;
    gen.ended = true;
    if (active === gen) active = null;

    var entry = gen.entry;
    entry.el.removeAttribute('aria-busy');

    if (outcome === 'done') {
      setEntryState(entry, 'done');
      announce('Answer finished.', 'done');
    } else if (outcome === 'stopped') {
      setEntryState(entry, 'stopped');
      announce(entry.content.length
        ? 'Answer stopped. The partial answer was kept.'
        : 'Answer stopped before any text arrived.', 'idle');
    } else {
      var msg = errorText(err);
      setEntryState(entry, 'error', msg);
      announce('Error: the answer failed. ' + msg, 'error');
    }
    updateControls();
  }

  function startGeneration() {
    if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
      announce('Error: the chat service is not available.', 'error');
      return;
    }

    var messages = apiMessages();
    var entry = { role: 'assistant', content: '', status: 'streaming' };
    addEntry(entry);
    entry.el.setAttribute('aria-busy', 'true');
    setEntryState(entry, 'streaming');

    var gen = { id: ++genCounter, entry: entry, ended: false, handle: null };
    active = gen;
    updateControls();
    announce('Generating answer…', 'busy');

    function isLive() { return active === gen && !gen.ended; }

    try {
      gen.handle = window.ChatSDK.streamChat(messages, {
        onToken: function (text) {
          // Ignore chunks from stopped, finished, failed or replaced answers.
          if (!isLive()) return;
          if (text == null) return;
          var chunk = String(text);
          if (!chunk) return;
          var near = isNearBottom();
          entry.content += chunk;
          entry.textNode.appendData(chunk);
          scrollToBottomIf(near);
        },
        onDone: function () {
          if (!isLive()) return;
          endGeneration(gen, 'done');
        },
        onError: function (error) {
          if (!isLive()) return;
          endGeneration(gen, 'error', error);
        }
      });
    } catch (e) {
      if (isLive()) endGeneration(gen, 'error', e);
    }
  }

  function stopGeneration() {
    var gen = active;
    if (!gen) {
      announce('Nothing is being generated.', 'idle');
      return;
    }
    // Mark as ended first so any late chunk is ignored.
    endGeneration(gen, 'stopped');
    try {
      if (gen.handle && typeof gen.handle.abort === 'function') gen.handle.abort();
    } catch (e) { /* ignore abort failures; the answer is already stopped locally */ }
  }

  /* ---------- Actions ---------- */
  function send() {
    if (active) {
      announce('An answer is still being generated. Stop it before sending a new message.', 'busy');
      return;
    }
    var text = promptEl.value;
    if (!text.trim()) {
      announce('Type a message before sending.', 'idle');
      return;
    }
    addEntry({ role: 'user', content: text, status: 'done' });
    setEntryState(history[history.length - 1], 'done');
    promptEl.value = '';
    startGeneration();
  }

  function regenerate() {
    if (active) {
      announce('An answer is still being generated. Stop it before regenerating.', 'busy');
      return;
    }
    var idx = lastUserIndex();
    if (idx === -1) {
      announce('There is no message to regenerate an answer for.', 'idle');
      return;
    }
    removeEntriesAfter(idx);
    startGeneration();
  }

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    send();
  });

  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.altKey && !e.ctrlKey && !e.metaKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });

  stopBtn.addEventListener('click', function () {
    if (stopBtn.getAttribute('aria-disabled') === 'true') {
      announce('Nothing is being generated.', 'idle');
      return;
    }
    stopGeneration();
  });

  regenBtn.addEventListener('click', function () {
    regenerate();
  });

  // Escape inside the message box stops a streaming answer without moving focus.
  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && active) {
      e.preventDefault();
      stopGeneration();
    }
  });

  updateControls();
})();
</script>
</body>
</html>
