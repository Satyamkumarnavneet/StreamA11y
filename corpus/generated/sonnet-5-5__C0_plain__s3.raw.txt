<script>
(function () {
  'use strict';

  var logEl = document.getElementById('log');
  var promptEl = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var statusEl = document.getElementById('status');
  var statusText = document.getElementById('status-text');

  // history entry: { role, content, state, errorText, el, bubble, note }
  // state (assistant only): 'generating' | 'done' | 'stopped' | 'error'
  var history = [];
  var activeId = 0;        // id of the stream whose callbacks are accepted; 0 = none
  var nextId = 1;
  var activeHandle = null;
  var activeEntry = null;

  var STATUS_LABELS = {
    idle: 'Ready',
    generating: 'Generating…',
    done: 'Finished',
    stopped: 'Stopped',
    error: 'Error'
  };

  function setStatus(state, detail) {
    statusEl.setAttribute('data-state', state);
    statusText.textContent = detail || STATUS_LABELS[state];
  }

  function isStreaming() { return activeId !== 0; }

  function updateControls() {
    var streaming = isStreaming();
    var hasUser = history.some(function (m) { return m.role === 'user'; });
    sendBtn.disabled = streaming;
    stopBtn.disabled = !streaming;
    regenBtn.disabled = !hasUser;
  }

  function nearBottom() {
    return logEl.scrollHeight - logEl.scrollTop - logEl.clientHeight < 80;
  }
  function scrollDown() { logEl.scrollTop = logEl.scrollHeight; }

  function noteText(entry) {
    switch (entry.state) {
      case 'generating': return 'Generating…';
      case 'done': return 'Finished';
      case 'stopped': return 'Stopped — partial answer kept';
      case 'error': return 'Error: ' + (entry.errorText || 'the response failed') +
        (entry.content ? ' (partial answer kept)' : '');
      default: return '';
    }
  }

  function paintAssistant(entry) {
    entry.el.className = 'msg assistant ' + entry.state;
    if (entry.content) {
      entry.bubble.classList.remove('placeholder');
      entry.bubble.textContent = entry.content;
    } else {
      var txt = entry.state === 'generating' ? '' :
                entry.state === 'stopped' ? '(stopped before any text arrived)' :
                entry.state === 'error' ? '(no answer received)' : '(empty answer)';
      entry.bubble.textContent = txt;
      entry.bubble.classList.toggle('placeholder', !!txt);
    }
    entry.note.className = 'note ' + entry.state;
    entry.note.textContent = noteText(entry);
  }

  function buildEntryDom(entry) {
    var wrap = document.createElement('div');
    wrap.className = 'msg ' + entry.role;
    var role = document.createElement('div');
    role.className = 'role';
    role.textContent = entry.role === 'user' ? 'You' : 'Assistant';
    var bubble = document.createElement('div');
    bubble.className = 'bubble';
    wrap.appendChild(role);
    wrap.appendChild(bubble);
    entry.el = wrap;
    entry.bubble = bubble;
    if (entry.role === 'assistant') {
      var note = document.createElement('div');
      note.className = 'note';
      wrap.appendChild(note);
      entry.note = note;
      paintAssistant(entry);
    } else {
      bubble.textContent = entry.content;
    }
    return wrap;
  }

  function renderAll() {
    logEl.textContent = '';
    if (!history.length) {
      var e = document.createElement('div');
      e.className = 'empty';
      e.textContent = 'No messages yet. Say hello to get started.';
      logEl.appendChild(e);
    } else {
      history.forEach(function (m) { logEl.appendChild(buildEntryDom(m)); });
    }
    scrollDown();
  }

  function appendEntry(entry) {
    if (!history.length || logEl.querySelector('.empty')) logEl.textContent = '';
    history.push(entry);
    logEl.appendChild(buildEntryDom(entry));
    scrollDown();
  }

  function apiMessages(upToExclusive) {
    return history.slice(0, upToExclusive)
      .filter(function (m) { return m.content && m.content.length; })
      .map(function (m) { return { role: m.role, content: m.content }; });
  }

  // Detach the current stream: later callbacks (including a late token) are ignored.
  function detachActive() {
    var h = activeHandle;
    activeId = 0;
    activeHandle = null;
    activeEntry = null;
    if (h && typeof h.abort === 'function') {
      try { h.abort(); } catch (e) { /* ignore */ }
    }
  }

  function startStream() {
    // History currently ends with a user message; create assistant entry.
    var messages = apiMessages(history.length);
    var entry = { role: 'assistant', content: '', state: 'generating', errorText: '' };
    appendEntry(entry);

    var id = nextId++;
    activeId = id;
    activeEntry = entry;
    activeHandle = null;
    setStatus('generating');
    updateControls();

    function finish(state, errText) {
      if (activeId !== id) return;
      activeId = 0; activeHandle = null; activeEntry = null;
      entry.state = state;
      entry.errorText = errText || '';
      var stick = nearBottom();
      paintAssistant(entry);
      if (stick) scrollDown();
      if (state === 'error') setStatus('error', 'Error: ' + (errText || 'the response failed'));
      else setStatus(state);
      updateControls();
      promptEl.focus();
    }

    var handle;
    try {
      if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
        throw new Error('Chat service is not available');
      }
      handle = window.ChatSDK.streamChat(messages, {
        onToken: function (text) {
          if (activeId !== id) return; // stale / aborted
          if (typeof text !== 'string' || !text) return;
          entry.content += text;
          var stick = nearBottom();
          entry.bubble.classList.remove('placeholder');
          entry.bubble.textContent = entry.content;
          if (stick) scrollDown();
        },
        onDone: function () { finish('done'); },
        onError: function (err) {
          var msg = (err && (err.message || String(err))) || 'the response failed';
          finish('error', msg);
        }
      });
    } catch (err) {
      finish('error', (err && err.message) || 'could not start the request');
      return;
    }
    // Callbacks may in theory fire synchronously before handle is returned.
    if (activeId === id) activeHandle = handle;
    else if (handle && typeof handle.abort === 'function') { /* already finished */ }
  }

  function send() {
    if (isStreaming()) return;
    var text = promptEl.value.trim();
    if (!text) return;
    promptEl.value = '';
    autoSize();
    appendEntry({ role: 'user', content: text });
    startStream();
  }

  function stop() {
    if (!isStreaming()) return;
    var entry = activeEntry;
    detachActive();
    if (entry) {
      entry.state = 'stopped';
      paintAssistant(entry);
    }
    setStatus('stopped');
    updateControls();
    promptEl.focus();
  }

  function regenerate() {
    var lastUser = -1;
    for (var i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') { lastUser = i; break; }
    }
    if (lastUser < 0) return;
    if (isStreaming()) detachActive();
    history.length = lastUser + 1; // drop every answer after the last user message
    renderAll();
    startStream();
  }

  function autoSize() {
    promptEl.style.height = 'auto';
    promptEl.style.height = Math.min(promptEl.scrollHeight, 160) + 'px';
  }

  sendBtn.addEventListener('click', send);
  stopBtn.addEventListener('click', stop);
  regenBtn.addEventListener('click', regenerate);
  promptEl.addEventListener('input', autoSize);
  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });

  renderAll();
  setStatus('idle');
  updateControls();
  promptEl.focus();
})();
</script>
</body>
</html>
