<script>
(function () {
  'use strict';

  var form = document.getElementById('chat-form');
  var promptEl = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var transcript = document.getElementById('transcript');
  var list = document.getElementById('messages');
  var emptyEl = document.getElementById('empty');
  var statusEl = document.getElementById('status-bar');

  /** history entries: {role, content, status, el, bodyEl, badgeEl} */
  var history = [];
  /** current generation: {id, msg, handle, active} or null */
  var current = null;
  var genCounter = 0;
  var lastAnnouncement = '';

  // ---------- Announcements ----------
  function announce(text) {
    // If the same text repeats, alter it invisibly so screen readers re-announce it.
    if (text === lastAnnouncement) text = text + '\u00A0';
    lastAnnouncement = text;
    statusEl.textContent = text;
  }

  // ---------- Button state (aria-disabled keeps focus on the button) ----------
  function setDisabled(btn, disabled) {
    btn.setAttribute('aria-disabled', disabled ? 'true' : 'false');
  }
  function isDisabled(btn) {
    return btn.getAttribute('aria-disabled') === 'true';
  }
  function lastUserIndex() {
    for (var i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') return i;
    }
    return -1;
  }
  function updateControls() {
    var streaming = !!current;
    setDisabled(sendBtn, streaming);
    setDisabled(stopBtn, !streaming);
    setDisabled(regenBtn, lastUserIndex() === -1);
    emptyEl.hidden = history.length > 0;
  }

  // ---------- Scrolling (never touches focus) ----------
  function isNearBottom() {
    return transcript.scrollHeight - transcript.scrollTop - transcript.clientHeight < 40;
  }
  function scrollToBottom() {
    transcript.scrollTop = transcript.scrollHeight;
  }

  // ---------- Rendering ----------
  function createMessage(role, content) {
    var msg = { role: role, content: content, status: role === 'user' ? 'sent' : 'generating' };
    var li = document.createElement('li');
    li.className = 'msg ' + role;

    var header = document.createElement('div');
    header.className = 'msg-header';
    var h = document.createElement('h2');
    h.textContent = role === 'user' ? 'You' : 'Assistant';
    header.appendChild(h);

    var badge = document.createElement('span');
    badge.className = 'badge';
    header.appendChild(badge);

    var body = document.createElement('div');
    body.className = 'msg-body';
    body.textContent = content;

    li.appendChild(header);
    li.appendChild(body);

    msg.el = li;
    msg.bodyEl = body;
    msg.badgeEl = badge;

    var stick = isNearBottom();
    list.appendChild(li);
    history.push(msg);
    if (stick || role === 'user') scrollToBottom();
    return msg;
  }

  function setBadge(msg, status, detail) {
    msg.status = status;
    var b = msg.badgeEl;
    b.className = 'badge';
    if (status === 'generating') {
      b.textContent = '(generating\u2026)';
    } else if (status === 'stopped') {
      b.classList.add('stopped');
      b.textContent = msg.content ? '(stopped \u2014 partial answer)' : '(stopped before any text was received)';
    } else if (status === 'done') {
      b.classList.add('done');
      b.textContent = '(finished)';
    } else if (status === 'error') {
      b.classList.add('error');
      b.textContent = '(error: ' + detail + ')';
    } else {
      b.textContent = '';
    }
  }

  function removeMessage(msg) {
    if (msg.el && msg.el.parentNode) msg.el.parentNode.removeChild(msg.el);
    var i = history.indexOf(msg);
    if (i !== -1) history.splice(i, 1);
  }

  // ---------- API messages ----------
  function buildApiMessages() {
    var out = [];
    for (var i = 0; i < history.length; i++) {
      var m = history[i];
      if (m.role === 'assistant' && (m.status === 'generating' || !m.content)) continue;
      out.push({ role: m.role, content: m.content });
    }
    return out;
  }

  function errorText(err) {
    if (!err) return 'unknown error';
    if (typeof err === 'string') return err;
    if (err.message) return String(err.message);
    try { return String(err); } catch (e) { return 'unknown error'; }
  }

  // ---------- Streaming ----------
  function finish(gen, kind, detail, silent) {
    if (!gen.active) return;
    gen.active = false;
    if (current === gen) current = null;

    var msg = gen.msg;
    msg.el.removeAttribute('aria-busy'); // never leave busy state behind
    setBadge(msg, kind, detail);
    updateControls();

    if (silent) return;
    if (kind === 'done') announce('Answer finished.');
    else if (kind === 'stopped') announce(msg.content ? 'Answer stopped. Partial answer kept.' : 'Answer stopped.');
    else if (kind === 'error') announce('Error: the answer could not be completed. ' + detail);
  }

  function startGeneration(announcement) {
    var apiMessages = buildApiMessages();
    var msg = createMessage('assistant', '');
    msg.el.setAttribute('aria-busy', 'true');
    setBadge(msg, 'generating');

    var gen = { id: ++genCounter, msg: msg, handle: null, active: true };
    current = gen;
    updateControls();
    announce(announcement || 'Assistant is generating an answer.');

    function live() { return gen.active && current === gen; }

    try {
      gen.handle = window.ChatSDK.streamChat(apiMessages, {
        onToken: function (text) {
          if (!live()) return; // ignore late chunks from stopped/replaced answers
          if (text == null) return;
          text = String(text);
          var stick = isNearBottom();
          msg.content += text;
          msg.bodyEl.appendChild(document.createTextNode(text));
          if (stick) scrollToBottom();
        },
        onDone: function () {
          if (!live()) return;
          finish(gen, 'done');
        },
        onError: function (err) {
          if (!live()) return;
          finish(gen, 'error', errorText(err));
        }
      });
    } catch (err) {
      if (live()) finish(gen, 'error', errorText(err));
    }
  }

  function abortCurrent(silent) {
    var gen = current;
    if (!gen) return false;
    finish(gen, 'stopped', null, silent); // mark inactive first so later chunks are dropped
    try { if (gen.handle && typeof gen.handle.abort === 'function') gen.handle.abort(); } catch (e) { /* ignore */ }
    return true;
  }

  // ---------- Actions ----------
  function send() {
    if (current) {
      announce('Please wait for the answer to finish, or press Stop.');
      return;
    }
    var text = promptEl.value;
    if (!text.trim()) {
      announce('Type a message before sending.');
      return;
    }
    if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
      announce('Error: the chat service is not available.');
      return;
    }
    promptEl.value = '';
    createMessage('user', text);
    startGeneration('Message sent. Assistant is generating an answer.');
  }

  function stop() {
    if (!current) {
      announce('No answer is being generated.');
      return;
    }
    abortCurrent(false);
  }

  function regenerate() {
    if (lastUserIndex() === -1) {
      announce('There is no message to regenerate an answer for.');
      return;
    }
    if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
      announce('Error: the chat service is not available.');
      return;
    }
    abortCurrent(true);
    var idx = lastUserIndex();
    // Remove every assistant answer after the last user message.
    var toRemove = history.slice(idx + 1);
    for (var i = 0; i < toRemove.length; i++) removeMessage(toRemove[i]);
    startGeneration('Regenerating answer. Assistant is generating.');
  }

  // ---------- Events ----------
  form.addEventListener('submit', function (e) {
    e.preventDefault();
    send();
  });

  stopBtn.addEventListener('click', function () {
    if (isDisabled(stopBtn)) { announce('No answer is being generated.'); return; }
    stop();
  });

  regenBtn.addEventListener('click', function () {
    if (isDisabled(regenBtn)) { announce('There is no message to regenerate an answer for.'); return; }
    regenerate();
  });

  promptEl.addEventListener('keydown', function (e) {
    if (e.isComposing || e.keyCode === 229) return;
    if (e.key === 'Enter' && !e.shiftKey && !e.altKey && !e.ctrlKey && !e.metaKey) {
      e.preventDefault();
      send();
    } else if (e.key === 'Escape' && current) {
      e.preventDefault();
      stop();
    }
  });

  updateControls();
})();
</script>
</body>
</html>
