<script>
(function () {
  "use strict";

  var els = {};
  ["transcript", "history", "empty", "status", "error", "composer",
   "prompt", "send", "stop", "regenerate"].forEach(function (id) {
    els[id] = document.getElementById(id);
  });

  // Conversation entries: {role, content, item}. `item` is the DOM node and is never sent to the SDK.
  var conversation = [];

  // The single answer currently streaming, or null.
  var active = null;

  function setStatus(text) {
    els.status.textContent = text;
  }

  function showError(text) {
    els.error.textContent = text;
  }

  function clearError() {
    els.error.textContent = "";
  }

  function scrollToEnd() {
    els.transcript.scrollTop = els.transcript.scrollHeight;
  }

  function updateEmpty() {
    els.empty.hidden = els.history.children.length > 0;
  }

  // aria-disabled (not the disabled attribute) keeps focus on the button when its state changes.
  function setAriaDisabled(btn, disabled) {
    btn.setAttribute("aria-disabled", disabled ? "true" : "false");
  }

  function isAriaDisabled(btn) {
    return btn.getAttribute("aria-disabled") === "true";
  }

  function refreshControls() {
    var streaming = active !== null;
    var hasUser = conversation.some(function (m) { return m.role === "user"; });
    setAriaDisabled(els.stop, !streaming);
    setAriaDisabled(els.regenerate, streaming || !hasUser);
    // Send stays focusable; while streaming it reports that it is waiting.
    setAriaDisabled(els.send, streaming);
  }

  function appendMessage(role, text, busy) {
    var item = document.createElement("li");
    item.className = "msg " + role;

    var who = document.createElement("p");
    who.className = "who";
    who.textContent = role === "user" ? "You" : "Assistant";

    var body = document.createElement("div");
    body.className = "body";
    body.textContent = text;
    if (busy) {
      body.setAttribute("aria-busy", "true");
    }

    item.appendChild(who);
    item.appendChild(body);
    els.history.appendChild(item);
    updateEmpty();
    scrollToEnd();
    return { item: item, body: body };
  }

  function addNote(item, text) {
    var note = document.createElement("p");
    note.className = "note";
    note.textContent = text;
    item.appendChild(note);
  }

  function describeError(err) {
    if (err && typeof err.message === "string" && err.message) {
      return err.message;
    }
    if (typeof err === "string" && err) {
      return err;
    }
    return "Unknown error.";
  }

  // Removes an answer that never received any text.
  function discardAnswer(state) {
    var idx = conversation.indexOf(state.entry);
    if (idx !== -1) {
      conversation.splice(idx, 1);
    }
    state.item.remove();
    updateEmpty();
  }

  // Ends the stream bookkeeping. Every end path (done, stop, error) goes through here.
  function endStream(state) {
    state.closed = true;
    state.body.removeAttribute("aria-busy");
    if (active === state) {
      active = null;
    }
    refreshControls();
  }

  function startAnswer(statusText) {
    // Copy only role and content for the SDK.
    var request = conversation.map(function (m) {
      return { role: m.role, content: m.content };
    });

    var view = appendMessage("assistant", "", true);
    var entry = { role: "assistant", content: "", item: view.item };
    conversation.push(entry);

    var state = {
      entry: entry,
      item: view.item,
      body: view.body,
      text: "",
      handle: null,
      closed: false,
      stopped: false
    };

    active = state;
    refreshControls();
    setStatus(statusText);

    try {
      state.handle = window.ChatSDK.streamChat(request, {
        onToken: function (chunk) {
          if (state.closed) { return; }
          state.text += String(chunk);
          entry.content = state.text;
          // Updates the visible text only. The body has no live region, so this is not announced.
          state.body.textContent = state.text;
          scrollToEnd();
        },
        onDone: function () {
          if (state.closed) { return; }
          endStream(state);
          setStatus("Answer finished.");
        },
        onError: function (err) {
          if (state.closed) { return; }
          endStream(state);
          if (state.text === "") {
            discardAnswer(state);
          } else {
            addNote(state.item, "Stopped by an error. Partial answer kept.");
          }
          setStatus("Answer failed.");
          showError("Answer failed: " + describeError(err));
        }
      });
    } catch (err) {
      if (!state.closed) {
        endStream(state);
        if (state.text === "") {
          discardAnswer(state);
        }
        setStatus("Answer failed.");
        showError("Answer failed: " + describeError(err));
      }
    }
  }

  function send() {
    if (active !== null) {
      setStatus("An answer is still generating. Press Stop to end it first.");
      return;
    }

    var text = els.prompt.value.trim();
    if (text === "") {
      setStatus("Type a message before sending.");
      return;
    }

    if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== "function") {
      showError("Chat service is not available.");
      setStatus("Answer failed.");
      return;
    }

    clearError();
    els.prompt.value = "";

    var view = appendMessage("user", text, false);
    conversation.push({ role: "user", content: text, item: view.item });

    startAnswer("Generating answer.");
  }

  function stop() {
    var state = active;
    if (state === null) { return; }

    state.closed = true;
    state.stopped = true;
    try {
      if (state.handle && typeof state.handle.abort === "function") {
        state.handle.abort();
      }
    } catch (err) {
      // The stream is already treated as closed, so an abort failure is ignored.
    }

    endStream(state);

    if (state.text === "") {
      discardAnswer(state);
      setStatus("Answer stopped.");
    } else {
      addNote(state.item, "Stopped. Partial answer kept.");
      setStatus("Answer stopped. Partial answer kept.");
    }
    // Focus is not moved; the Stop button keeps focus.
  }

  function regenerate() {
    if (active !== null) { return; }

    var lastUser = -1;
    for (var i = conversation.length - 1; i >= 0; i--) {
      if (conversation[i].role === "user") {
        lastUser = i;
        break;
      }
    }
    if (lastUser === -1) { return; }

    // Remove the answers that follow the last user message.
    conversation.slice(lastUser + 1).forEach(function (m) {
      if (m.item) { m.item.remove(); }
    });
    conversation.length = lastUser + 1;
    updateEmpty();

    clearError();
    startAnswer("Regenerating answer.");
  }

  // Event wiring
  els.composer.addEventListener("submit", function (e) {
    e.preventDefault();
    send();
  });

  els.prompt.addEventListener("keydown", function (e) {
    if (e.key === "Enter" && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });

  els.stop.addEventListener("click", function () {
    if (isAriaDisabled(els.stop)) { return; }
    stop();
  });

  els.regenerate.addEventListener("click", function () {
    if (isAriaDisabled(els.regenerate)) { return; }
    regenerate();
  });

  updateEmpty();
  refreshControls();
})();
</script>
</body>
</html>
