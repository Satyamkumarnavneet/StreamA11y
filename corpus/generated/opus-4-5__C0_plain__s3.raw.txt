<body>
    <div class="chat-container">
        <div class="chat-header">Chat Assistant</div>
        <div class="chat-messages" id="messages">
            <div class="empty-state" id="emptyState">Start a conversation by typing a message below</div>
        </div>
        <div class="status-indicator" id="status"></div>
        <div class="chat-input-area">
            <div class="input-row">
                <textarea id="prompt" rows="2" placeholder="Type your message..."></textarea>
            </div>
            <div class="button-row">

                <button id="regenerate" disabled>Regenerate</button>
            </div>
        </div>

    <script>
        (function() {
            const messagesContainer = document.getElementById('messages');
            const emptyState = document.getElementById('emptyState');
            const statusIndicator = document.getElementById('status');
            const promptInput = document.getElementById('prompt');
            const sendButton = document.getElementById('send');
            const stopButton = document.getElementById('stop');
            const regenerateButton = document.getElementById('regenerate');

            let conversationHistory = [];
            let currentHandle = null;
            let isStreaming = false;
            let currentAssistantMessage = null;

            function setStatus(text, type) {
                statusIndicator.textContent = text;
                statusIndicator.className = 'status-indicator';
                if (type) {
                    statusIndicator.classList.add(type);
                }
            }

            function clearStatus() {
                statusIndicator.textContent = '';
                statusIndicator.className = 'status-indicator';
            }

            function updateButtonStates() {
                const hasHistory = conversationHistory.length > 0;
                const lastIsUser = hasHistory && conversationHistory[conversationHistory.length - 1].role === 'user';
                
                sendButton.disabled = isStreaming;
                stopButton.disabled = !isStreaming;
                regenerateButton.disabled = isStreaming || !hasHistory || lastIsUser;
            }

            function scrollToBottom() {
                messagesContainer.scrollTop = messagesContainer.scrollHeight;
            }

            function addMessageElement(role, content) {
                if (emptyState.parentNode) {
                    emptyState.remove();
                }

                const messageDiv = document.createElement('div');
                messageDiv.className = `message ${role}`;
                messageDiv.textContent = content;
                messagesContainer.appendChild(messageDiv);
                scrollToBottom();
                return messageDiv;
            }

            function startStream(messages) {
                isStreaming = true;
                updateButtonStates();
                setStatus('Generating...', 'generating');

                currentAssistantMessage = addMessageElement('assistant', '');
                currentAssistantMessage.classList.add('streaming');

                let accumulatedContent = '';

                currentHandle = window.ChatSDK.streamChat(messages, {
                    onToken: function(text) {
                        accumulatedContent += text;
                        currentAssistantMessage.textContent = accumulatedContent;
                        scrollToBottom();
                    },
                    onDone: function() {
                        isStreaming = false;
                        currentHandle = null;
                        currentAssistantMessage.classList.remove('streaming');
                        
                        conversationHistory.push({
                            role: 'assistant',
                            content: accumulatedContent
                        });

                        setStatus('Finished', 'finished');
                        setTimeout(clearStatus, 2000);
                        updateButtonStates();
                        currentAssistantMessage = null;
                    },
                    onError: function(error) {
                        isStreaming = false;
                        currentHandle = null;
                        currentAssistantMessage.classList.remove('streaming');

                        if (accumulatedContent) {
                            conversationHistory.push({
                                role: 'assistant',
                                content: accumulatedContent
                            });
                        } else {
                            currentAssistantMessage.remove();
                        }

                        setStatus('Error: ' + (error.message || 'Something went wrong'), 'error');
                        updateButtonStates();
                        currentAssistantMessage = null;
                    }
                });
            }

            function sendMessage() {
                const content = promptInput.value.trim();
                if (!content || isStreaming) return;

                promptInput.value = '';
                
                conversationHistory.push({
                    role: 'user',
                    content: content
                });

                addMessageElement('user', content);
                startStream([...conversationHistory]);
            }

            function stopGeneration() {
                if (currentHandle && isStreaming) {
                    currentHandle.abort();
                    isStreaming = false;
                    currentHandle = null;

                    if (currentAssistantMessage) {
                        currentAssistantMessage.classList.remove('streaming');
                        const partialContent = currentAssistantMessage.textContent;
                        
                        if (partialContent) {
                            conversationHistory.push({
                                role: 'assistant',
                                content: partialContent
                            });
                        } else {
                            currentAssistantMessage.remove();
                        }
                        currentAssistantMessage = null;
                    }

                    setStatus('Stopped', 'stopped');
                    setTimeout(clearStatus, 2000);
                    updateButtonStates();
                }
            }

            function regenerate() {
                if (isStreaming || conversationHistory.length === 0) return;

                // Remove the last assistant message if it exists
                if (conversationHistory[conversationHistory.length - 1].role === 'assistant') {
                    conversationHistory.pop();
                    
                    // Remove from DOM
                    const messages = messagesContainer.querySelectorAll('.message.assistant');
                    if (messages.length > 0) {
                        messages[messages.length - 1].remove();
                    }
                }

                // Now regenerate based on the conversation up to and including the last user message
                if (conversationHistory.length > 0 && conversationHistory[conversationHistory.length - 1].role === 'user') {
                    startStream([...conversationHistory]);
                }
            }

            sendButton.addEventListener('click', sendMessage);
            stopButton.addEventListener('click', stopGeneration);
            regenerateButton.addEventListener('click', regenerate);

            promptInput.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    sendMessage();
                }
            });

            updateButtonStates();
        })();
    </script>
</body>
