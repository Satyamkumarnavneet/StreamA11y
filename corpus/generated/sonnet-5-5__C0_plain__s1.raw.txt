<script>
(function () {
  'use strict';

  var conversationEl = document.getElementById('conversation');
  var emptyHintEl = document.getElementById('empty-hint');
  var promptEl = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var statusEl = document.getElementById('status');
  var statusTextEl = document.getElementById('status-text');

  // history entries: { role, content, el, bubbleEl, metaEl }
  var history = [];
  // the stream currently in progress (null if none)
  var current = null;

  /* ---------- helpers ---------- */

  function setStatus(state, text) {
    statusEl.setAttribute('data-state', state);
    statusTextEl.textContent = text;
  }

  function nearBottom() {
    return conversationEl.scrollHeight - conversationEl.scrollTop - conversationEl.clientHeight < 80;
  }
  function scrollToBottom() {
    conversationEl.scrollTop = conversationEl.scrollHeight;
  }

  function errorText(err) {
    if (!err) return 'Unknown error';
    if (typeof err === 'string') return err;
    if (err.message) return String(err.message);
    try { return String(err); } catch (e) { return 'Unknown error'; }
  }

  function lastUserIndex() {
    for (var i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') return i;
    }
    return -1;
  }

  function updateControls() {
    var generating = !!current;
    sendBtn.disabled = generating;
    stopBtn.disabled = !generating;
    regenBtn.disabled = lastUserIndex() === -1;
    emptyHintEl.style.display = history.length ? 'none' : '';
  }

  function addMessage(role, content) {
    var stick = nearBottom();
    var wrap = document.createElement('div');
    wrap.className = 'msg ' + role;

    var roleEl = document.createElement('div');
    roleEl.className = 'role';
    roleEl.textContent = role === 'user' ? 'You' : 'Assistant';

    var bubble = document.createElement('div');
    bubble.className = 'bubble';
    bubble.textContent = content;

    var meta = document.createElement('div');
    meta.className = 'meta';
    meta.style.display = 'none';

    wrap.appendChild(roleEl);
    wrap.appendChild(bubble);
    wrap.appendChild(meta);
    conversationEl.appendChild(wrap);

    var entry = { role: role, content: content, el: wrap, bubbleEl: bubble, metaEl: meta };
    history.push(entry);
    if (stick || role === 'user') scrollToBottom();
    return entry;
  }

  function setMeta(entry, kind, text) {
    entry.metaEl.className = 'meta ' + kind;
    entry.metaEl.textContent = text;
    entry.metaEl.style.display = text ? '' : 'none';
  }

  function showTyping(entry) {
    var t = document.createElement('span');
    t.className = 'typing';
    t.innerHTML = '<span></span><span></span><span></span>';
    entry.typingEl = t;
    entry.bubbleEl.style.display = 'block';
    entry.bubbleEl.appendChild(t);
  }
  function hideTyping(entry) {
    if (entry.typingEl) {
      if (entry.typingEl.parentNode) entry.typingEl.parentNode.removeChild(entry.typingEl);
      entry.typingEl = null;
    }
    entry.bubbleEl.style.display = '';
  }

  function removeAfter(index) {
    while (history.length > index + 1) {
      var e = history.pop();
      if (e.el.parentNode) e.el.parentNode.removeChild(e.el);
    }
  }

  /* ---------- streaming ---------- */

  function finish(stream, state, message) {
    // state: 'done' | 'stopped' | 'error'
    stream.ended = true;
    if (current === stream) current = null;
    var entry = stream.entry;
    hideTyping(entry);
    entry.el.classList.remove('streaming');
    var hasText = entry.content.length > 0;

    if (state === 'done') {
      setMeta(entry, 'done', hasText ? '' : '(The assistant returned an empty answer.)');
      entry.metaEl.style.display = hasText ? 'none' : '';
      setStatus('done', 'Finished');
    } else if (state === 'stopped') {
      entry.el.classList.add('stopped');
      setMeta(entry, 'stopped', hasText ? '■ Stopped — partial answer kept' : '■ Stopped before any text arrived');
      setStatus('stopped', 'Stopped');
    } else {
      entry.el.classList.add('error');
      setMeta(entry, 'error', '⚠ Error: ' + message + (hasText ? ' (partial answer kept)' : ''));
      setStatus('error', 'Error: ' + message);
    }
    updateControls();
  }

  function startStream() {
    // Build payload from history (skip empty assistant placeholders from stopped/failed runs)
    var payload = [];
    for (var i = 0; i < history.length; i++) {
      var m = history[i];
      if (m.role === 'assistant' && !m.content) continue;
      payload.push({ role: m.role, content: m.content });
    }

    var entry = addMessage('assistant', '');
    entry.el.classList.add('streaming');
    showTyping(entry);
    setMeta(entry, 'generating', 'Generating…');

    var stream = { aborted: false, ended: false, handle: null, entry: entry };
    current = stream;
    setStatus('generating', 'Generating answer…');
    updateControls();

    function onToken(text) {
      // Ignore anything that arrives after stop / regenerate / completion
      if (stream.aborted || stream.ended || current !== stream) return;
      if (typeof text !== 'string' || !text) return;
      var stick = nearBottom();
      if (entry.typingEl) hideTyping(entry);
      entry.content += text;
      entry.bubbleEl.textContent = entry.content;
      if (stick) scrollToBottom();
    }
    function onDone() {
      if (stream.aborted || stream.ended || current !== stream) return;
      finish(stream, 'done');
    }
    function onError(err) {
      if (stream.aborted || stream.ended || current !== stream) return;
      finish(stream, 'error', errorText(err));
    }

    try {
      if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
        throw new Error('Chat service is not available.');
      }
      stream.handle = window.ChatSDK.streamChat(payload, {
        onToken: onToken,
        onDone: onDone,
        onError: onError
      });
    } catch (e) {
      if (!stream.ended) finish(stream, 'error', errorText(e));
    }
  }

  function abortCurrent() {
    var s = current;
    if (!s) return null;
    s.aborted = true;
    current = null;
    try {
      if (s.handle && typeof s.handle.abort === 'function') s.handle.abort();
    } catch (e) { /* ignore abort failures; stream is already ignored locally */ }
    return s;
  }

  /* ---------- actions ---------- */

  function send() {
    if (current) return;
    var text = promptEl.value.trim();
    if (!text) return;
    promptEl.value = '';
    autoResize();
    addMessage('user', text);
    startStream();
    promptEl.focus();
  }

  function stop() {
    var s = abortCurrent();
    if (!s) return;
    finish(s, 'stopped');
  }

  function regenerate() {
    var idx = lastUserIndex();
    if (idx === -1) return;
    var s = abortCurrent();
    if (s) s.ended = true;
    removeAfter(idx);   // drop the previous answer (and any partial/error answer)
    startStream();
  }

  /* ---------- UI wiring ---------- */

  function autoResize() {
    promptEl.style.height = 'auto';
    promptEl.style.height = Math.min(promptEl.scrollHeight, 160) + 'px';
  }

  sendBtn.addEventListener('click', send);
  stopBtn.addEventListener('click', stop);
  regenBtn.addEventListener('click', regenerate);
  promptEl.addEventListener('input', autoResize);
  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });

  setStatus('idle', 'Ready');
  updateControls();
  promptEl.focus();
})();
</script>
</body>
</html>
