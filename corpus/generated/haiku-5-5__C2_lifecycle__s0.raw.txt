<script>
(function () {
  'use strict';

  var els = {
    transcript: document.getElementById('transcript'),
    empty: document.getElementById('empty'),
    prompt: document.getElementById('prompt'),
    send: document.getElementById('send'),
    stop: document.getElementById('stop'),
    regenerate: document.getElementById('regenerate'),
    status: document.getElementById('status')
  };

  // Conversation history sent to the model: {role, content}.
  var history = [];
  // The answer currently streaming, or null.
  var current = null;
  var messageSeq = 0;
  var announceTimer = null;

  // Status messages go to the polite status region only.
  // Clearing and then setting the text makes repeated messages announce again.
  function announce(text) {
    els.status.textContent = '';
    clearTimeout(announceTimer);
    announceTimer = setTimeout(function () {
      els.status.textContent = text;
    }, 80);
  }

  // aria-disabled (not the disabled attribute) keeps focus on the control
  // so keyboard focus is not lost when a button becomes unavailable.
  function setAriaDisabled(btn, on) {
    btn.setAttribute('aria-disabled', on ? 'true' : 'false');
  }

  function hasUserMessage() {
    for (var i = 0; i < history.length; i++) {
      if (history[i].role === 'user') return true;
    }
    return false;
  }

  function updateControls() {
    var busy = current !== null;
    setAriaDisabled(els.send, busy);
    setAriaDisabled(els.stop, !busy);
    setAriaDisabled(els.regenerate, busy || !hasUserMessage());
  }

  function updateEmptyState() {
    els.empty.hidden = els.transcript.querySelector('.msg') !== null;
  }

  function scrollToEnd(force) {
    var t = els.transcript;
    var nearBottom = t.scrollHeight - t.scrollTop - t.clientHeight < 80;
    if (force || nearBottom) {
      t.scrollTop = t.scrollHeight;
    }
  }

  // Builds a message article. Text is set with textContent only (no HTML injection).
  function buildMessage(role) {
    messageSeq += 1;
    var labelId = 'msg-label-' + messageSeq;

    var article = document.createElement('article');
    article.className = 'msg msg-' + role;
    article.setAttribute('aria-labelledby', labelId);

    var label = document.createElement('p');
    label.id = labelId;
    label.className = 'msg-label';
    label.textContent = role === 'user' ? 'You' : 'Assistant';

    var body = document.createElement('div');
    body.className = 'msg-body';
    var textNode = document.createTextNode('');
    body.appendChild(textNode);

    article.appendChild(label);
    article.appendChild(body);

    return { article: article, body: body, textNode: textNode, caret: null };
  }

  function appendUserMessage(text) {
    var m = buildMessage('user');
    m.textNode.data = text;
    els.transcript.appendChild(m.article);
    history.push({ role: 'user', content: text });
    updateEmptyState();
    scrollToEnd(true);
  }

  // Ends a request exactly once. Partial text is kept for stopped and errored answers.
  function settle(req, outcome, detail) {
    req.closed = true;
    if (current === req) current = null;

    if (req.caret) req.caret.remove();
    req.article.removeAttribute('aria-busy');

    if (req.text) {
      history.push({ role: 'assistant', content: req.text });
    }

    var note = '';
    var status = '';
    if (outcome === 'done') {
      status = 'Answer finished.';
    } else if (outcome === 'stopped') {
      note = 'Stopped. Partial answer kept.';
      status = 'Answer stopped. Partial answer kept.';
    } else {
      var reason = detail || 'Something went wrong.';
      note = 'Error: ' + reason;
      status = 'Error: ' + reason + ' Use Regenerate to try again.';
    }

    if (note) {
      var p = document.createElement('p');
      p.className = 'msg-note';
      p.textContent = note;
      req.article.appendChild(p);
    }

    updateControls();
    announce(status);
    scrollToEnd(false);
  }

  function startStream() {
    var m = buildMessage('assistant');
    var caret = document.createElement('span');
    caret.className = 'caret';
    caret.setAttribute('aria-hidden', 'true');
    m.body.appendChild(caret);
    m.caret = caret;
    m.article.setAttribute('aria-busy', 'true');
    els.transcript.appendChild(m.article);
    updateEmptyState();

    var req = {
      article: m.article,
      textNode: m.textNode,
      caret: caret,
      text: '',
      closed: false,
      handle: null
    };
    current = req;
    updateControls();
    announce('Generating answer.');
    scrollToEnd(true);

    var messages = history.map(function (x) {
      return { role: x.role, content: x.content };
    });

    try {
      if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
        throw new Error('Chat service is unavailable.');
      }
      req.handle = window.ChatSDK.streamChat(messages, {
        onToken: function (chunk) {
          // Ignore chunks from a stopped or finished answer, including ones that were in flight.
          if (req.closed) return;
          req.text += String(chunk);
          req.textNode.data = req.text;
          scrollToEnd(false);
        },
        onDone: function () {
          if (req.closed) return;
          settle(req, 'done');
        },
        onError: function (err) {
          if (req.closed) return;
          var msg = (err && err.message) ? err.message : 'The answer could not be completed.';
          settle(req, 'error', msg);
        }
      });
    } catch (err) {
      if (!req.closed) {
        settle(req, 'error', (err && err.message) ? err.message : 'The answer could not be started.');
      }
    }
  }

  function sendPrompt() {
    if (current !== null) {
      announce('An answer is still generating. Press Stop to end it first.');
      return;
    }
    var text = els.prompt.value.trim();
    if (!text) {
      announce('Type a message before sending.');
      return;
    }
    // Focus stays where it is: the textarea when sent with Enter, the Send button when clicked.
    els.prompt.value = '';
    appendUserMessage(text);
    startStream();
  }

  function stopStream() {
    var req = current;
    if (!req || req.closed) return;
    req.closed = true;
    try {
      if (req.handle && typeof req.handle.abort === 'function') {
        req.handle.abort();
      }
    } catch (e) {
      // The partial answer is kept regardless of abort errors.
    }
    settle(req, 'stopped');
  }

  function regenerate() {
    if (current !== null) {
      announce('An answer is still generating. Press Stop to end it first.');
      return;
    }
    if (!hasUserMessage()) {
      announce('There is no message to regenerate.');
      return;
    }

    // Remove the last assistant bubble (including an empty or errored one) and its history entry.
    var lastChild = els.transcript.lastElementChild;
    if (lastChild && lastChild.classList.contains('msg-assistant')) {
      lastChild.remove();
    }
    if (history.length && history[history.length - 1].role === 'assistant') {
      history.pop();
    }

    updateEmptyState();
    startStream();
  }

  els.send.addEventListener('click', sendPrompt);
  els.stop.addEventListener('click', function () {
    if (current === null) {
      announce('No answer is generating.');
      return;
    }
    stopStream();
  });
  els.regenerate.addEventListener('click', regenerate);

  els.prompt.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      sendPrompt();
    }
  });

  updateControls();
  updateEmptyState();
  els.prompt.focus();
})();
</script>
</body>
</html>
