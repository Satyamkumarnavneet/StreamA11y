<script>
(function () {
  'use strict';

  var form = document.getElementById('composer');
  var promptEl = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var transcriptEl = document.getElementById('transcript');
  var regionEl = document.getElementById('transcript-region');
  var emptyNote = document.getElementById('empty-note');
  var statusEl = document.getElementById('status');

  // Conversation history: [{role, content, el}]
  var history = [];
  // The single active generation, or null.
  var current = null;
  var generationCounter = 0;
  var messageCounter = 0;
  var statusTimer = null;

  function setStatus(text, kind) {
    // Clear first, then set, so that repeated identical messages are still announced.
    if (statusTimer) { clearTimeout(statusTimer); statusTimer = null; }
    statusEl.textContent = '';
    statusEl.setAttribute('data-kind', kind || 'idle');
    statusTimer = setTimeout(function () {
      statusEl.textContent = text;
      statusTimer = null;
    }, 50);
  }

  // aria-disabled keeps buttons focusable, so focus is never lost when state changes.
  function setDisabled(btn, disabled) {
    btn.setAttribute('aria-disabled', disabled ? 'true' : 'false');
  }
  function isDisabled(btn) {
    return btn.getAttribute('aria-disabled') === 'true';
  }

  function lastUserIndex() {
    for (var i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') return i;
    }
    return -1;
  }

  function updateControls() {
    var streaming = current !== null;
    setDisabled(sendBtn, streaming);
    setDisabled(stopBtn, !streaming);
    setDisabled(regenBtn, streaming || lastUserIndex() === -1);
    emptyNote.hidden = transcriptEl.children.length > 0;
  }

  function isNearBottom() {
    return regionEl.scrollHeight - regionEl.scrollTop - regionEl.clientHeight < 40;
  }
  function scrollToBottom() {
    regionEl.scrollTop = regionEl.scrollHeight;
  }

  function createMessageElement(role) {
    var id = 'msg-' + (++messageCounter);
    var li = document.createElement('li');
    li.className = 'msg ' + role;
    var h = document.createElement('h3');
    h.id = id + '-label';
    h.textContent = role === 'user' ? 'You' : 'Assistant';
    var content = document.createElement('p');
    content.className = 'content';
    var state = document.createElement('span');
    state.className = 'state';
    li.setAttribute('aria-labelledby', h.id);
    li.appendChild(h);
    li.appendChild(content);
    li.appendChild(state);
    transcriptEl.appendChild(li);
    return { li: li, content: content, state: state };
  }

  function addUserMessage(text) {
    var stick = true;
    var parts = createMessageElement('user');
    parts.content.textContent = text;
    history.push({ role: 'user', content: text, el: parts.li });
    if (stick) scrollToBottom();
  }

  function setMessageState(parts, text, cls) {
    parts.state.textContent = text;
    parts.state.className = 'state' + (cls ? ' ' + cls : '');
  }

  function messagesForApi() {
    return history.map(function (m) { return { role: m.role, content: m.content }; });
  }

  function startGeneration() {
    if (current) return;
    var payload = messagesForApi();
    var parts = createMessageElement('assistant');
    parts.li.setAttribute('aria-busy', 'true');
    setMessageState(parts, 'Generating…', '');
    var gen = {
      id: ++generationCounter,
      parts: parts,
      text: '',
      textNode: document.createTextNode(''),
      handle: null
    };
    parts.content.appendChild(gen.textNode);
    current = gen;
    updateControls();
    scrollToBottom();
    setStatus('Assistant is generating an answer.', 'busy');

    function isActive() { return current !== null && current.id === gen.id; }

    try {
      var handle = window.ChatSDK.streamChat(payload, {
        onToken: function (text) {
          // Ignore chunks from stopped, failed, finished, or replaced generations.
          if (!isActive()) return;
          if (typeof text !== 'string' || text === '') return;
          var stick = isNearBottom();
          gen.text += text;
          gen.textNode.appendData(text);
          if (stick) scrollToBottom();
        },
        onDone: function () {
          if (!isActive()) return;
          finish(gen, 'done');
        },
        onError: function (err) {
          if (!isActive()) return;
          finish(gen, 'error', err);
        }
      });
      gen.handle = handle;
    } catch (err) {
      if (isActive()) finish(gen, 'error', err);
    }
  }

  function finish(gen, outcome, err) {
    if (!current || current.id !== gen.id) return;
    current = null; // From here on, any late chunk is ignored.
    var parts = gen.parts;
    parts.li.removeAttribute('aria-busy');

    if (gen.text.length > 0) {
      history.push({ role: 'assistant', content: gen.text, el: parts.li, outcome: outcome });
    } else {
      // Keep the element visible as a record, but it carries no content for the API.
      parts.content.textContent = outcome === 'stopped' ? '(No text was generated.)' :
        outcome === 'error' ? '(No answer.)' : '(Empty answer.)';
      parts.li.setAttribute('data-empty', 'true');
    }

    if (outcome === 'done') {
      setMessageState(parts, 'Finished.', 'done');
      setStatus('Answer finished.', 'idle');
    } else if (outcome === 'stopped') {
      setMessageState(parts, 'Stopped. Partial answer kept.', '');
      setStatus('Generation stopped. The partial answer was kept.', 'idle');
    } else {
      var msg = err && err.message ? err.message : (err ? String(err) : 'Unknown error');
      setMessageState(parts, 'Error: ' + msg, 'error');
      setStatus('Error: the answer could not be completed. ' + msg, 'error');
    }
    updateControls();
  }

  function stopGeneration() {
    if (!current) return;
    var gen = current;
    try {
      if (gen.handle && typeof gen.handle.abort === 'function') gen.handle.abort();
    } catch (e) { /* ignore abort errors; we still finalize locally */ }
    finish(gen, 'stopped');
  }

  function send() {
    if (current) {
      setStatus('Please wait for the current answer to finish, or stop it first.', 'busy');
      return;
    }
    var text = promptEl.value.trim();
    if (!text) {
      setStatus('Type a message before sending.', 'idle');
      return;
    }
    promptEl.value = '';
    addUserMessage(text);
    startGeneration();
  }

  function regenerate() {
    if (current) return;
    var idx = lastUserIndex();
    if (idx === -1) {
      setStatus('There is no message to regenerate an answer for.', 'idle');
      return;
    }
    // Remove assistant answers after the last user message, from both history and the transcript.
    var lastUserEl = history[idx].el;
    history = history.slice(0, idx + 1);
    while (lastUserEl.nextSibling) {
      transcriptEl.removeChild(lastUserEl.nextSibling);
    }
    startGeneration();
  }

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    if (isDisabled(sendBtn)) {
      setStatus('Please wait for the current answer to finish, or stop it first.', 'busy');
      return;
    }
    send();
  });

  stopBtn.addEventListener('click', function () {
    if (isDisabled(stopBtn)) return;
    stopGeneration();
  });

  regenBtn.addEventListener('click', function () {
    if (isDisabled(regenBtn)) return;
    regenerate();
  });

  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      if (isDisabled(sendBtn)) {
        setStatus('Please wait for the current answer to finish, or stop it first.', 'busy');
        return;
      }
      send();
    } else if (e.key === 'Escape' && current) {
      e.preventDefault();
      stopGeneration();
    }
  });

  if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
    setStatus('Error: the chat service is not available.', 'error');
  }
  updateControls();
})();
</script>
</body>
</html>
