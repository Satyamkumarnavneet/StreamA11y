<script>
(function () {
  'use strict';

  var form = document.getElementById('chat-form');
  var promptEl = document.getElementById('prompt');
  var promptError = document.getElementById('prompt-error');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var listEl = document.getElementById('conversation');
  var regionEl = document.getElementById('conversation-region');
  var emptyEl = document.getElementById('empty');
  var statusBar = document.getElementById('status');
  var statusText = document.getElementById('status-text');
  var alertEl = document.getElementById('alert');

  // Conversation state: {role, content, state: 'complete'|'streaming'|'stopped'|'error', el, contentEl, noteEl}
  var messages = [];
  // Active stream: {id, handle, msg} or null
  var active = null;
  var nextStreamId = 1;

  function announceStatus(state, text) {
    statusBar.setAttribute('data-state', state);
    // Clear first so repeated identical messages are re-announced.
    statusText.textContent = '';
    window.setTimeout(function () { statusText.textContent = text; }, 50);
  }

  function announceAlert(text) {
    alertEl.textContent = '';
    window.setTimeout(function () { alertEl.textContent = text; }, 50);
  }

  function isNearBottom() {
    return regionEl.scrollHeight - regionEl.scrollTop - regionEl.clientHeight < 60;
  }
  function scrollToBottom() { regionEl.scrollTop = regionEl.scrollHeight; }

  function renderMessage(msg) {
    var li = document.createElement('li');
    li.className = 'message ' + msg.role;
    var h = document.createElement('h3');
    h.textContent = msg.role === 'user' ? 'You' : 'Assistant';
    var content = document.createElement('p');
    content.className = 'content';
    content.textContent = msg.content;
    var note = document.createElement('span');
    note.className = 'note';
    note.hidden = true;
    li.appendChild(h);
    li.appendChild(content);
    li.appendChild(note);
    msg.el = li;
    msg.contentEl = content;
    msg.noteEl = note;
    listEl.appendChild(li);
    emptyEl.hidden = true;
    return li;
  }

  function setNote(msg, text, isError) {
    msg.noteEl.textContent = text;
    msg.noteEl.hidden = !text;
    msg.el.classList.toggle('error', !!isError);
  }

  function addMessage(role, content, state) {
    var msg = { role: role, content: content, state: state || 'complete' };
    messages.push(msg);
    renderMessage(msg);
    return msg;
  }

  function removeMessage(msg) {
    var i = messages.indexOf(msg);
    if (i !== -1) messages.splice(i, 1);
    if (msg.el && msg.el.parentNode) msg.el.parentNode.removeChild(msg.el);
    emptyEl.hidden = messages.length > 0;
  }

  function lastUserIndex() {
    for (var i = messages.length - 1; i >= 0; i--) {
      if (messages[i].role === 'user') return i;
    }
    return -1;
  }

  function historyForApi(upTo) {
    var out = [];
    for (var i = 0; i < upTo; i++) {
      var m = messages[i];
      if (m.state === 'streaming') continue;
      if (m.role === 'assistant' && m.content.trim() === '') continue;
      out.push({ role: m.role, content: m.content });
    }
    return out;
  }

  function updateControls() {
    var streaming = !!active;
    var focused = document.activeElement;
    sendBtn.disabled = streaming;
    stopBtn.disabled = !streaming;
    regenBtn.disabled = streaming || lastUserIndex() === -1;
    // Never leave keyboard focus on a control that just became disabled.
    if (focused && focused.disabled) promptEl.focus();
  }

  function finishStream(state, errorText) {
    if (!active) return;
    var msg = active.msg;
    active = null;
    msg.state = state;
    msg.el.removeAttribute('aria-busy');
    if (state === 'complete') {
      setNote(msg, '', false);
      announceStatus('done', 'Answer finished.');
    } else if (state === 'stopped') {
      setNote(msg, msg.content ? 'Stopped. Partial answer kept.' : 'Stopped before any text was received.', false);
      announceStatus('stopped', 'Stopped. The partial answer was kept.');
    } else if (state === 'error') {
      var text = 'Error: ' + (errorText || 'The answer could not be completed.');
      setNote(msg, text + ' Use Regenerate to try again.', true);
      announceStatus('error', text);
      announceAlert(text + ' Use Regenerate to try again.');
    }
    updateControls();
  }

  function startAssistant(history) {
    var msg = addMessage('assistant', '', 'streaming');
    msg.el.setAttribute('aria-busy', 'true');
    setNote(msg, 'Generating…', false);
    var id = nextStreamId++;
    active = { id: id, handle: null, msg: msg };
    updateControls();
    announceStatus('generating', 'Assistant is generating an answer…');
    scrollToBottom();

    function isCurrent() { return active && active.id === id; }

    var callbacks = {
      onToken: function (text) {
        // Ignores late tokens that arrive after abort() or after the stream ended.
        if (!isCurrent()) return;
        var stick = isNearBottom();
        msg.content += String(text == null ? '' : text);
        msg.contentEl.textContent = msg.content;
        if (stick) scrollToBottom();
      },
      onDone: function () {
        if (!isCurrent()) return;
        finishStream('complete');
      },
      onError: function (err) {
        if (!isCurrent()) return;
        var t = err && err.message ? err.message : (err ? String(err) : '');
        finishStream('error', t);
      }
    };

    try {
      if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
        throw new Error('The chat service is not available.');
      }
      var handle = window.ChatSDK.streamChat(history, callbacks);
      if (isCurrent()) active.handle = handle;
    } catch (e) {
      if (isCurrent()) finishStream('error', e && e.message ? e.message : String(e));
    }
  }

  function stopStream() {
    if (!active) return;
    var handle = active.handle;
    // Mark stopped first so any in-flight token is ignored.
    finishStream('stopped');
    try {
      if (handle && typeof handle.abort === 'function') handle.abort();
    } catch (e) { /* ignore abort failures; UI already stopped */ }
  }

  function clearPromptError() {
    promptError.textContent = '';
    promptEl.removeAttribute('aria-invalid');
  }

  function send() {
    if (active) return;
    var text = promptEl.value;
    if (text.trim() === '') {
      promptEl.setAttribute('aria-invalid', 'true');
      promptError.textContent = 'Please type a message before sending.';
      promptEl.focus();
      return;
    }
    clearPromptError();
    addMessage('user', text, 'complete');
    promptEl.value = '';
    promptEl.focus();
    startAssistant(historyForApi(messages.length));
  }

  function regenerate() {
    if (active) return;
    var idx = lastUserIndex();
    if (idx === -1) return;
    // Remove all assistant answers after the last user message.
    var toRemove = messages.slice(idx + 1);
    toRemove.forEach(removeMessage);
    promptEl.focus();
    startAssistant(historyForApi(idx + 1));
  }

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    send();
  });

  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      if (active) {
        announceStatus('generating', 'Please wait for the current answer, or press Stop.');
        return;
      }
      send();
    } else if (e.key === 'Escape' && active) {
      e.preventDefault();
      stopStream();
    }
  });

  promptEl.addEventListener('input', function () {
    if (promptEl.getAttribute('aria-invalid') === 'true' && promptEl.value.trim() !== '') {
      clearPromptError();
    }
  });

  stopBtn.addEventListener('click', function () {
    stopStream();
    promptEl.focus();
  });

  regenBtn.addEventListener('click', regenerate);

  updateControls();
})();
</script>
</body>
</html>
