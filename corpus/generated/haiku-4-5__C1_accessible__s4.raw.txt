    <script>
        // State management
        const state = {
            messages: [],
            currentHandle: null,
            isGenerating: false,
            lastUserMessageIndex: null
        };

        // DOM elements
        const promptInput = document.getElementById('prompt');
        const sendButton = document.getElementById('send');
        const stopButton = document.getElementById('stop');
        const regenerateButton = document.getElementById('regenerate');
        const messagesContainer = document.getElementById('messages');
        const statusElement = document.getElementById('status');
        const emptyState = document.getElementById('emptyState');

        // Utility functions
        function showStatus(message, type = 'info') {
            statusElement.textContent = message;
            statusElement.className = `status show ${type}`;
        }

        function hideStatus() {
            statusElement.className = 'status';
        }

        function updateButtonStates() {
            const hasMessages = state.messages.length > 0;
            const canSend = promptInput.value.trim().length > 0 && !state.isGenerating;

            sendButton.disabled = !canSend;
            stopButton.disabled = !state.isGenerating;
            regenerateButton.disabled = state.isGenerating || state.lastUserMessageIndex === null;
        }

        function addMessageToDOM(role, content) {
            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${role}`;

            const roleLabel = document.createElement('div');
            roleLabel.className = 'message-role';
            roleLabel.textContent = role === 'user' ? 'You' : 'Assistant';

            const contentDiv = document.createElement('div');
            contentDiv.className = 'message-content';
            contentDiv.textContent = content;

            messageDiv.appendChild(roleLabel);
            messageDiv.appendChild(contentDiv);
            messagesContainer.appendChild(messageDiv);

            // Remove empty state if present
            if (emptyState.parentNode) {
                emptyState.remove();
            }

            // Scroll to bottom
            messagesContainer.scrollTop = messagesContainer.scrollHeight;

            return contentDiv;
        }

        function findOrCreateLastAssistantMessage() {
            const messages = messagesContainer.querySelectorAll('.message.assistant');
            if (messages.length > 0) {
                const lastAssistant = messages[messages.length - 1];
                return lastAssistant.querySelector('.message-content');
            }
            return null;
        }

        async function sendMessage() {
            const text = promptInput.value.trim();
            if (!text || state.isGenerating) return;

            // Add user message to state and DOM
            state.messages.push({ role: 'user', content: text });
            state.lastUserMessageIndex = state.messages.length - 1;
            addMessageToDOM('user', text);

            promptInput.value = '';
            updateButtonStates();

            // Stream assistant response
            await streamAssistantResponse();
        }

        async function streamAssistantResponse() {
            state.isGenerating = true;
            updateButtonStates();
            showStatus('Assistant is generating...', 'generating');

            // Add empty assistant message
            state.messages.push({ role: 'assistant', content: '' });
            const contentElement = addMessageToDOM('assistant', '');

            let currentContent = '';

            const onToken = (text) => {
                currentContent += text;
                state.messages[state.messages.length - 1].content = currentContent;
                contentElement.textContent = currentContent;
                messagesContainer.scrollTop = messagesContainer.scrollHeight;
            };

            const onDone = () => {
                state.isGenerating = false;
                state.currentHandle = null;
                updateButtonStates();
                showStatus('Response complete', 'success');
                setTimeout(hideStatus, 3000);
            };

            const onError = (error) => {
                state.isGenerating = false;
                state.currentHandle = null;
                updateButtonStates();
                const errorMessage = error.message || 'An error occurred while generating the response';
                showStatus(`Error: ${errorMessage}`, 'error');
            };

            try {
                state.currentHandle = window.ChatSDK.streamChat(state.messages, {
                    onToken,
                    onDone,
                    onError
                });
            } catch (error) {
                onError(error);
            }
        }

        function stopGeneration() {
            if (state.currentHandle) {
                state.currentHandle.abort();
                state.isGenerating = false;
                state.currentHandle = null;
                updateButtonStates();
                showStatus('Response stopped', 'info');
                setTimeout(hideStatus, 3000);
            }
        }

        function regenerateResponse() {
            if (state.lastUserMessageIndex === null || state.isGenerating) return;

            // Remove the last assistant message if it exists
            if (state.messages[state.messages.length - 1].role === 'assistant') {
                state.messages.pop();
            }

            // Remove last assistant message from DOM
            const assistantMessages = messagesContainer.querySelectorAll('.message.assistant');
            if (assistantMessages.length > 0) {
                assistantMessages[assistantMessages.length - 1].remove();
            }

            // Stream new response
            streamAssistantResponse();
        }

        // Event listeners
        sendButton.addEventListener('click', sendMessage);
        stopButton.addEventListener('click', stopGeneration);
        regenerateButton.addEventListener('click', regenerateResponse);

        promptInput.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' && !e.shiftKey) {
                e.preventDefault();
                sendMessage();
            }
        });

        promptInput.addEventListener('input', updateButtonStates);

        // Initialize button states
        updateButtonStates();

        // Focus management for accessibility
        promptInput.focus();
    </script>
</body>
