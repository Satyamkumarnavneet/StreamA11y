<script>
(function () {
  'use strict';

  var transcript = document.getElementById('transcript');
  var list = document.getElementById('messages');
  var emptyNote = document.getElementById('empty');
  var statusBox = document.getElementById('status');
  var statusIcon = document.getElementById('status-icon');
  var statusText = document.getElementById('status-text');
  var composer = document.getElementById('composer');
  var prompt = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');

  var messages = [];   // {role, content, state, el, textNode, badge, note, err}
  var current = null;  // active generation: {msg, handle, closed}

  var ICONS = { idle: '○', generating: '●', stopped: '■', done: '✓', error: '!' };
  var BADGES = { streaming: '— Generating', stopped: '— Stopped', done: '— Finished', error: '— Error' };

  function setStatus(kind, text) {
    statusBox.setAttribute('data-kind', kind);
    statusIcon.textContent = ICONS[kind] || '○';
    statusText.textContent = text;
  }

  function lastUserIndex() {
    for (var i = messages.length - 1; i >= 0; i--) {
      if (messages[i].role === 'user') return i;
    }
    return -1;
  }

  // Buttons use aria-disabled (not the disabled attribute) so a focused button
  // never loses keyboard focus when its state changes.
  function updateControls() {
    var generating = !!current;
    sendBtn.setAttribute('aria-disabled', generating ? 'true' : 'false');
    stopBtn.setAttribute('aria-disabled', generating ? 'false' : 'true');
    regenBtn.setAttribute('aria-disabled', lastUserIndex() === -1 ? 'true' : 'false');
    emptyNote.hidden = messages.length > 0;
  }

  function isDisabled(btn) { return btn.getAttribute('aria-disabled') === 'true'; }

  function withAutoScroll(fn) {
    var near = transcript.scrollHeight - transcript.scrollTop - transcript.clientHeight < 48;
    fn();
    if (near) transcript.scrollTop = transcript.scrollHeight;
  }

  function buildMessage(role, content) {
    var m = { role: role, content: content, state: role === 'assistant' ? 'streaming' : 'done' };
    var li = document.createElement('li');
    li.className = 'message ' + role;

    var meta = document.createElement('div');
    meta.className = 'meta';
    var who = document.createElement('span');
    who.className = 'who';
    who.textContent = role === 'user' ? 'You' : 'Assistant';
    meta.appendChild(who);

    if (role === 'assistant') {
      meta.appendChild(document.createTextNode(' '));
      m.badge = document.createElement('span');
      m.badge.className = 'badge';
      meta.appendChild(m.badge);
    }
    li.appendChild(meta);

    if (role === 'assistant') {
      m.note = document.createElement('p');
      m.note.className = 'note';
      li.appendChild(m.note);
    }

    var body = document.createElement('div');
    body.className = 'text';
    m.textNode = document.createTextNode(content);
    body.appendChild(m.textNode);
    li.appendChild(body);

    if (role === 'assistant') {
      m.err = document.createElement('p');
      m.err.className = 'err';
      m.err.hidden = true;
      li.appendChild(m.err);
    }

    m.el = li;
    return m;
  }

  function refreshMessage(m) {
    if (m.role !== 'assistant') return;
    m.el.setAttribute('data-state', m.state);
    // aria-busy only while this answer streams; always removed otherwise.
    if (m.state === 'streaming') m.el.setAttribute('aria-busy', 'true');
    else m.el.removeAttribute('aria-busy');
    m.badge.textContent = BADGES[m.state] || '';
    if (m.content.length > 0) {
      m.note.hidden = true;
    } else {
      m.note.hidden = false;
      m.note.textContent =
        m.state === 'streaming' ? 'Waiting for the first words…' :
        m.state === 'stopped' ? 'Stopped before any text arrived.' :
        m.state === 'error' ? 'No text was received.' : '(Empty answer)';
    }
  }

  function addMessage(m) {
    messages.push(m);
    refreshMessage(m);
    list.appendChild(m.el);
    transcript.scrollTop = transcript.scrollHeight;
  }

  function apiMessages() {
    var out = [];
    for (var i = 0; i < messages.length; i++) {
      var m = messages[i];
      if (m.role === 'assistant' && !m.content) continue;
      out.push({ role: m.role, content: m.content });
    }
    return out;
  }

  // Detach the active generation so nothing it delivers can change the transcript.
  function detachCurrent() {
    var g = current;
    if (!g) return null;
    g.closed = true;
    current = null;
    try { if (g.handle && typeof g.handle.abort === 'function') g.handle.abort(); } catch (e) { /* ignore */ }
    return g;
  }

  function finish(g, state) {
    g.closed = true;
    if (current === g) current = null;
    g.msg.state = state;
    refreshMessage(g.msg);
    updateControls();
  }

  function startGeneration(statusMessage) {
    var m = buildMessage('assistant', '');
    var g = { msg: m, handle: null, closed: false };
    current = g;
    addMessage(m);
    updateControls();
    setStatus('generating', statusMessage);

    var payload = apiMessages();

    function onToken(text) {
      if (g.closed || current !== g) return; // ignore chunks from stopped/replaced answers
      var t = (text === undefined || text === null) ? '' : String(text);
      if (!t) return;
      var wasEmpty = m.content.length === 0;
      withAutoScroll(function () {
        m.content += t;
        m.textNode.appendData(t); // append only; no re-render, no live region
        if (wasEmpty) refreshMessage(m);
      });
    }
    function onDone() {
      if (g.closed || current !== g) return;
      finish(g, 'done');
      setStatus('done', m.content ? 'Answer finished.' : 'Answer finished, but it was empty.');
    }
    function onError(error) { fail(g, error); }

    try {
      g.handle = window.ChatSDK.streamChat(payload, { onToken: onToken, onDone: onDone, onError: onError });
    } catch (e) {
      fail(g, e);
    }
  }

  function fail(g, error) {
    if (g.closed || current !== g) return;
    var msg = (error && error.message) ? error.message : (error ? String(error) : 'Unknown error');
    finish(g, 'error');
    g.msg.err.textContent = 'Error: ' + msg;
    g.msg.err.hidden = false;
    setStatus('error', 'Error: ' + msg + '. ' +
      (g.msg.content ? 'The partial answer was kept. ' : '') +
      'Use Regenerate to try again.');
  }

  function send() {
    if (current) {
      setStatus('generating', 'Still generating. Press Stop to interrupt, then send your message.');
      return;
    }
    var text = prompt.value.trim();
    if (!text) {
      setStatus('error', 'Please type a message before sending.');
      return;
    }
    if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
      setStatus('error', 'Error: the chat service is not available.');
      return;
    }
    addMessage(buildMessage('user', text));
    prompt.value = '';
    startGeneration('Generating answer…');
  }

  function stop() {
    if (!current) return;
    var g = detachCurrent();
    g.msg.state = 'stopped';
    refreshMessage(g.msg);
    updateControls();
    setStatus('stopped', g.msg.content
      ? 'Generation stopped. The partial answer was kept.'
      : 'Generation stopped before any text arrived.');
  }

  function regenerate() {
    var idx = lastUserIndex();
    if (idx === -1) return;
    if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
      setStatus('error', 'Error: the chat service is not available.');
      return;
    }
    detachCurrent();
    while (messages.length > idx + 1) {
      var removed = messages.pop();
      if (removed.el.parentNode) removed.el.parentNode.removeChild(removed.el);
    }
    startGeneration('Regenerating answer…');
  }

  composer.addEventListener('submit', function (e) {
    e.preventDefault();
    send();
  });
  sendBtn.addEventListener('click', function (e) {
    if (isDisabled(sendBtn)) {
      e.preventDefault();
      setStatus('generating', 'Still generating. Press Stop to interrupt, then send your message.');
    }
  });
  stopBtn.addEventListener('click', function () {
    if (isDisabled(stopBtn)) return;
    stop();
  });
  regenBtn.addEventListener('click', function () {
    if (isDisabled(regenBtn)) return;
    regenerate();
  });
  prompt.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.ctrlKey && !e.altKey && !e.metaKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });

  updateControls();
})();
</script>
</body>
</html>
