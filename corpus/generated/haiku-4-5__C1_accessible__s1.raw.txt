    <script>
        class ChatAssistant {
            constructor() {
                this.messages = [];
                this.isStreaming = false;
                this.currentStreamHandle = null;
                this.currentAssistantMessage = null;

                this.elements = {
                    prompt: document.getElementById('prompt'),
                    send: document.getElementById('send'),
                    stop: document.getElementById('stop'),
                    regenerate: document.getElementById('regenerate'),
                    messagesContainer: document.getElementById('messages'),
                    errorContainer: document.getElementById('error-container')
                };

                this.initializeEventListeners();
            }

            initializeEventListeners() {
                this.elements.send.addEventListener('click', () => this.sendMessage());
                this.elements.stop.addEventListener('click', () => this.stopStreaming());
                this.elements.regenerate.addEventListener('click', () => this.regenerateLastResponse());

                this.elements.prompt.addEventListener('keydown', (e) => {
                    if ((e.ctrlKey || e.metaKey) && e.key === 'Enter') {
                        e.preventDefault();
                        this.sendMessage();
                    }
                });

                this.elements.prompt.addEventListener('input', () => {
                    this.updateSendButton();
                });
            }

            updateSendButton() {
                const hasText = this.elements.prompt.value.trim().length > 0;
                this.elements.send.disabled = !hasText || this.isStreaming;
            }

            clearError() {
                this.elements.errorContainer.innerHTML = '';
                this.elements.errorContainer.setAttribute('role', 'alert');
                this.elements.errorContainer.setAttribute('aria-live', 'polite');
            }

            showError(message) {
                this.clearError();
                const errorDiv = document.createElement('div');
                errorDiv.className = 'error-message';
                errorDiv.textContent = `Error: ${message}`;
                this.elements.errorContainer.appendChild(errorDiv);
            }

            async sendMessage() {
                const userInput = this.elements.prompt.value.trim();
                if (!userInput || this.isStreaming) return;

                this.clearError();
                this.removeEmptyState();

                const userMessage = {
                    role: 'user',
                    content: userInput
                };

                this.messages.push(userMessage);
                this.addMessageToUI(userMessage);

                this.elements.prompt.value = '';
                this.updateSendButton();

                await this.streamAssistantResponse();
            }

            async streamAssistantResponse() {
                if (this.isStreaming) return;

                this.isStreaming = true;
                this.updateUIState();

                this.currentAssistantMessage = {
                    role: 'assistant',
                    content: ''
                };

                const messageElement = this.addMessageToUI(this.currentAssistantMessage, true);
                const contentElement = messageElement.querySelector('.message-content');
                const statusElement = messageElement.querySelector('.status-indicator');

                try {
                    this.currentStreamHandle = window.ChatSDK.streamChat(
                        this.messages,
                        {
                            onToken: (text) => {
                                this.currentAssistantMessage.content += text;
                                contentElement.textContent = this.currentAssistantMessage.content;
                                this.scrollToBottom();
                            },
                            onDone: () => {
                                this.isStreaming = false;
                                this.currentStreamHandle = null;

                                if (this.currentAssistantMessage) {
                                    this.messages.push(this.currentAssistantMessage);
                                }

                                this.updateStatusIndicator(statusElement, 'done');
                                this.updateUIState();
                                this.announceCompletion('Response generation completed');
                            },
                            onError: (error) => {
                                this.isStreaming = false;
                                this.currentStreamHandle = null;

                                this.updateStatusIndicator(statusElement, 'error');
                                this.showError(error.message || 'An error occurred during generation');
                                this.updateUIState();
                                this.announceCompletion('Response generation failed with an error');
                            }
                        }
                    );

                    this.updateStatusIndicator(statusElement, 'generating');
                } catch (error) {
                    this.isStreaming = false;
                    this.currentStreamHandle = null;
                    this.showError(error.message || 'Failed to start message stream');
                    this.updateStatusIndicator(statusElement, 'error');
                    this.updateUIState();
                }
            }

            stopStreaming() {
                if (!this.isStreaming || !this.currentStreamHandle) return;

                this.currentStreamHandle.abort();
                this.currentStreamHandle = null;
                this.isStreaming = false;

                const messageElement = this.elements.messagesContainer.querySelector('.message:last-child');
                if (messageElement) {
                    const statusElement = messageElement.querySelector('.status-indicator');
                    this.updateStatusIndicator(statusElement, 'stopped');
                }

                this.updateUIState();
                this.announceCompletion('Response generation stopped by user');
            }

            regenerateLastResponse() {
                if (this.messages.length === 0) return;

                const lastMessage = this.messages[this.messages.length - 1];
                if (lastMessage.role !== 'assistant') return;

                this.messages.pop();
                const messageElements = this.elements.messagesContainer.querySelectorAll('.message');
                if (messageElements.length > 0) {
                    messageElements[messageElements.length - 1].remove();
                }

                this.currentAssistantMessage = null;
                this.clearError();
                this.streamAssistantResponse();
            }

            removeEmptyState() {
                const emptyState = this.elements.messagesContainer.querySelector('.empty-state');
                if (emptyState) {
                    emptyState.remove();
                }
            }

            addMessageToUI(message, isTemporary = false) {
                const messageElement = document.createElement('div');
                messageElement.className = `message ${message.role}`;
                messageElement.setAttribute('role', 'article');

                const avatarElement = document.createElement('div');
                avatarElement.className = 'message-avatar';
                avatarElement.setAttribute('aria-hidden', 'true');
                avatarElement.textContent = message.role === 'user' ? 'You' : 'AI';

                const contentElement = document.createElement('div');
                contentElement.className = 'message-content';
                contentElement.textContent = message.content;

                messageElement.appendChild(avatarElement);
                messageElement.appendChild(contentElement);

                if (isTemporary) {
                    const statusElement = document.createElement('div');
                    statusElement.className = 'status-indicator';
                    statusElement.setAttribute('aria-live', 'polite');
                    statusElement.setAttribute('aria-label', 'Assistant generating response');

                    const statusDot = document.createElement('span');
                    statusDot.className = 'status-dot generating';

                    statusElement.appendChild(statusDot);
                    statusElement.appendChild(document.createTextNode('Generating...'));

                    messageElement.appendChild(statusElement);
                }

                this.elements.messagesContainer.appendChild(messageElement);
                this.scrollToBottom();

                return messageElement;
            }

            updateStatusIndicator(statusElement, status) {
                const statusDot = statusElement.querySelector('.status-dot');
                const statusTextNode = statusElement.childNodes[1];

                statusDot.className = 'status-dot';

                if (status === 'generating') {
                    statusDot.classList.add('generating');
                    statusTextNode.textContent = 'Generating...';
                } else if (status === 'done') {
                    statusDot.classList.add('done');
                    statusTextNode.textContent = 'Completed';
                } else if (status === 'error') {
                    statusDot.classList.add('error');
                    statusTextNode.textContent = 'Error';
                } else if (status === 'stopped') {
                    statusElement.style.color = '#ff9800';
                    statusDot.style.backgroundColor = '#ff9800';
                    statusTextNode.textContent = 'Stopped';
                }

                statusElement.setAttribute('aria-label', `Assistant response ${status}`);
            }

            updateUIState() {
                const hasMessages = this.messages.length > 0;
                const hasUserMessages = this.messages.some(m => m.role === 'user');

                this.elements.send.disabled = !this.elements.prompt.value.trim() || this.isStreaming;
                this.elements.stop.disabled = !this.isStreaming;
                this.elements.regenerate.disabled = !hasUserMessages || this.isStreaming;
            }

            scrollToBottom() {
                this.elements.messagesContainer.scrollTop = this.elements.messagesContainer.scrollHeight;
            }

            announceCompletion(message) {
                const announcement = document.createElement('div');
                announcement.className = 'sr-only';
                announcement.setAttribute('role', 'status');
                announcement.setAttribute('aria-live', 'polite');
                announcement.textContent = message;

                document.body.appendChild(announcement);

                setTimeout(() => announcement.remove(), 1000);
            }
        }

        const chatAssistant = new ChatAssistant();
    </script>
</body>
</html>
