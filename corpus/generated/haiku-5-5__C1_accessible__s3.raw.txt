<script>
(function () {
  'use strict';

  const log = document.getElementById('log');
  const statusEl = document.getElementById('status');
  const form = document.getElementById('composer');
  const promptEl = document.getElementById('prompt');
  const sendBtn = document.getElementById('send');
  const stopBtn = document.getElementById('stop');
  const regenBtn = document.getElementById('regenerate');

  const sdkReady = !!(window.ChatSDK && typeof window.ChatSDK.streamChat === 'function');

  // Conversation state: {role, content, el}
  const history = [];
  // Active request: {entry, view, handle, finished}
  let current = null;
  let busy = false;

  function setStatus(text) {
    statusEl.textContent = text;
  }

  function removeEmptyPlaceholder() {
    const empty = document.getElementById('empty');
    if (empty) empty.remove();
  }

  function isNearBottom() {
    return log.scrollHeight - log.scrollTop - log.clientHeight < 100;
  }

  function scrollToEnd() {
    log.scrollTop = log.scrollHeight;
  }

  function addNotice(text) {
    const n = document.createElement('div');
    n.className = 'notice';
    n.setAttribute('role', 'alert');
    n.textContent = text;
    log.appendChild(n);
    scrollToEnd();
  }

  // Builds a message bubble using textContent only (no HTML injection).
  function renderMessage(role, text, streaming) {
    removeEmptyPlaceholder();

    const wrap = document.createElement('div');
    wrap.className = 'msg ' + role;

    const who = document.createElement('p');
    who.className = 'who';
    who.textContent = role === 'user' ? 'You' : 'Assistant';

    const body = document.createElement('div');
    body.className = 'body';

    let pending = null;
    if (streaming) {
      pending = document.createElement('span');
      pending.className = 'pending';
      pending.textContent = 'Generating…';
      body.appendChild(pending);
    }

    const textEl = document.createElement('span');
    textEl.textContent = text;
    body.appendChild(textEl);

    let caret = null;
    if (streaming) {
      caret = document.createElement('span');
      caret.className = 'caret';
      caret.setAttribute('aria-hidden', 'true');
      body.appendChild(caret);
    }

    wrap.appendChild(who);
    wrap.appendChild(body);
    log.appendChild(wrap);
    scrollToEnd();

    return { wrap, text: textEl, caret, pending };
  }

  function lastUserIndex() {
    for (let i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') return i;
    }
    return -1;
  }

  function setBusy(value) {
    busy = value;
    sendBtn.disabled = value || !sdkReady;
    stopBtn.disabled = !value;
    regenBtn.disabled = value || !sdkReady || lastUserIndex() < 0;
    log.setAttribute('aria-busy', String(value));

    // Never leave keyboard focus on a button that just became disabled.
    const active = document.activeElement;
    if (active && active.disabled) {
      promptEl.focus();
    }
  }

  function removeEntry(entry) {
    const i = history.indexOf(entry);
    if (i >= 0) history.splice(i, 1);
    if (entry.el) entry.el.remove();
  }

  function errorText(err) {
    const detail = err && err.message ? String(err.message) : '';
    return 'The answer could not be completed.' +
      (detail ? ' Details: ' + detail : '') +
      ' Use Regenerate to try again.';
  }

  function finish(req, outcome, err) {
    if (req.finished) return;
    req.finished = true;
    if (current === req) current = null;

    const entry = req.entry;
    const view = req.view;
    if (view.caret) view.caret.remove();
    if (view.pending) { view.pending.remove(); view.pending = null; }

    const empty = entry.content === '';

    if (outcome === 'done') {
      setStatus('Answer finished.');
    } else if (outcome === 'stopped') {
      setStatus(empty
        ? 'Generation stopped. No text had been received.'
        : 'Generation stopped. Partial answer kept.');
    } else {
      setStatus('Error: the answer failed. Use Regenerate to try again.');
    }

    // An empty answer that was stopped or failed has nothing worth keeping.
    if (empty && outcome !== 'done') {
      removeEntry(entry);
    }

    if (outcome === 'error') {
      addNotice(errorText(err));
    }

    setBusy(false);
  }

  function startRequest() {
    if (!sdkReady || busy) return;

    const apiMessages = history.map(m => ({ role: m.role, content: m.content }));
    const entry = { role: 'assistant', content: '', el: null };
    const view = renderMessage('assistant', '', true);
    entry.el = view.wrap;
    history.push(entry);

    const req = { entry, view, handle: null, finished: false };
    current = req;
    setBusy(true);
    setStatus('Generating answer…');

    try {
      req.handle = window.ChatSDK.streamChat(apiMessages, {
        onToken(text) {
          // Ignore anything that arrives after abort or completion.
          if (req !== current || req.finished) return;
          const follow = isNearBottom();
          if (view.pending) { view.pending.remove(); view.pending = null; }
          entry.content += String(text);
          view.text.textContent = entry.content;
          if (follow) scrollToEnd();
        },
        onDone() {
          if (req !== current || req.finished) return;
          finish(req, 'done');
        },
        onError(err) {
          if (req !== current || req.finished) return;
          finish(req, 'error', err);
        }
      });
    } catch (err) {
      finish(req, 'error', err);
    }
  }

  function stopGeneration() {
    const req = current;
    if (!req || req.finished) return;
    // Mark stopped first so late tokens are ignored, then ask the server to stop.
    current = null;
    try {
      if (req.handle && typeof req.handle.abort === 'function') {
        req.handle.abort();
      }
    } catch (e) {
      // Abort failures are not fatal; the local stream is already closed.
    }
    finish(req, 'stopped');
  }

  function regenerate() {
    if (busy || !sdkReady) return;
    const idx = lastUserIndex();
    if (idx < 0) return;

    // Remove everything after the last user message (old answers and notices).
    const userEntry = history[idx];
    if (userEntry.el) {
      let node = userEntry.el.nextSibling;
      while (node) {
        const next = node.nextSibling;
        log.removeChild(node);
        node = next;
      }
    }
    history.length = idx + 1;
    startRequest();
  }

  // Event wiring
  form.addEventListener('submit', function (e) {
    e.preventDefault();
    if (busy || !sdkReady) return;

    const text = promptEl.value.trim();
    if (!text) {
      setStatus('Type a message before sending.');
      promptEl.focus();
      return;
    }

    const view = renderMessage('user', text, false);
    history.push({ role: 'user', content: text, el: view.wrap });
    promptEl.value = '';
    startRequest();
  });

  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      form.requestSubmit();
    }
  });

  stopBtn.addEventListener('click', stopGeneration);
  regenBtn.addEventListener('click', regenerate);

  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && busy) {
      e.preventDefault();
      stopGeneration();
    }
  });

  // Initial state
  setBusy(false);
  if (sdkReady) {
    setStatus('Ready.');
  } else {
    setStatus('Error: the chat service is not available.');
    addNotice('The chat service (window.ChatSDK.streamChat) is not available. Reload the page or check the host integration.');
  }
})();
</script>
</body>
</html>
