<script>
(function () {
  'use strict';
  var $ = function (id) { return document.getElementById(id); };
  var promptEl = $('prompt'), sendBtn = $('send'), stopBtn = $('stop'), regenBtn = $('regenerate');
  var transcript = $('transcript'), statusText = $('status'), statusBox = $('statusbox');
  var emptyEl = $('empty'), form = $('form');

  // history entries: {role, content, status: 'done'|'streaming'|'stopped'|'error', errorText, li, textNode, metaEl}
  var history = [];
  var active = null; // the one request whose chunks are allowed to change the transcript

  function setStatus(text, kind) {
    statusText.textContent = text;
    statusBox.className = 'statusbox' + (kind ? ' ' + kind : '');
  }

  function setAria(btn, disabled) {
    btn.setAttribute('aria-disabled', disabled ? 'true' : 'false');
  }
  function isDisabled(btn) { return btn.getAttribute('aria-disabled') === 'true'; }

  function hasUser() {
    for (var i = 0; i < history.length; i++) if (history[i].role === 'user') return true;
    return false;
  }

  // aria-disabled (not the disabled attribute) keeps focus on the button when its state changes.
  function updateControls() {
    setAria(sendBtn, !!active);
    setAria(stopBtn, !active);
    setAria(regenBtn, !!active || !hasUser());
  }

  function renderMeta(entry) {
    if (!entry.metaEl) return;
    var t = '', err = false;
    if (entry.status === 'streaming') t = 'Generating\u2026';
    else if (entry.status === 'stopped') t = 'Stopped \u2013 partial answer kept.';
    else if (entry.status === 'error') { t = 'Error: ' + entry.errorText; err = true; }
    entry.metaEl.textContent = t;
    entry.metaEl.hidden = !t;
    entry.metaEl.className = 'meta' + (err ? ' error' : '');
  }

  function nearBottom() {
    return window.innerHeight + window.scrollY >= document.documentElement.scrollHeight - 100;
  }
  function scrollDown() {
    window.scrollTo(0, document.documentElement.scrollHeight);
  }

  function addEntry(role, content, status) {
    var li = document.createElement('li');
    li.className = 'msg ' + (role === 'user' ? 'user' : 'assistant');
    var who = document.createElement('p');
    who.className = 'who';
    who.textContent = role === 'user' ? 'You' : 'Assistant';
    var text = document.createElement('p');
    text.className = 'text';
    var node = document.createTextNode(content);
    text.appendChild(node);
    li.appendChild(who);
    li.appendChild(text);
    var entry = { role: role, content: content, status: status, errorText: '', li: li, textNode: node, metaEl: null };
    if (role === 'assistant') {
      var meta = document.createElement('p');
      meta.hidden = true;
      li.appendChild(meta);
      entry.metaEl = meta;
    }
    var stick = nearBottom();
    transcript.appendChild(li);
    emptyEl.hidden = true;
    history.push(entry);
    renderMeta(entry);
    if (stick) scrollDown();
    return entry;
  }

  function removeTrailingAssistants() {
    while (history.length && history[history.length - 1].role === 'assistant') {
      var e = history.pop();
      if (e.li.parentNode) e.li.parentNode.removeChild(e.li);
    }
  }

  function errorMessage(err) {
    var m = '';
    if (err && typeof err === 'object' && err.message) m = String(err.message);
    else if (typeof err === 'string') m = err;
    m = m.trim().slice(0, 200);
    return m || 'The connection to the assistant failed.';
  }

  function startAnswer() {
    var apiMessages = history
      .filter(function (m) { return m.content.trim() !== ''; })
      .map(function (m) { return { role: m.role, content: m.content }; });

    var entry = addEntry('assistant', '', 'streaming');
    var req = { entry: entry, ended: false, handle: null };
    active = req;
    updateControls();
    setStatus('Generating answer\u2026', 'generating');

    function live() { return active === req && !req.ended; }

    function fail(err) {
      if (!live()) return;
      req.ended = true;
      active = null;
      var msg = errorMessage(err);
      entry.status = 'error';
      entry.errorText = msg;
      renderMeta(entry);
      setStatus('Error: ' + msg + (entry.content ? ' The partial answer was kept.' : '') +
        ' Press Regenerate to try again.', 'error');
      updateControls();
    }

    var callbacks = {
      onToken: function (t) {
        if (!live()) return; // late chunk from a stopped/replaced answer: ignore
        t = (t === undefined || t === null) ? '' : String(t);
        if (!t) return;
        var stick = nearBottom();
        entry.content += t;
        entry.textNode.appendData(t);
        if (stick) scrollDown();
      },
      onDone: function () {
        if (!live()) return;
        req.ended = true;
        active = null;
        entry.status = 'done';
        renderMeta(entry);
        if (!entry.content) {
          entry.metaEl.hidden = false;
          entry.metaEl.textContent = 'The assistant returned an empty answer.';
          setStatus('Answer finished, but it was empty.', 'done');
        } else {
          setStatus('Answer finished.', 'done');
        }
        updateControls();
      },
      onError: fail
    };

    try {
      if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
        throw new Error('The chat service is not available.');
      }
      req.handle = window.ChatSDK.streamChat(apiMessages, callbacks);
    } catch (e) {
      fail(e);
    }
  }

  function stopAnswer() {
    var req = active;
    if (!req) return;
    // Mark as ended first so any chunk still in flight is ignored.
    req.ended = true;
    active = null;
    try { if (req.handle && typeof req.handle.abort === 'function') req.handle.abort(); } catch (e) { /* ignore */ }
    req.entry.status = 'stopped';
    renderMeta(req.entry);
    setStatus(req.entry.content
      ? 'Generation stopped. The partial answer was kept.'
      : 'Generation stopped before any text arrived.', 'stopped');
    updateControls();
  }

  function trySend() {
    if (active) {
      setStatus('An answer is still generating. Press Stop first, or wait for it to finish.', 'generating');
      return;
    }
    var text = promptEl.value.trim();
    if (!text) {
      setStatus('Type a message before sending.', '');
      return;
    }
    promptEl.value = '';
    addEntry('user', text, 'done');
    startAnswer();
  }

  function tryRegenerate() {
    if (active) {
      setStatus('Stop the current answer before regenerating.', 'generating');
      return;
    }
    if (!hasUser()) return;
    removeTrailingAssistants();
    startAnswer();
  }

  form.addEventListener('submit', function (e) { e.preventDefault(); trySend(); });
  stopBtn.addEventListener('click', function () { if (!isDisabled(stopBtn)) stopAnswer(); });
  regenBtn.addEventListener('click', function () { if (!isDisabled(regenBtn) || active) tryRegenerate(); });

  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      trySend();
    } else if (e.key === 'Escape' && active) {
      e.preventDefault();
      stopAnswer();
    }
  });

  updateControls();
})();
</script>
</body>
</html>
