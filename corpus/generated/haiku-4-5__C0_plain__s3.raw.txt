<body>
    <div class="container">
        <div class="header">Chat Assistant</div>
        <div class="messages-container" id="messagesContainer">
            <div class="empty-state">Start a conversation</div>
        </div>
        <div class="input-area">
            <textarea id="prompt" placeholder="Type your message..." rows="1"></textarea>
            <div class="button-group">
                <button id="send">Send</button>
                <button id="stop">Stop</button>
                <button id="regenerate">Regenerate</button>
            </div>
        </div>
    </div>

    <script>
        const messagesContainer = document.getElementById('messagesContainer');
        const promptInput = document.getElementById('prompt');
        const sendBtn = document.getElementById('send');
        const stopBtn = document.getElementById('stop');
        const regenerateBtn = document.getElementById('regenerate');

        let messages = [];
        let isStreaming = false;
        let currentStreamHandle = null;

        function clearEmptyState() {
            const emptyState = messagesContainer.querySelector('.empty-state');
            if (emptyState) {
                emptyState.remove();
            }
        }

        function addMessage(role, content) {
            clearEmptyState();
            const messageEl = document.createElement('div');
            messageEl.className = `message ${role}`;
            const contentEl = document.createElement('div');
            contentEl.className = 'message-content';
            contentEl.textContent = content;
            messageEl.appendChild(contentEl);
            messagesContainer.appendChild(messageEl);
            messagesContainer.scrollTop = messagesContainer.scrollHeight;
            return contentEl;
        }

        function addStatus(text, type) {
            clearEmptyState();
            const statusEl = document.createElement('div');
            statusEl.className = `status ${type}`;
            statusEl.textContent = text;
            statusEl.id = 'currentStatus';
            messagesContainer.appendChild(statusEl);
            messagesContainer.scrollTop = messagesContainer.scrollHeight;
            return statusEl;
        }

        function updateStatus(text, type) {
            let statusEl = document.getElementById('currentStatus');
            if (!statusEl) {
                statusEl = addStatus(text, type);
            } else {
                statusEl.textContent = text;
                statusEl.className = `status ${type}`;
            }
            return statusEl;
        }

        function removeStatus() {
            const statusEl = document.getElementById('currentStatus');
            if (statusEl) {
                statusEl.remove();
            }
        }

        function updatePromptHeight() {
            promptInput.style.height = 'auto';
            promptInput.style.height = Math.min(promptInput.scrollHeight, 100) + 'px';
        }

        function updateButtonStates() {
            const hasMessages = messages.length > 0;
            const lastIsUser = messages.length > 0 && messages[messages.length - 1].role === 'user';
            
            sendBtn.disabled = isStreaming;
            stopBtn.style.display = isStreaming ? 'block' : 'none';
            regenerateBtn.style.display = !isStreaming && lastIsUser ? 'block' : 'none';
        }

        async function sendMessage() {
            const content = promptInput.value.trim();
            if (!content || isStreaming) return;

            messages.push({ role: 'user', content });
            addMessage('user', content);
            promptInput.value = '';
            updatePromptHeight();
            updateButtonStates();

            streamAssistantResponse();
        }

        function streamAssistantResponse() {
            if (isStreaming) return;

            isStreaming = true;
            updateButtonStates();
            updateStatus('Generating...', 'generating');

            let assistantContent = '';
            let contentEl = null;

            const onToken = (text) => {
                if (!contentEl) {
                    removeStatus();
                    contentEl = addMessage('assistant', '');
                }
                assistantContent += text;
                contentEl.textContent = assistantContent;
                messagesContainer.scrollTop = messagesContainer.scrollHeight;
            };

            const onDone = () => {
                isStreaming = false;
                currentStreamHandle = null;
                messages.push({ role: 'assistant', content: assistantContent });
                removeStatus();
                updateStatus('Done', 'done');
                setTimeout(() => removeStatus(), 2000);
                updateButtonStates();
            };

            const onError = (error) => {
                isStreaming = false;
                currentStreamHandle = null;
                removeStatus();
                const errorMsg = error instanceof Error ? error.message : String(error);
                updateStatus(`Error: ${errorMsg}`, 'error');
                updateButtonStates();
            };

            currentStreamHandle = window.ChatSDK.streamChat(messages, {
                onToken,
                onDone,
                onError
            });
        }

        function stopStreaming() {
            if (currentStreamHandle) {
                currentStreamHandle.abort();
                currentStreamHandle = null;
                isStreaming = false;
                updateStatus('Stopped', 'stopped');
                setTimeout(() => removeStatus(), 2000);
                updateButtonStates();
            }
        }

        function regenerateLastResponse() {
            if (messages.length === 0) return;
            
            // Remove last assistant message if it exists
            if (messages[messages.length - 1].role === 'assistant') {
                messages.pop();
            }

            // Remove last message element if it's an assistant message
            const lastMessage = messagesContainer.querySelector('.message:last-of-type');
            if (lastMessage && lastMessage.classList.contains('assistant')) {
                lastMessage.remove();
            }

            streamAssistantResponse();
        }

        promptInput.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' && !e.shiftKey && !isStreaming) {
                e.preventDefault();
                sendMessage();
            }
        });

        promptInput.addEventListener('input', updatePromptHeight);

        sendBtn.addEventListener('click', sendMessage);
        stopBtn.addEventListener('click', stopStreaming);
        regenerateBtn.addEventListener('click', regenerateLastResponse);

        updateButtonStates();
    </script>
</body>
</html>
