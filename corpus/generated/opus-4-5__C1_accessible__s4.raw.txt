    <script>
        (function() {
            'use strict';

            // State
            let conversationHistory = [];
            let currentHandle = null;
            let isGenerating = false;
            let currentAssistantMessageEl = null;
            let wasStopped = false;

            // DOM Elements
            const messagesContainer = document.getElementById('messages');
            const statusEl = document.getElementById('status');
            const promptInput = document.getElementById('prompt');
            const sendBtn = document.getElementById('send');
            const stopBtn = document.getElementById('stop');
            const regenerateBtn = document.getElementById('regenerate');
            const chatForm = document.getElementById('chat-form');
            const emptyState = document.getElementById('empty-state');

            // Utility functions
            function escapeHtml(text) {
                const div = document.createElement('div');
                div.textContent = text;
                return div.innerHTML;
            }

            function scrollToBottom() {
                messagesContainer.scrollTop = messagesContainer.scrollHeight;
            }

            function setStatus(message, type) {
                statusEl.textContent = message;
                statusEl.className = 'status-indicator';
                if (type === 'generating') {
                    statusEl.classList.add('status-generating');
                } else if (type === 'error') {
                    statusEl.classList.add('status-error');
                } else if (type === 'stopped') {
                    statusEl.classList.add('status-stopped');
                }
            }

            function clearStatus() {
                statusEl.textContent = '';
                statusEl.className = 'status-indicator';
            }

            function updateButtonStates() {
                sendBtn.disabled = isGenerating || promptInput.value.trim() === '';
                stopBtn.disabled = !isGenerating;
                regenerateBtn.disabled = isGenerating || !hasLastUserMessage();
            }

            function hasLastUserMessage() {
                for (let i = conversationHistory.length - 1; i >= 0; i--) {
                    if (conversationHistory[i].role === 'user') {
                        return true;
                    }
                }
                return false;
            }

            function hideEmptyState() {
                if (emptyState) {
                    emptyState.style.display = 'none';
                }
            }

            function createMessageElement(role, content) {
                const messageDiv = document.createElement('article');
                messageDiv.className = `message message-${role}`;
                messageDiv.setAttribute('aria-label', `${role === 'user' ? 'You' : 'Assistant'} said`);

                const roleLabel = document.createElement('div');
                roleLabel.className = 'message-role';
                roleLabel.textContent = role === 'user' ? 'You' : 'Assistant';

                const contentDiv = document.createElement('div');
                contentDiv.className = 'message-content';
                contentDiv.textContent = content;

                messageDiv.appendChild(roleLabel);
                messageDiv.appendChild(contentDiv);

                return messageDiv;
            }

            function addUserMessage(content) {
                hideEmptyState();
                const messageEl = createMessageElement('user', content);
                messagesContainer.appendChild(messageEl);
                conversationHistory.push({ role: 'user', content: content });
                scrollToBottom();
            }

            function startAssistantMessage() {
                hideEmptyState();
                currentAssistantMessageEl = createMessageElement('assistant', '');
                messagesContainer.appendChild(currentAssistantMessageEl);
                scrollToBottom();
            }

            function appendToAssistantMessage(text) {
                if (currentAssistantMessageEl) {
                    const contentDiv = currentAssistantMessageEl.querySelector('.message-content');
                    if (contentDiv) {
                        contentDiv.textContent += text;
                        scrollToBottom();
                    }
                }
            }

            function finalizeAssistantMessage() {
                if (currentAssistantMessageEl) {
                    const contentDiv = currentAssistantMessageEl.querySelector('.message-content');
                    if (contentDiv) {
                        const content = contentDiv.textContent;
                        conversationHistory.push({ role: 'assistant', content: content });
                    }
                    currentAssistantMessageEl = null;
                }
            }

            function removeLastAssistantMessage() {
                // Remove from DOM
                const messages = messagesContainer.querySelectorAll('.message-assistant');
                if (messages.length > 0) {
                    const lastMessage = messages[messages.length - 1];
                    lastMessage.remove();
                }
                // Remove from history
                if (conversationHistory.length > 0 && 
                    conversationHistory[conversationHistory.length - 1].role === 'assistant') {
                    conversationHistory.pop();
                }
            }

            function getMessagesForAPI() {
                return conversationHistory.map(msg => ({
                    role: msg.role,
                    content: msg.content
                }));
            }

            function startStreaming() {
                isGenerating = true;
                wasStopped = false;
                updateButtonStates();
                setStatus('Generating response...', 'generating');
                startAssistantMessage();

                const messages = getMessagesForAPI();

                currentHandle = window.ChatSDK.streamChat(messages, {
                    onToken: function(text) {
                        appendToAssistantMessage(text);
                    },
                    onDone: function() {
                        isGenerating = false;
                        currentHandle = null;
                        finalizeAssistantMessage();
                        if (!wasStopped) {
                            setStatus('Response complete', 'complete');
                            setTimeout(clearStatus, 2000);
                        }
                        updateButtonStates();
                        promptInput.focus();
                    },
                    onError: function(error) {
                        isGenerating = false;
                        currentHandle = null;
                        finalizeAssistantMessage();
                        const errorMessage = error && error.message ? error.message : 'An error occurred';
                        setStatus('Error: ' + errorMessage, 'error');
                        updateButtonStates();
                        promptInput.focus();
                    }
                });
            }

            // Event handlers
            function handleSend(e) {
                if (e) e.preventDefault();
                
                const content = promptInput.value.trim();
                if (content === '' || isGenerating) {
                    return;
                }

                addUserMessage(content);
                promptInput.value = '';
                updateButtonStates();
                startStreaming();
            }

            function handleStop() {
                if (currentHandle && isGenerating) {
                    wasStopped = true;
                    currentHandle.abort();
                    isGenerating = false;
                    currentHandle = null;
                    finalizeAssistantMessage();
                    setStatus('Generation stopped', 'stopped');
                    updateButtonStates();
                    promptInput.focus();
                }
            }

            function handleRegenerate() {
                if (isGenerating) {
                    return;
                }

                // Find and keep the last user message, remove everything after it
                let lastUserIndex = -1;
                for (let i = conversationHistory.length - 1; i >= 0; i--) {
                    if (conversationHistory[i].role === 'user') {
                        lastUserIndex = i;
                        break;
                    }
                }

                if (lastUserIndex === -1) {
                    return;
                }

                // Remove any assistant messages after the last user message
                while (conversationHistory.length > lastUserIndex + 1) {
                    removeLastAssistantMessage();
                }

                // Also remove from DOM if there's a trailing assistant message
                removeLastAssistantMessage();

                // Start new stream
                startStreaming();
            }

            function handleKeydown(e) {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    handleSend();
                }
            }

            function handleInput() {
                updateButtonStates();
                // Auto-resize textarea
                promptInput.style.height = 'auto';
                promptInput.style.height = Math.min(promptInput.scrollHeight, 200) + 'px';
            }

            // Initialize
            function init() {
                chatForm.addEventListener('submit', handleSend);
                stopBtn.addEventListener('click', handleStop);
                regenerateBtn.addEventListener('click', handleRegenerate);
                promptInput.addEventListener('keydown', handleKeydown);
                promptInput.addEventListener('input', handleInput);

                updateButtonStates();
                promptInput.focus();
            }

            // Run initialization when DOM is ready
            if (document.readyState === 'loading') {
                document.addEventListener('DOMContentLoaded', init);
            } else {
                init();
            }
        })();
    </script>
</body>
</html>
