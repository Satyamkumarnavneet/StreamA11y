<script>
(function () {
  'use strict';

  var $ = function (id) { return document.getElementById(id); };
  var historyEl = $('history'), listEl = $('list'), emptyEl = $('empty');
  var promptEl = $('prompt'), sendBtn = $('send'), stopBtn = $('stop'), regenBtn = $('regenerate');
  var statusEl = $('status'), statusIcon = $('statusIcon'), statusBar = $('statusbar');
  var alertEl = $('alert'), form = $('composer');

  var ICONS = { idle: 'i', generating: '\u25CF', stopped: '\u25A0', done: '\u2713', error: '!' };

  var messages = [];   // {role, content, state, li, textEl, textNode, stateEl}
  var current = null;  // the active run: {cancelled, msg, handle}

  /* ---------- UI helpers ---------- */
  function setStatus(kind, text) {
    statusBar.setAttribute('data-state', kind);
    statusIcon.textContent = ICONS[kind] || '';
    statusEl.textContent = text;
  }

  function clearError() { alertEl.textContent = ''; }

  function showError(text) {
    clearError();
    var p = document.createElement('p');
    p.className = 'error-msg';
    p.textContent = text;
    alertEl.appendChild(p);
  }

  function nearBottom() {
    return historyEl.scrollHeight - historyEl.scrollTop - historyEl.clientHeight < 80;
  }
  function scrollDown() { historyEl.scrollTop = historyEl.scrollHeight; }

  function updateControls() {
    sendBtn.disabled = !!current;
    stopBtn.disabled = !current;
    regenBtn.disabled = !messages.some(function (m) { return m.role === 'user'; });
  }

  function addMessage(role, content, state) {
    var li = document.createElement('li');
    li.className = 'msg ' + role;
    var who = document.createElement('span');
    who.className = 'who';
    who.textContent = role === 'user' ? 'You' : 'Assistant';
    var textEl = document.createElement('div');
    textEl.className = 'text';
    var textNode = document.createTextNode(content);
    textEl.appendChild(textNode);
    var stateEl = document.createElement('div');
    stateEl.className = 'state';
    li.appendChild(who);
    li.appendChild(textEl);
    li.appendChild(stateEl);

    var stick = nearBottom();
    listEl.appendChild(li);
    emptyEl.hidden = true;

    var m = { role: role, content: content, state: state || 'done',
              li: li, textEl: textEl, textNode: textNode, stateEl: stateEl };
    messages.push(m);
    renderState(m);
    if (stick || role === 'user') scrollDown();
    return m;
  }

  function renderState(m) {
    var t = '';
    if (m.role === 'assistant') {
      if (m.state === 'streaming') t = 'Generating\u2026';
      else if (m.state === 'stopped') t = m.content ? 'Stopped (partial answer kept)' : 'Stopped before any text arrived';
      else if (m.state === 'error') t = m.content ? 'Error (answer incomplete)' : 'Error (no answer received)';
      else if (m.state === 'done' && !m.content) t = 'No text was returned';
    }
    m.stateEl.textContent = t;
    m.li.classList.toggle('is-error', m.state === 'error');
    if (m.state === 'streaming') m.li.setAttribute('aria-busy', 'true');
    else m.li.removeAttribute('aria-busy');
  }

  function setState(m, state) { m.state = state; renderState(m); }

  function removeAfter(index) {
    while (messages.length > index + 1) {
      var m = messages.pop();
      if (m.li.parentNode) m.li.parentNode.removeChild(m.li);
    }
  }

  function payload() {
    return messages
      .filter(function (m) { return m.role === 'user' || m.content.length > 0; })
      .map(function (m) { return { role: m.role, content: m.content }; });
  }

  /* ---------- Streaming ---------- */
  function cancelRun(run) {
    run.cancelled = true;
    if (current === run) current = null;
    try { if (run.handle && typeof run.handle.abort === 'function') run.handle.abort(); } catch (e) { /* ignore */ }
  }

  function finishRun(run, state, statusKind, statusText) {
    var hadFocus = document.activeElement === stopBtn;
    run.cancelled = true; // ignore anything after the terminal event
    if (current === run) current = null;
    setState(run.msg, state);
    setStatus(statusKind, statusText);
    updateControls();
    if (hadFocus) promptEl.focus();
  }

  function startRun(startText) {
    clearError();
    var history = payload();
    var msg = addMessage('assistant', '', 'streaming');
    var run = { cancelled: false, msg: msg, handle: null };
    current = run;
    setStatus('generating', startText);
    updateControls();

    function onToken(text) {
      if (run.cancelled || current !== run) return; // late chunk after stop/regenerate
      var s = String(text == null ? '' : text);
      if (!s) return;
      var stick = nearBottom();
      msg.content += s;
      msg.textNode.appendData(s);
      if (stick) scrollDown();
    }
    function onDone() {
      if (run.cancelled || current !== run) return;
      finishRun(run, 'done', 'done', msg.content ? 'Answer finished.' : 'Answer finished, but no text was returned.');
    }
    function onError(err) {
      if (run.cancelled || current !== run) return;
      var detail = (err && err.message) ? err.message : (err ? String(err) : 'Unknown error');
      finishRun(run, 'error', 'error', '');
      showError('Error: the answer could not be completed (' + detail + '). ' +
                (msg.content ? 'The partial answer is kept above. ' : '') +
                'Press Regenerate to try again.');
    }

    try {
      if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
        throw new Error('The chat service is not available');
      }
      run.handle = window.ChatSDK.streamChat(history, { onToken: onToken, onDone: onDone, onError: onError });
    } catch (e) {
      onError(e);
    }
  }

  /* ---------- Actions ---------- */
  function send() {
    if (current) {
      setStatus('generating', 'Please wait for the answer to finish, or press Stop.');
      return;
    }
    var text = promptEl.value.trim();
    if (!text) {
      setStatus('idle', 'Please type a message before sending.');
      promptEl.focus();
      return;
    }
    addMessage('user', text, 'done');
    promptEl.value = '';
    startRun('Assistant is generating an answer\u2026');
    promptEl.focus();
  }

  function stop() {
    if (!current) return;
    var run = current;
    cancelRun(run);
    setState(run.msg, 'stopped');
    setStatus('stopped', run.msg.content ? 'Stopped. The partial answer was kept.' : 'Stopped before any text arrived.');
    updateControls();
    promptEl.focus();
  }

  function regenerate() {
    var idx = -1, i;
    for (i = messages.length - 1; i >= 0; i--) {
      if (messages[i].role === 'user') { idx = i; break; }
    }
    if (idx < 0) return;
    if (current) cancelRun(current);
    removeAfter(idx);
    startRun('Assistant is generating a new answer\u2026');
  }

  /* ---------- Wiring ---------- */
  form.addEventListener('submit', function (e) { e.preventDefault(); send(); });
  stopBtn.addEventListener('click', stop);
  regenBtn.addEventListener('click', regenerate);
  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.ctrlKey && !e.altKey && !e.metaKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });

  updateControls();
})();
</script>
</body>
</html>
