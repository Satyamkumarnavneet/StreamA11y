<body>
    <div id="chat-container">
        <div id="messages" role="log" aria-label="Chat conversation"></div>
        <div id="input-area">
            <textarea
                id="prompt"
                placeholder="Type your message..."
                aria-label="Message input"
            ></textarea>
            <div class="button-group">
                <button id="send" aria-label="Send message">Send</button>
                <button id="stop" disabled aria-label="Stop generating response">Stop</button>
                <button id="regenerate" disabled aria-label="Regenerate last response">Regenerate</button>
            </div>
        </div>
    </div>

    <div id="status" role="status" aria-live="polite" aria-atomic="true"></div>

    <script>
        // State management
        const state = {
            messages: [],
            isGenerating: false,
            currentHandle: null,
            lastUserMessage: null,
            currentAssistantMessage: null
        };

        // DOM elements
        const messagesContainer = document.getElementById('messages');
        const promptInput = document.getElementById('prompt');
        const sendButton = document.getElementById('send');
        const stopButton = document.getElementById('stop');
        const regenerateButton = document.getElementById('regenerate');
        const statusElement = document.getElementById('status');

        // Utility functions
        function setStatus(message) {
            statusElement.textContent = message;
        }

        function updateButtonStates() {
            const hasMessages = state.messages.length > 0;
            const lastMessageIsUser = hasMessages && state.messages[state.messages.length - 1].role === 'user';

            sendButton.disabled = state.isGenerating;
            stopButton.disabled = !state.isGenerating;
            regenerateButton.disabled = state.isGenerating || !lastMessageIsUser;
            promptInput.disabled = state.isGenerating;
        }

        function scrollToBottom() {
            messagesContainer.scrollTop = messagesContainer.scrollHeight;
        }

        function addMessageToDOM(role, content, isPartial = false) {
            const messageEl = document.createElement('div');
            messageEl.className = `message ${role}`;

            const roleEl = document.createElement('div');
            roleEl.className = 'message-role';
            roleEl.textContent = role === 'user' ? 'You' : 'Assistant';

            const contentEl = document.createElement('div');
            contentEl.className = 'message-content';
            contentEl.textContent = content;

            if (isPartial && role === 'assistant') {
                contentEl.id = 'current-assistant-message';
                contentEl.setAttribute('aria-live', 'off');
            }

            messageEl.appendChild(roleEl);
            messageEl.appendChild(contentEl);
            messagesContainer.appendChild(messageEl);

            scrollToBottom();

            return contentEl;
        }

        function addErrorMessageToDOM(error) {
            const messageEl = document.createElement('div');
            messageEl.className = 'message error';

            const roleEl = document.createElement('div');
            roleEl.className = 'message-role';
            roleEl.textContent = 'Error';

            const contentEl = document.createElement('div');
            contentEl.className = 'message-content';
            contentEl.textContent = `An error occurred: ${error}`;

            messageEl.appendChild(roleEl);
            messageEl.appendChild(contentEl);
            messagesContainer.appendChild(messageEl);

            scrollToBottom();
        }

        function createGeneratingIndicator() {
            const indicatorEl = document.createElement('div');
            indicatorEl.className = 'generating-indicator';
            indicatorEl.id = 'generating-indicator';
            indicatorEl.setAttribute('aria-live', 'off');
            indicatorEl.textContent = 'Generating response...';
            messagesContainer.appendChild(indicatorEl);
            scrollToBottom();
            return indicatorEl;
        }

        function removeGeneratingIndicator() {
            const indicator = document.getElementById('generating-indicator');
            if (indicator) {
                indicator.remove();
            }
        }

        function addUserMessage(text) {
            state.messages.push({ role: 'user', content: text });
            state.lastUserMessage = text;
            addMessageToDOM('user', text);
        }

        async function sendMessage() {
            const text = promptInput.value.trim();
            if (!text) return;

            promptInput.value = '';
            addUserMessage(text);
            await generateResponse();
        }

        async function generateResponse() {
            if (state.isGenerating) return;

            state.isGenerating = true;
            updateButtonStates();
            createGeneratingIndicator();
            setStatus('Assistant is generating a response');

            let fullResponse = '';
            let messageContentEl = null;

            state.currentHandle = window.ChatSDK.streamChat(state.messages, {
                onToken: (text) => {
                    if (state.currentHandle.aborted) return;

                    fullResponse += text;

                    if (!messageContentEl) {
                        removeGeneratingIndicator();
                        messageContentEl = addMessageToDOM('assistant', '', true);
                    }

                    messageContentEl.textContent = fullResponse;
                    scrollToBottom();
                },
                onDone: () => {
                    removeGeneratingIndicator();
                    state.isGenerating = false;
                    state.currentHandle = null;
                    state.currentAssistantMessage = fullResponse;

                    if (messageContentEl) {
                        messageContentEl.id = '';
                        messageContentEl.removeAttribute('aria-live');
                    }

                    state.messages.push({ role: 'assistant', content: fullResponse });
                    updateButtonStates();
                    setStatus('Response complete');
                    promptInput.focus();
                },
                onError: (error) => {
                    removeGeneratingIndicator();
                    state.isGenerating = false;
                    state.currentHandle = null;

                    if (messageContentEl) {
                        messageContentEl.id = '';
                        messageContentEl.removeAttribute('aria-live');
                    }

                    addErrorMessageToDOM(error.message || 'Unknown error');
                    updateButtonStates();
                    setStatus(`Error: ${error.message || 'Unknown error'}`);
                    promptInput.focus();
                }
            });
        }

        function stopGeneration() {
            if (!state.isGenerating || !state.currentHandle) return;

            state.currentHandle.abort();
            removeGeneratingIndicator();
            state.isGenerating = false;
            state.currentHandle = null;

            const messageContentEl = document.getElementById('current-assistant-message');
            if (messageContentEl) {
                messageContentEl.id = '';
                messageContentEl.removeAttribute('aria-live');
                // Keep the partial response in the conversation
                if (messageContentEl.textContent) {
                    state.messages.push({ role: 'assistant', content: messageContentEl.textContent });
                    state.currentAssistantMessage = messageContentEl.textContent;
                }
            }

            updateButtonStates();
            setStatus('Response generation stopped');
            promptInput.focus();
        }

        async function regenerateLastResponse() {
            if (!state.lastUserMessage) return;

            // Remove the last assistant message if it exists
            if (state.messages.length > 0 && state.messages[state.messages.length - 1].role === 'assistant') {
                state.messages.pop();
            }

            // Remove the last assistant message from DOM
            const messages = messagesContainer.querySelectorAll('.message');
            if (messages.length > 0) {
                const lastMessage = messages[messages.length - 1];
                if (lastMessage.querySelector('.message-role').textContent === 'Assistant') {
                    lastMessage.remove();
                }
            }

            await generateResponse();
        }

        // Event listeners
        sendButton.addEventListener('click', sendMessage);
        stopButton.addEventListener('click', stopGeneration);
        regenerateButton.addEventListener('click', regenerateLastResponse);

        promptInput.addEventListener('keydown', (event) => {
            if (event.key === 'Enter' && !event.shiftKey) {
                event.preventDefault();
                sendMessage();
            }
        });

        // Initialize
        updateButtonStates();
        promptInput.focus();
    </script>
</body>
</html>
