    <script>
        class ChatAssistant {
            constructor() {
                this.messages = [];
                this.currentStreamHandle = null;
                this.isStreaming = false;
                this.lastAssistantMessageElement = null;

                this.elements = {
                    prompt: document.getElementById('prompt'),
                    send: document.getElementById('send'),
                    stop: document.getElementById('stop'),
                    regenerate: document.getElementById('regenerate'),
                    chatMessages: document.getElementById('chatMessages'),
                    statusLive: document.getElementById('statusLive'),
                };

                this.attachEventListeners();
            }

            attachEventListeners() {
                this.elements.send.addEventListener('click', () => this.sendMessage());
                this.elements.prompt.addEventListener('keydown', (e) => {
                    if (e.key === 'Enter' && !e.shiftKey) {
                        e.preventDefault();
                        this.sendMessage();
                    }
                });

                this.elements.stop.addEventListener('click', () => this.stopStreaming());
                this.elements.regenerate.addEventListener('click', () => this.regenerateLastMessage());
            }

            sendMessage() {
                const content = this.elements.prompt.value.trim();
                if (!content) return;

                this.addMessage('user', content);
                this.elements.prompt.value = '';
                this.elements.prompt.focus();
                this.updateButtonStates();

                this.startStreaming();
            }

            addMessage(role, content) {
                this.messages.push({ role, content });

                const messageDiv = document.createElement('div');
                messageDiv.className = `message ${role}`;
                messageDiv.setAttribute('role', 'article');

                const contentDiv = document.createElement('div');
                contentDiv.className = 'message-content';
                if (role === 'assistant') {
                    contentDiv.setAttribute('aria-label', 'Assistant message');
                }

                contentDiv.textContent = content;
                messageDiv.appendChild(contentDiv);

                if (this.elements.chatMessages.querySelector('.empty-state')) {
                    this.elements.chatMessages.innerHTML = '';
                }

                this.elements.chatMessages.appendChild(messageDiv);
                this.scrollToBottom();

                if (role === 'assistant') {
                    this.lastAssistantMessageElement = contentDiv;
                }
            }

            startStreaming() {
                if (this.isStreaming) return;

                this.isStreaming = true;
                this.updateButtonStates();

                const messageDiv = document.createElement('div');
                messageDiv.className = 'message assistant';
                messageDiv.setAttribute('role', 'article');

                const contentDiv = document.createElement('div');
                contentDiv.className = 'message-content streaming';
                contentDiv.setAttribute('aria-live', 'off');
                contentDiv.setAttribute('aria-label', 'Assistant is generating a response');

                messageDiv.appendChild(contentDiv);
                this.elements.chatMessages.appendChild(messageDiv);
                this.lastAssistantMessageElement = contentDiv;

                this.announceToScreenReader('Assistant is generating a response');
                this.scrollToBottom();

                this.currentStreamHandle = window.ChatSDK.streamChat(this.messages, {
                    onToken: (text) => this.handleToken(text),
                    onDone: () => this.handleStreamDone(),
                    onError: (error) => this.handleStreamError(error),
                });
            }

            handleToken(text) {
                if (!this.isStreaming || !this.lastAssistantMessageElement) return;

                this.lastAssistantMessageElement.textContent += text;
                this.scrollToBottom();
            }

            handleStreamDone() {
                if (!this.isStreaming) return;

                this.isStreaming = false;
                this.currentStreamHandle = null;

                if (this.lastAssistantMessageElement) {
                    this.lastAssistantMessageElement.classList.remove('streaming');
                    this.lastAssistantMessageElement.removeAttribute('aria-live');
                    const content = this.lastAssistantMessageElement.textContent;
                    if (this.messages[this.messages.length - 1]?.role !== 'assistant') {
                        this.messages.push({ role: 'assistant', content });
                    } else {
                        this.messages[this.messages.length - 1].content = content;
                    }
                }

                this.updateButtonStates();
                this.announceToScreenReader('Response generated successfully');
            }

            handleStreamError(error) {
                if (!this.isStreaming) return;

                this.isStreaming = false;
                this.currentStreamHandle = null;

                if (this.lastAssistantMessageElement) {
                    this.lastAssistantMessageElement.classList.remove('streaming');
                    this.lastAssistantMessageElement.removeAttribute('aria-live');

                    const errorDiv = document.createElement('div');
                    errorDiv.className = 'status-message error';
                    errorDiv.textContent = `Error: ${error.message || 'An error occurred'}`;
                    this.lastAssistantMessageElement.parentElement.appendChild(errorDiv);
                }

                this.updateButtonStates();
                this.announceToScreenReader(`Error: ${error.message || 'An error occurred while generating the response'}`);
            }

            stopStreaming() {
                if (!this.isStreaming || !this.currentStreamHandle) return;

                this.currentStreamHandle.abort();
                this.isStreaming = false;
                this.currentStreamHandle = null;

                if (this.lastAssistantMessageElement) {
                    this.lastAssistantMessageElement.classList.remove('streaming');
                    this.lastAssistantMessageElement.removeAttribute('aria-live');
                    const content = this.lastAssistantMessageElement.textContent;
                    if (this.messages[this.messages.length - 1]?.role !== 'assistant') {
                        this.messages.push({ role: 'assistant', content });
                    } else {
                        this.messages[this.messages.length - 1].content = content;
                    }

                    const statusDiv = document.createElement('div');
                    statusDiv.className = 'status-message';
                    statusDiv.textContent = 'Response stopped by user';
                    this.lastAssistantMessageElement.parentElement.appendChild(statusDiv);
                }

                this.updateButtonStates();
                this.announceToScreenReader('Response generation stopped by user');
                this.elements.stop.focus();
            }

            regenerateLastMessage() {
                const lastUserIndex = this.messages.length - 1;

                for (let i = this.messages.length - 1; i >= 0; i--) {
                    if (this.messages[i].role === 'user') {
                        this.messages = this.messages.slice(0, i + 1);
                        break;
                    }
                }

                while (this.elements.chatMessages.children.length > 0) {
                    const lastChild = this.elements.chatMessages.lastChild;
                    if (
                        lastChild.classList &&
                        lastChild.classList.contains('message') &&
                        lastChild.classList.contains('assistant')
                    ) {
                        lastChild.remove();
                    } else if (
                        lastChild.classList &&
                        lastChild.classList.contains('message')
                    ) {
                        break;
                    } else {
                        lastChild.remove();
                    }
                }

                this.updateButtonStates();
                this.startStreaming();
                this.announceToScreenReader('Regenerating response');
            }

            updateButtonStates() {
                const hasMessages = this.messages.length > 0;
                const hasUserMessages = this.messages.some((m) => m.role === 'user');

                this.elements.send.disabled = !this.elements.prompt.value.trim();
                this.elements.stop.disabled = !this.isStreaming;
                this.elements.regenerate.disabled = !hasUserMessages || this.isStreaming;
            }

            scrollToBottom() {
                requestAnimationFrame(() => {
                    this.elements.chatMessages.scrollTop = this.elements.chatMessages.scrollHeight;
                });
            }

            announceToScreenReader(message) {
                this.elements.statusLive.textContent = message;
            }
        }

        document.addEventListener('DOMContentLoaded', () => {
            new ChatAssistant();
        });
    </script>
</body>
