<script>
(function () {
  'use strict';

  var promptEl = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var logEl = document.getElementById('log');
  var emptyEl = document.getElementById('empty');
  var statusEl = document.getElementById('status');
  var alertEl = document.getElementById('alert');

  // Conversation as the user sees it. Each entry: {role, content, el}.
  var turns = [];
  // The stream currently running, or null.
  var run = null;
  var runSeq = 0;
  var msgSeq = 0;
  var stick = true;

  function setStatus(msg) { statusEl.textContent = msg; }
  function setAlert(msg) { alertEl.textContent = msg || ''; }

  function isNearBottom() {
    return logEl.scrollHeight - logEl.scrollTop - logEl.clientHeight < 40;
  }

  logEl.addEventListener('scroll', function () {
    stick = isNearBottom();
  });

  function scrollToEnd() {
    if (stick) logEl.scrollTop = logEl.scrollHeight;
  }

  function createMessage(role) {
    if (emptyEl.parentNode) emptyEl.parentNode.removeChild(emptyEl);
    msgSeq += 1;
    var article = document.createElement('article');
    article.className = 'msg ' + role;

    var heading = document.createElement('h2');
    heading.id = 'msg-title-' + msgSeq;
    heading.textContent = role === 'user' ? 'You' : 'Assistant';
    article.setAttribute('aria-labelledby', heading.id);

    var textEl = document.createElement('div');
    textEl.className = 'text';

    article.appendChild(heading);
    article.appendChild(textEl);
    logEl.appendChild(article);
    return { article: article, textEl: textEl };
  }

  function addNote(article, text) {
    var p = document.createElement('p');
    p.className = 'note';
    p.textContent = text;
    article.appendChild(p);
  }

  function updateControls() {
    var busy = !!run;
    var hasUser = turns.some(function (t) { return t.role === 'user'; });
    sendBtn.disabled = busy;
    stopBtn.disabled = !busy;
    regenBtn.disabled = busy || !hasUser;
  }

  // Keep keyboard focus on a usable control when the focused button gets disabled.
  function ensureFocus() {
    var a = document.activeElement;
    if (!a || a === document.body || a.disabled) {
      promptEl.focus();
    }
  }

  function startStream() {
    var id = ++runSeq;
    var parts = createMessage('assistant');
    parts.article.setAttribute('aria-busy', 'true');

    var indicator = document.createElement('p');
    indicator.className = 'note';
    var dots = document.createElement('span');
    dots.className = 'dots';
    dots.setAttribute('aria-hidden', 'true');
    for (var i = 0; i < 3; i++) dots.appendChild(document.createElement('i'));
    indicator.appendChild(dots);
    indicator.appendChild(document.createTextNode('Generating answer…'));
    parts.article.appendChild(indicator);

    var r = {
      id: id,
      handle: null,
      article: parts.article,
      textEl: parts.textEl,
      indicator: indicator,
      text: ''
    };

    run = r;
    stick = true;
    setAlert('');
    setStatus('Generating answer…');
    updateControls();
    ensureFocus();
    scrollToEnd();

    var messages = turns.map(function (t) {
      return { role: t.role, content: t.content };
    });

    var callbacks = {
      onToken: function (chunk) {
        // Ignore tokens from a stream that is no longer active (for example,
        // a chunk that was in flight when the user pressed Stop).
        if (run !== r) return;
        r.text += String(chunk);
        r.textEl.textContent = r.text;
        scrollToEnd();
      },
      onDone: function () {
        if (run !== r) return;
        complete(r);
      },
      onError: function (err) {
        if (run !== r) return;
        fail(r, err);
      }
    };

    try {
      r.handle = window.ChatSDK.streamChat(messages, callbacks);
    } catch (err) {
      if (run === r) fail(r, err);
    }
  }

  function settle(r) {
    if (run === r) run = null;
    if (r.indicator.parentNode) r.indicator.parentNode.removeChild(r.indicator);
    r.article.setAttribute('aria-busy', 'false');
  }

  function complete(r) {
    settle(r);
    if (r.text) {
      turns.push({ role: 'assistant', content: r.text, el: r.article });
      setStatus('Answer finished.');
    } else {
      r.article.parentNode.removeChild(r.article);
      setStatus('Answer finished with no text.');
    }
    updateControls();
    ensureFocus();
  }

  function fail(r, err) {
    settle(r);
    var msg = (err && err.message) ? err.message : String(err || 'Unknown error');
    if (r.text) {
      addNote(r.article, 'Stopped by an error. The answer is incomplete.');
      turns.push({ role: 'assistant', content: r.text, el: r.article });
    } else if (r.article.parentNode) {
      r.article.parentNode.removeChild(r.article);
    }
    setAlert('Error: ' + msg + '. Use Regenerate answer to try again.');
    setStatus('Stopped because of an error.');
    updateControls();
    ensureFocus();
  }

  function stop() {
    var r = run;
    if (!r) return;
    // Clear the active stream first so any in-flight token is ignored.
    run = null;
    if (r.handle && typeof r.handle.abort === 'function') {
      try { r.handle.abort(); } catch (e) { /* ignore */ }
    }
    if (r.indicator.parentNode) r.indicator.parentNode.removeChild(r.indicator);
    r.article.setAttribute('aria-busy', 'false');

    if (r.text) {
      addNote(r.article, 'Stopped. Partial answer kept.');
      turns.push({ role: 'assistant', content: r.text, el: r.article });
      setStatus('Stopped. Partial answer kept.');
    } else {
      r.article.parentNode.removeChild(r.article);
      setStatus('Stopped before any answer text arrived.');
    }
    updateControls();
    ensureFocus();
  }

  function send() {
    if (run) {
      setStatus('An answer is still generating. Press Stop to end it first.');
      return;
    }
    var text = promptEl.value.trim();
    if (!text) {
      setStatus('Type a message before sending.');
      promptEl.focus();
      return;
    }
    promptEl.value = '';
    var parts = createMessage('user');
    parts.textEl.textContent = text;
    turns.push({ role: 'user', content: text, el: parts.article });
    startStream();
    promptEl.focus();
  }

  function regenerate() {
    if (run) return;
    var last = turns[turns.length - 1];
    if (last && last.role === 'assistant') {
      turns.pop();
      if (last.el.parentNode) last.el.parentNode.removeChild(last.el);
    }
    var lastNow = turns[turns.length - 1];
    if (!lastNow || lastNow.role !== 'user') {
      updateControls();
      return;
    }
    startStream();
  }

  sendBtn.addEventListener('click', send);
  stopBtn.addEventListener('click', stop);
  regenBtn.addEventListener('click', regenerate);

  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });

  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && run) {
      e.preventDefault();
      stop();
    }
  });

  if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
    setAlert('The chat service is not available on this page.');
    setStatus('Unavailable.');
  }

  updateControls();
})();
</script>
</body>
</html>
