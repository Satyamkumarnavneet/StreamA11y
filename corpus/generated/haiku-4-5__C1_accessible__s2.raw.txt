    <script>
        const messagesContainer = document.getElementById('messages');
        const promptInput = document.getElementById('prompt');
        const sendButton = document.getElementById('send');
        const stopButton = document.getElementById('stop');
        const regenerateButton = document.getElementById('regenerate');

        let messages = [];
        let isGenerating = false;
        let currentHandle = null;
        let lastUserMessage = null;

        // Auto-resize textarea
        function resizeTextarea() {
            promptInput.style.height = 'auto';
            promptInput.style.height = Math.min(promptInput.scrollHeight, 100) + 'px';
        }

        // Clear the empty state
        function clearEmptyState() {
            const emptyState = messagesContainer.querySelector('.empty-state');
            if (emptyState) {
                emptyState.remove();
            }
        }

        // Add message to conversation
        function addMessage(role, content) {
            clearEmptyState();

            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${role}`;
            messageDiv.setAttribute('role', 'article');

            const contentDiv = document.createElement('div');
            contentDiv.className = 'message-content';
            contentDiv.textContent = content;

            messageDiv.appendChild(contentDiv);
            messagesContainer.appendChild(messageDiv);
            messagesContainer.scrollTop = messagesContainer.scrollHeight;

            return messageDiv;
        }

        // Update the last message (for streaming)
        function updateLastMessage(content) {
            const lastMessage = messagesContainer.querySelector('.message:last-child .message-content');
            if (lastMessage) {
                lastMessage.textContent = content;
            }
            messagesContainer.scrollTop = messagesContainer.scrollHeight;
        }

        // Show status below the last message
        function setStatus(status, message) {
            let statusDiv = messagesContainer.querySelector('.status-message');
            if (!statusDiv) {
                statusDiv = document.createElement('div');
                statusDiv.className = 'status-message';
                messagesContainer.appendChild(statusDiv);
            }

            const indicator = document.createElement('span');
            indicator.className = `status-indicator ${status}`;
            indicator.setAttribute('aria-hidden', 'true');

            statusDiv.innerHTML = '';
            statusDiv.appendChild(indicator);
            statusDiv.appendChild(document.createTextNode(' ' + message));

            messagesContainer.scrollTop = messagesContainer.scrollHeight;
        }

        // Clear status message
        function clearStatus() {
            const statusDiv = messagesContainer.querySelector('.status-message');
            if (statusDiv) {
                statusDiv.remove();
            }
        }

        // Update UI state
        function updateUIState() {
            sendButton.disabled = isGenerating || promptInput.value.trim() === '';
            stopButton.disabled = !isGenerating;
            regenerateButton.disabled = isGenerating || messages.length === 0 || messages[messages.length - 1].role !== 'assistant';
        }

        // Send message
        async function sendMessage() {
            const content = promptInput.value.trim();
            if (!content || isGenerating) return;

            clearStatus();
            messages.push({ role: 'user', content });
            lastUserMessage = content;
            addMessage('user', content);
            promptInput.value = '';
            resizeTextarea();

            await getAssistantResponse();
        }

        // Get assistant response
        async function getAssistantResponse() {
            isGenerating = true;
            updateUIState();

            let assistantMessage = '';
            addMessage('assistant', '');
            setStatus('generating', 'Generating response...');

            try {
                currentHandle = window.ChatSDK.streamChat(messages, {
                    onToken: (text) => {
                        assistantMessage += text;
                        updateLastMessage(assistantMessage);
                    },
                    onDone: () => {
                        messages.push({ role: 'assistant', content: assistantMessage });
                        isGenerating = false;
                        currentHandle = null;
                        clearStatus();
                        setStatus('done', 'Response complete');
                        updateUIState();
                    },
                    onError: (error) => {
                        isGenerating = false;
                        currentHandle = null;
                        clearStatus();
                        
                        const errorDiv = document.createElement('div');
                        errorDiv.className = 'error-message';
                        errorDiv.setAttribute('role', 'alert');
                        errorDiv.textContent = `Error: ${error.message || 'Failed to get response'}`;
                        messagesContainer.appendChild(errorDiv);
                        
                        setStatus('error', 'Error occurred');
                        updateUIState();
                    }
                });
            } catch (error) {
                isGenerating = false;
                currentHandle = null;
                clearStatus();
                
                const errorDiv = document.createElement('div');
                errorDiv.className = 'error-message';
                errorDiv.setAttribute('role', 'alert');
                errorDiv.textContent = `Error: ${error.message || 'Failed to get response'}`;
                messagesContainer.appendChild(errorDiv);
                
                setStatus('error', 'Error occurred');
                updateUIState();
            }
        }

        // Stop generation
        function stopGeneration() {
            if (currentHandle) {
                currentHandle.abort();
                isGenerating = false;
                currentHandle = null;
                clearStatus();
                setStatus('stopped', 'Response stopped');
                updateUIState();
            }
        }

        // Regenerate last response
        async function regenerateResponse() {
            if (!lastUserMessage) return;

            clearStatus();
            // Remove the last assistant message
            messages = messages.filter(m => !(m.role === 'assistant' && m.content === messages[messages.length - 1].content));
            
            // Remove the last assistant message DOM element
            const lastMessage = messagesContainer.querySelector('.message:last-child');
            if (lastMessage) {
                lastMessage.remove();
            }

            await getAssistantResponse();
        }

        // Event listeners
        sendButton.addEventListener('click', sendMessage);
        stopButton.addEventListener('click', stopGeneration);
        regenerateButton.addEventListener('click', regenerateResponse);

        promptInput.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' && !e.shiftKey) {
                e.preventDefault();
                sendMessage();
            }
        });

        promptInput.addEventListener('input', () => {
            resizeTextarea();
            updateUIState();
        });

        // Initial state
        updateUIState();
    </script>
</body>
