<script>
(function () {
  'use strict';

  var $ = function (id) { return document.getElementById(id); };
  var logEl = $('log');
  var promptEl = $('prompt');
  var sendBtn = $('send');
  var stopBtn = $('stop');
  var regenBtn = $('regenerate');
  var statusEl = $('status');
  var form = $('composer');

  var history = [];   // { role, content, note }
  var current = null; // active stream state
  var genSeq = 0;

  function setStatus(text) {
    statusEl.textContent = text;
  }

  function hasUserMessage() {
    return history.some(function (m) { return m.role === 'user'; });
  }

  function chatAvailable() {
    return !!(window.ChatSDK && typeof window.ChatSDK.streamChat === 'function');
  }

  function updateControls() {
    var busy = !!current;
    var hasText = promptEl.value.trim().length > 0;
    sendBtn.disabled = busy || !hasText || !chatAvailable();
    stopBtn.disabled = !busy;
    regenBtn.disabled = busy || !hasUserMessage() || !chatAvailable();
    logEl.setAttribute('aria-busy', String(busy));
  }

  // If the focused control was just disabled, the browser drops focus.
  // Move focus to the message box so keyboard users are not stranded.
  function ensureFocus() {
    var a = document.activeElement;
    if (!a || a === document.body || a.disabled) {
      promptEl.focus();
    }
  }

  function refresh() {
    updateControls();
    ensureFocus();
  }

  function describeError(err) {
    if (err && typeof err.message === 'string' && err.message) {
      return err.message.slice(0, 300);
    }
    if (typeof err === 'string' && err) {
      return err.slice(0, 300);
    }
    return 'The answer could not be completed.';
  }

  function buildMessage(entry) {
    var article = document.createElement('article');
    article.className = 'msg ' + entry.role;

    var head = document.createElement('div');
    head.className = 'msg-head';
    head.textContent = entry.role === 'user' ? 'You' : 'Assistant';
    article.appendChild(head);

    var body = document.createElement('div');
    body.className = 'msg-body';
    body.textContent = entry.content;
    article.appendChild(body);

    var note = document.createElement('div');
    note.className = 'msg-note';
    note.textContent = entry.note || '';
    note.hidden = !entry.note;
    article.appendChild(note);

    return { article: article, body: body, note: note };
  }

  function renderLog() {
    logEl.textContent = '';
    history.forEach(function (entry) {
      logEl.appendChild(buildMessage(entry).article);
    });
  }

  function scrollToEnd(force) {
    var nearBottom = logEl.scrollHeight - logEl.scrollTop - logEl.clientHeight < 80;
    if (force || nearBottom) {
      logEl.scrollTop = logEl.scrollHeight;
    }
  }

  function finish(st, kind, errMsg) {
    if (st.closed) return;      // only the first outcome counts
    st.closed = true;
    if (current === st) current = null;

    if (kind === 'complete') {
      st.entry.note = '';
      st.note.textContent = '';
      st.note.hidden = true;
      setStatus('Answer finished.');
    } else if (kind === 'stopped') {
      st.entry.note = 'Stopped. The partial answer is kept.';
      st.note.textContent = st.entry.note;
      st.note.hidden = false;
      setStatus('Answer stopped. The partial answer is kept.');
    } else if (kind === 'error') {
      st.entry.note = 'Stopped because of an error.';
      st.note.textContent = st.entry.note;
      st.note.hidden = false;
      setStatus('The answer stopped because of an error.');

      var alertEl = document.createElement('p');
      alertEl.className = 'msg-error';
      alertEl.setAttribute('role', 'alert');
      alertEl.textContent = 'Error: ' + errMsg + ' Use "Regenerate last answer" to try again.';
      st.article.appendChild(alertEl);
    }

    st.article.removeAttribute('aria-busy');
    refresh();
    scrollToEnd(false);
  }

  function startAssistant() {
    var entry = { role: 'assistant', content: '', note: 'Generating…' };
    history.push(entry);

    var view = buildMessage(entry);
    view.article.setAttribute('aria-busy', 'true');
    logEl.appendChild(view.article);

    var st = {
      id: ++genSeq,
      entry: entry,
      article: view.article,
      body: view.body,
      note: view.note,
      handle: null,
      closed: false
    };
    current = st;

    refresh();
    setStatus('Generating answer…');
    scrollToEnd(true);

    // Send only finished turns; skip the empty placeholder and empty partials.
    var payload = history
      .filter(function (m) { return m.role === 'user' || m.content.length > 0; })
      .map(function (m) { return { role: m.role, content: m.content }; });

    try {
      st.handle = window.ChatSDK.streamChat(payload, {
        onToken: function (text) {
          if (st.closed || current !== st) return;   // ignore late chunks after stop
          var t = String(text);
          st.entry.content += t;
          st.body.appendChild(document.createTextNode(t));
          scrollToEnd(false);
        },
        onDone: function () {
          if (st.closed || current !== st) return;
          finish(st, 'complete');
        },
        onError: function (err) {
          if (st.closed || current !== st) return;
          finish(st, 'error', describeError(err));
        }
      });
    } catch (err) {
      finish(st, 'error', describeError(err));
    }
  }

  function stopAnswer() {
    var st = current;
    if (!st || st.closed) return;
    try {
      if (st.handle && typeof st.handle.abort === 'function') {
        st.handle.abort();
      }
    } catch (e) {
      // Ignore abort failures; the local stream is closed either way.
    }
    finish(st, 'stopped');
  }

  function regenerate() {
    if (current || !hasUserMessage()) return;
    var idx = -1;
    for (var i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') { idx = i; break; }
    }
    history = history.slice(0, idx + 1);
    renderLog();
    startAssistant();
  }

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    if (current) {
      setStatus('An answer is still generating. Press Stop to end it first.');
      return;
    }
    var text = promptEl.value.trim();
    if (!text) {
      setStatus('Type a message before sending.');
      promptEl.focus();
      return;
    }
    if (!chatAvailable()) {
      setStatus('The chat service is not available.');
      return;
    }

    promptEl.value = '';
    var userEntry = { role: 'user', content: text, note: '' };
    history.push(userEntry);
    logEl.appendChild(buildMessage(userEntry).article);
    startAssistant();
  });

  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      form.requestSubmit();
    }
  });

  promptEl.addEventListener('input', updateControls);

  stopBtn.addEventListener('click', stopAnswer);
  regenBtn.addEventListener('click', regenerate);

  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && current) {
      e.preventDefault();
      stopAnswer();
    }
  });

  // Initial state
  if (!chatAvailable()) {
    setStatus('The chat service is not available. Reload the page or contact the site owner.');
  } else {
    setStatus('Ready. Type a message and press Send.');
  }
  updateControls();
  promptEl.focus();
})();
</script>
</body>
</html>
