<script>
(function () {
  'use strict';

  var $ = function (id) { return document.getElementById(id); };
  var els = {
    log: $('log'), empty: $('empty'), list: $('messages'),
    banner: $('error-banner'), state: $('state'), stateText: $('state-text'),
    announcer: $('announcer'), form: $('composer'), prompt: $('prompt'),
    promptError: $('prompt-error'), send: $('send'), stop: $('stop'),
    regenerate: $('regenerate')
  };

  var history = [];      // {role, content, status, li, textNode, pill, note}
  var current = null;    // active streaming session, or null
  var announceTimer = null;
  var bannerTimer = null;

  var STATE_LABELS = {
    idle: 'Ready',
    generating: 'Assistant is generating an answer…',
    stopped: 'Stopped. The partial answer was kept.',
    done: 'Answer finished.',
    error: 'Error. The answer could not be completed.'
  };

  function setState(kind, textOverride) {
    els.state.setAttribute('data-kind', kind);
    els.stateText.textContent = textOverride || STATE_LABELS[kind];
  }

  function announce(text) {
    clearTimeout(announceTimer);
    els.announcer.textContent = '';
    announceTimer = setTimeout(function () { els.announcer.textContent = text; }, 60);
  }

  function showBanner(text) {
    clearTimeout(bannerTimer);
    els.banner.textContent = '';
    if (!text) return;
    bannerTimer = setTimeout(function () { els.banner.textContent = text; }, 60);
  }

  function setFieldError(text) {
    els.promptError.textContent = text || '';
    if (text) els.prompt.setAttribute('aria-invalid', 'true');
    else els.prompt.removeAttribute('aria-invalid');
  }

  function nearBottom() {
    return els.log.scrollHeight - els.log.scrollTop - els.log.clientHeight < 80;
  }
  function scrollToBottom() { els.log.scrollTop = els.log.scrollHeight; }

  function addMessage(role, content, status) {
    var msg = { role: role, content: content, status: status };
    var li = document.createElement('li');
    li.className = 'msg ' + role;

    var head = document.createElement('div');
    head.className = 'msg-head';
    var who = document.createElement('span');
    who.className = 'who';
    who.textContent = role === 'user' ? 'You' : 'Assistant';
    var pill = document.createElement('span');
    pill.className = 'pill';
    head.appendChild(who);
    head.appendChild(pill);

    var body = document.createElement('div');
    body.className = 'msg-body';
    var textNode = document.createTextNode(content);
    body.appendChild(textNode);

    var note = document.createElement('p');
    note.className = 'msg-note';

    li.appendChild(head);
    li.appendChild(body);
    li.appendChild(note);

    msg.li = li; msg.textNode = textNode; msg.pill = pill; msg.note = note;
    var stick = nearBottom() || role === 'user';
    els.list.appendChild(li);
    els.empty.hidden = true;
    history.push(msg);
    refreshMessage(msg);
    if (stick) scrollToBottom();
    return msg;
  }

  function refreshMessage(msg) {
    if (msg.role !== 'assistant') return;
    var pillText = '', noteText = '', err = false;
    if (msg.status === 'generating') {
      pillText = 'Generating…';
      msg.li.setAttribute('aria-busy', 'true');
    } else {
      msg.li.removeAttribute('aria-busy');
      if (msg.status === 'stopped') {
        pillText = 'Stopped';
        noteText = msg.content
          ? 'You stopped this answer. The partial text above was kept.'
          : 'You stopped this answer before any text arrived.';
      } else if (msg.status === 'error') {
        pillText = 'Error'; err = true;
        noteText = msg.content
          ? 'This answer was interrupted by an error. The partial text above was kept.'
          : 'An error occurred before any text arrived.';
      } else if (msg.status === 'done' && !msg.content) {
        noteText = 'The assistant returned an empty answer.';
      }
    }
    msg.pill.textContent = pillText;
    msg.pill.className = 'pill' + (err ? ' err' : '');
    msg.note.textContent = noteText;
    msg.note.className = 'msg-note' + (err ? ' err' : '');
  }

  function updateControls() {
    var ae = document.activeElement;
    var streaming = !!current;
    var hasUser = history.some(function (m) { return m.role === 'user'; });
    els.send.disabled = streaming;
    els.stop.disabled = !streaming;
    els.regenerate.disabled = streaming || !hasUser;
    if (ae && (ae === els.send || ae === els.stop || ae === els.regenerate) && ae.disabled) {
      (streaming ? els.stop : els.prompt).focus();
    }
  }

  function buildPayload() {
    return history
      .filter(function (m) { return m.content && m.content.trim() !== ''; })
      .map(function (m) { return { role: m.role, content: m.content }; });
  }

  function closeSession(session, status) {
    session.closed = true;
    if (current === session) current = null;
    session.msg.status = status;
    refreshMessage(session.msg);
  }

  function errorMessage(err) {
    if (err && typeof err.message === 'string' && err.message) return err.message;
    if (typeof err === 'string' && err) return err;
    return 'Unknown error';
  }

  function fail(session, err) {
    var hadText = !!session.msg.content;
    closeSession(session, 'error');
    setState('error');
    showBanner('Error: ' + errorMessage(err) + '. ' +
      (hadText ? 'The partial answer was kept. ' : '') +
      'Select Regenerate to try again.');
    updateControls();
  }

  function startStream() {
    var payload = buildPayload();
    var msg = addMessage('assistant', '', 'generating');
    var session = { msg: msg, closed: false, handle: null };
    current = session;
    showBanner('');
    setFieldError('');
    setState('generating');
    announce('Assistant is generating an answer.');
    updateControls();

    var callbacks = {
      onToken: function (text) {
        if (session.closed) return;           // ignore late chunks after stop/finish/error
        if (typeof text !== 'string') text = text == null ? '' : String(text);
        if (!text) return;
        var stick = nearBottom();
        msg.content += text;
        msg.textNode.appendData(text);
        if (stick) scrollToBottom();
      },
      onDone: function () {
        if (session.closed) return;
        closeSession(session, 'done');
        setState('done');
        announce('Answer finished.');
        updateControls();
      },
      onError: function (err) {
        if (session.closed) return;
        fail(session, err);
      }
    };

    try {
      if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
        throw new Error('The chat service is not available');
      }
      session.handle = window.ChatSDK.streamChat(payload, callbacks);
    } catch (e) {
      if (!session.closed) fail(session, e);
    }
  }

  function send() {
    if (current) {
      announce('Please wait for the current answer to finish, or stop it first.');
      return;
    }
    var text = els.prompt.value.trim();
    if (!text) {
      setFieldError('Enter a message before sending.');
      els.prompt.focus();
      return;
    }
    setFieldError('');
    addMessage('user', text, 'sent');
    els.prompt.value = '';
    autoResize();
    startStream();
  }

  function stop() {
    var session = current;
    if (!session) return;
    session.closed = true;
    try { if (session.handle && typeof session.handle.abort === 'function') session.handle.abort(); }
    catch (e) { /* ignore abort failures; the answer is already treated as stopped */ }
    closeSession(session, 'stopped');
    setState('stopped');
    announce(session.msg.content
      ? 'Stopped. The partial answer was kept.'
      : 'Stopped before any text arrived.');
    updateControls();
  }

  function regenerate() {
    if (current) return;
    var idx = -1;
    for (var i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') { idx = i; break; }
    }
    if (idx < 0) return;
    while (history.length > idx + 1) {
      var m = history.pop();
      if (m.li && m.li.parentNode) m.li.parentNode.removeChild(m.li);
    }
    startStream();
  }

  function autoResize() {
    els.prompt.style.height = 'auto';
    var h = els.prompt.scrollHeight + 4;
    els.prompt.style.height = h + 'px';
  }

  els.form.addEventListener('submit', function (e) { e.preventDefault(); send(); });
  els.stop.addEventListener('click', stop);
  els.regenerate.addEventListener('click', regenerate);
  els.send.addEventListener('click', function () {
    // keep keyboard/mouse users in the message box after sending
    setTimeout(function () { if (current) els.prompt.focus(); }, 0);
  });
  els.prompt.addEventListener('input', function () {
    if (els.promptError.textContent) setFieldError('');
    autoResize();
  });
  els.prompt.addEventListener('keydown', function (e) {
    if (e.isComposing || e.keyCode === 229) return;
    if (e.key === 'Enter' && !e.shiftKey && !e.ctrlKey && !e.altKey && !e.metaKey) {
      e.preventDefault();
      send();
    } else if (e.key === 'Escape' && current) {
      e.preventDefault();
      stop();
    }
  });
  window.addEventListener('pagehide', function () {
    if (current && current.handle && typeof current.handle.abort === 'function') {
      try { current.handle.abort(); } catch (e) {}
    }
  });

  setState('idle');
  updateControls();
  autoResize();
})();
</script>
</body>
</html>
