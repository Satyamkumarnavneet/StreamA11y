    <script>
        (function() {
            'use strict';

            // DOM elements
            const messagesContainer = document.getElementById('messages');
            const emptyState = document.getElementById('empty-state');
            const promptInput = document.getElementById('prompt');
            const sendBtn = document.getElementById('send');
            const stopBtn = document.getElementById('stop');
            const regenerateBtn = document.getElementById('regenerate');
            const statusAnnouncer = document.getElementById('status-announcer');

            // State
            let conversationHistory = []; // Array of {role, content}
            let currentStreamHandle = null;
            let currentStreamId = 0; // Incremented each stream to detect stale callbacks
            let isStreaming = false;
            let currentAssistantMessageEl = null;
            let currentAssistantContentEl = null;

            // Announce status to screen readers
            function announceStatus(message) {
                statusAnnouncer.textContent = '';
                // Force reannouncement by clearing then setting
                requestAnimationFrame(() => {
                    statusAnnouncer.textContent = message;
                });
            }

            // Update UI state based on streaming status
            function updateUIState() {
                sendBtn.disabled = isStreaming || promptInput.value.trim() === '';
                stopBtn.disabled = !isStreaming;
                regenerateBtn.disabled = isStreaming || conversationHistory.length === 0 || conversationHistory[conversationHistory.length - 1].role !== 'assistant';
                promptInput.disabled = isStreaming;

                if (emptyState && conversationHistory.length > 0) {
                    emptyState.remove();
                }
            }

            // Create a message element
            function createMessageElement(role, content, status) {
                const messageDiv = document.createElement('div');
                messageDiv.className = `message message-${role}`;
                messageDiv.setAttribute('role', 'article');

                const labelDiv = document.createElement('div');
                labelDiv.className = 'message-label';
                labelDiv.textContent = role === 'user' ? 'You' : 'Assistant';
                messageDiv.appendChild(labelDiv);

                const contentP = document.createElement('p');
                contentP.className = 'message-content';
                contentP.textContent = content;
                messageDiv.appendChild(contentP);

                if (status) {
                    const statusDiv = document.createElement('div');
                    statusDiv.className = 'message-status';
                    statusDiv.textContent = status;
                    statusDiv.setAttribute('aria-live', 'off');
                    messageDiv.appendChild(statusDiv);
                }

                return { messageDiv, contentP };
            }

            // Add user message to display
            function addUserMessage(content) {
                const { messageDiv } = createMessageElement('user', content);
                messagesContainer.appendChild(messageDiv);
                scrollToBottom();
            }

            // Start assistant message (for streaming)
            function startAssistantMessage() {
                const { messageDiv, contentP } = createMessageElement('assistant', '', 'Generating...');
                messageDiv.setAttribute('aria-busy', 'true');
                currentAssistantMessageEl = messageDiv;
                currentAssistantContentEl = contentP;
                messagesContainer.appendChild(messageDiv);
                scrollToBottom();
            }

            // Update assistant message status
            function updateAssistantStatus(status, isError) {
                if (!currentAssistantMessageEl) return;

                let statusEl = currentAssistantMessageEl.querySelector('.message-status');
                if (!statusEl) {
                    statusEl = document.createElement('div');
                    statusEl.className = 'message-status';
                    statusEl.setAttribute('aria-live', 'off');
                    currentAssistantMessageEl.appendChild(statusEl);
                }

                statusEl.textContent = status;
                if (isError) {
                    statusEl.classList.add('message-error');
                } else {
                    statusEl.classList.remove('message-error');
                }
            }

            // Finalize assistant message
            function finalizeAssistantMessage() {
                if (currentAssistantMessageEl) {
                    currentAssistantMessageEl.removeAttribute('aria-busy');
                }
                currentAssistantMessageEl = null;
                currentAssistantContentEl = null;
            }

            // Scroll to bottom of messages
            function scrollToBottom() {
                messagesContainer.scrollTop = messagesContainer.scrollHeight;
            }

            // Send a message and start streaming
            function sendMessage() {
                const content = promptInput.value.trim();
                if (!content || isStreaming) return;

                // Add user message to history and display
                conversationHistory.push({ role: 'user', content });
                addUserMessage(content);
                promptInput.value = '';

                // Start streaming
                startStream();
            }

            // Start streaming response
            function startStream() {
                isStreaming = true;
                currentStreamId++;
                const myStreamId = currentStreamId;

                updateUIState();
                startAssistantMessage();
                announceStatus('Assistant is generating a response.');

                // Prepare messages for API
                const messages = conversationHistory.map(m => ({
                    role: m.role,
                    content: m.content
                }));

                let accumulatedContent = '';

                currentStreamHandle = window.ChatSDK.streamChat(messages, {
                    onToken: function(text) {
                        // Ignore if this is a stale stream
                        if (myStreamId !== currentStreamId) return;

                        accumulatedContent += text;
                        if (currentAssistantContentEl) {
                            currentAssistantContentEl.textContent = accumulatedContent;
                            scrollToBottom();
                        }
                    },
                    onDone: function() {
                        // Ignore if this is a stale stream
                        if (myStreamId !== currentStreamId) return;

                        // Save to history
                        conversationHistory.push({ role: 'assistant', content: accumulatedContent });

                        updateAssistantStatus('');
                        // Remove the status element if empty
                        if (currentAssistantMessageEl) {
                            const statusEl = currentAssistantMessageEl.querySelector('.message-status');
                            if (statusEl && !statusEl.textContent) {
                                statusEl.remove();
                            }
                        }

                        finalizeAssistantMessage();
                        isStreaming = false;
                        currentStreamHandle = null;
                        updateUIState();
                        announceStatus('Response complete.');
                        promptInput.focus();
                    },
                    onError: function(error) {
                        // Ignore if this is a stale stream
                        if (myStreamId !== currentStreamId) return;

                        const errorMessage = error && error.message ? error.message : 'An error occurred';
                        
                        // If we have partial content, keep it
                        if (accumulatedContent) {
                            conversationHistory.push({ role: 'assistant', content: accumulatedContent });
                        }

                        updateAssistantStatus('Error: ' + errorMessage, true);
                        finalizeAssistantMessage();
                        isStreaming = false;
                        currentStreamHandle = null;
                        updateUIState();
                        announceStatus('Error: ' + errorMessage);
                        promptInput.focus();
                    }
                });
            }

            // Stop current stream
            function stopStream() {
                if (!isStreaming || !currentStreamHandle) return;

                // Invalidate current stream
                currentStreamId++;
                
                currentStreamHandle.abort();
                currentStreamHandle = null;

                // Save partial content to history
                if (currentAssistantContentEl) {
                    const partialContent = currentAssistantContentEl.textContent;
                    if (partialContent) {
                        conversationHistory.push({ role: 'assistant', content: partialContent });
                    }
                }

                updateAssistantStatus('Stopped');
                finalizeAssistantMessage();
                isStreaming = false;
                updateUIState();
                announceStatus('Generation stopped.');
                promptInput.focus();
            }

            // Regenerate last response
            function regenerate() {
                if (isStreaming) return;
                if (conversationHistory.length === 0) return;

                // Find and remove the last assistant message
                if (conversationHistory[conversationHistory.length - 1].role === 'assistant') {
                    conversationHistory.pop();

                    // Remove from display
                    const lastMessage = messagesContainer.querySelector('.message-assistant:last-of-type');
                    if (lastMessage) {
                        lastMessage.remove();
                    }
                }

                // Make sure we have a user message to regenerate from
                if (conversationHistory.length === 0 || conversationHistory[conversationHistory.length - 1].role !== 'user') {
                    updateUIState();
                    return;
                }

                // Start new stream
                startStream();
            }

            // Event listeners
            sendBtn.addEventListener('click', sendMessage);

            stopBtn.addEventListener('click', stopStream);

            regenerateBtn.addEventListener('click', regenerate);

            promptInput.addEventListener('input', function() {
                updateUIState();
                // Auto-resize textarea
                this.style.height = 'auto';
                this.style.height = Math.min(this.scrollHeight, 128) + 'px';
            });

            promptInput.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    if (!sendBtn.disabled) {
                        sendMessage();
                    }
                }
            });

            // Initial state
            updateUIState();
            promptInput.focus();
        })();
    </script>
</body>
</html>
