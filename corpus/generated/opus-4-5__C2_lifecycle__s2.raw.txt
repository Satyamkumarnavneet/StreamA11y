    <div class="chat-container">
        <header class="chat-header">
            <h1>Chat Assistant</h1>
        </header>

        <main 
            id="messages" 
            class="messages-container" 
            role="log" 
            aria-label="Conversation history"
            aria-live="off"
            tabindex="0"
        >
            <div class="empty-state" id="empty-state">
                <p>Start a conversation by typing a message below.</p>
            </div>

            id="status-announcer"

    <script>
        (function() {
            'use strict';

            const messagesContainer = document.getElementById('messages');
            const emptyState = document.getElementById('empty-state');
            const promptInput = document.getElementById('prompt');
            const sendButton = document.getElementById('send');
            const stopButton = document.getElementById('stop');
            const regenerateButton = document.getElementById('regenerate');
            const statusAnnouncer = document.getElementById('status-announcer');

            let conversationHistory = [];
            let currentStreamHandle = null;
            let currentStreamId = null;
            let isStreaming = false;
            let currentAssistantMessageEl = null;
            let currentContentEl = null;
            let currentStatusEl = null;
            let streamedContent = '';
            let focusedElementBeforeStream = null;

            function announceStatus(message) {
                statusAnnouncer.textContent = '';
                void statusAnnouncer.offsetHeight;
                statusAnnouncer.textContent = message;
            }

            function updateButtonStates() {
                const hasContent = promptInput.value.trim().length > 0;
                const hasAssistantMessage = conversationHistory.some(m => m.role === 'assistant');

                sendButton.disabled = isStreaming || !hasContent;
                sendButton.setAttribute('aria-disabled', sendButton.disabled);

                stopButton.disabled = !isStreaming;
                stopButton.setAttribute('aria-disabled', stopButton.disabled);

                regenerateButton.disabled = isStreaming || !hasAssistantMessage;
                regenerateButton.setAttribute('aria-disabled', regenerateButton.disabled);
            }

            function scrollToBottom() {
                messagesContainer.scrollTop = messagesContainer.scrollHeight;
            }

            function createMessageElement(role, content, status) {
                const messageDiv = document.createElement('div');
                messageDiv.className = `message message-${role}`;
                if (status) {
                    messageDiv.classList.add(`message-${status}`);
                }

                const labelDiv = document.createElement('div');
                labelDiv.className = 'message-label';
                labelDiv.textContent = role === 'user' ? 'You' : 'Assistant';

                const contentDiv = document.createElement('div');
                contentDiv.className = 'message-content';
                contentDiv.textContent = content || '';

                messageDiv.appendChild(labelDiv);
                messageDiv.appendChild(contentDiv);

                return { messageDiv, contentDiv };
            }

            function addUserMessage(content) {
                if (emptyState && emptyState.parentNode) {
                    emptyState.parentNode.removeChild(emptyState);
                }

                const { messageDiv } = createMessageElement('user', content);
                messagesContainer.appendChild(messageDiv);
                scrollToBottom();
            }

            function startAssistantMessage() {
                if (emptyState && emptyState.parentNode) {
                    emptyState.parentNode.removeChild(emptyState);
                }

                const { messageDiv, contentDiv } = createMessageElement('assistant', '', 'streaming');
                
                contentDiv.setAttribute('aria-busy', 'true');

                const statusDiv = document.createElement('div');
                statusDiv.className = 'status-indicator status-streaming';
                statusDiv.textContent = 'Generating...';
                statusDiv.setAttribute('aria-hidden', 'true');

                messageDiv.appendChild(statusDiv);
                messagesContainer.appendChild(messageDiv);

                currentAssistantMessageEl = messageDiv;
                currentContentEl = contentDiv;
                currentStatusEl = statusDiv;
                streamedContent = '';

                scrollToBottom();
            }

            function appendToAssistantMessage(text) {
                if (currentContentEl) {
                    streamedContent += text;
                    currentContentEl.textContent = streamedContent;
                    scrollToBottom();
                }
            }

            function finalizeAssistantMessage(status, statusText) {
                if (currentAssistantMessageEl) {
                    currentAssistantMessageEl.classList.remove('message-streaming');
                    if (status !== 'complete') {
                        currentAssistantMessageEl.classList.add(`message-${status}`);
                    }
                }

                if (currentContentEl) {
                    currentContentEl.removeAttribute('aria-busy');
                }

                if (currentStatusEl) {
                    currentStatusEl.className = `status-indicator status-${status}`;
                    currentStatusEl.textContent = statusText;
                }

                currentAssistantMessageEl = null;
                currentContentEl = null;
                currentStatusEl = null;
            }

            function removeLastAssistantMessage() {
                const messages = messagesContainer.querySelectorAll('.message-assistant');
                if (messages.length > 0) {
                    const lastMessage = messages[messages.length - 1];
                    lastMessage.parentNode.removeChild(lastMessage);
                }
            }

            function sendMessage(userContent) {
                if (isStreaming || !userContent.trim()) return;

                focusedElementBeforeStream = document.activeElement;

                conversationHistory.push({ role: 'user', content: userContent.trim() });
                addUserMessage(userContent.trim());

                promptInput.value = '';
                updateButtonStates();

                startStream();
            }

            function startStream() {
                isStreaming = true;
                const streamId = Date.now();
                currentStreamId = streamId;
                updateButtonStates();

                startAssistantMessage();
                announceStatus('Assistant is generating a response');

                const messagesToSend = conversationHistory.map(m => ({
                    role: m.role,
                    content: m.content
                }));

                currentStreamHandle = window.ChatSDK.streamChat(messagesToSend, {
                    onToken: function(text) {
                        if (currentStreamId !== streamId) return;
                        appendToAssistantMessage(text);
                    },
                    onDone: function() {
                        if (currentStreamId !== streamId) return;

                        conversationHistory.push({ role: 'assistant', content: streamedContent });
                        finalizeAssistantMessage('complete', 'Complete');
                        announceStatus('Response complete');

                        isStreaming = false;
                        currentStreamHandle = null;
                        currentStreamId = null;
                        updateButtonStates();

                        restoreFocus();
                    },
                    onError: function(error) {
                        if (currentStreamId !== streamId) return;

                        const errorMessage = error && error.message ? error.message : 'An error occurred';
                        
                        if (streamedContent) {
                            conversationHistory.push({ role: 'assistant', content: streamedContent });
                        }
                        
                        finalizeAssistantMessage('error', 'Error: ' + errorMessage);
                        announceStatus('Error: ' + errorMessage);

                        isStreaming = false;
                        currentStreamHandle = null;
                        currentStreamId = null;
                        updateButtonStates();

                        restoreFocus();
                    }
                });
            }

            function stopStream() {
                if (!isStreaming || !currentStreamHandle) return;

                const stoppedStreamId = currentStreamId;
                currentStreamId = null;

                currentStreamHandle.abort();
                currentStreamHandle = null;

                if (streamedContent) {
                    conversationHistory.push({ role: 'assistant', content: streamedContent });
                }

                finalizeAssistantMessage('stopped', 'Stopped');
                announceStatus('Generation stopped');

                isStreaming = false;
                updateButtonStates();

                restoreFocus();
            }

            function regenerate() {
                if (isStreaming) return;

                const lastAssistantIndex = conversationHistory.map(m => m.role).lastIndexOf('assistant');
                if (lastAssistantIndex === -1) return;

                focusedElementBeforeStream = document.activeElement;

                conversationHistory = conversationHistory.slice(0, lastAssistantIndex);
                removeLastAssistantMessage();

                startStream();
            }

            function restoreFocus() {
                if (focusedElementBeforeStream && document.body.contains(focusedElementBeforeStream)) {
                    focusedElementBeforeStream.focus();
                } else {
                    promptInput.focus();
                }
                focusedElementBeforeStream = null;
            }

            sendButton.addEventListener('click', function() {
                sendMessage(promptInput.value);
            });

            stopButton.addEventListener('click', function() {
                stopStream();
            });

            regenerateButton.addEventListener('click', function() {
                regenerate();
            });

            promptInput.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    if (!sendButton.disabled) {
                        sendMessage(promptInput.value);
                    }
                }
            });

            promptInput.addEventListener('input', function() {
                updateButtonStates();
                
                this.style.height = 'auto';
                this.style.height = Math.min(this.scrollHeight, 200) + 'px';
            });

            updateButtonStates();
            promptInput.focus();
        })();
    </script>
</body>
</html>
