    <script>
        (function() {
            'use strict';

            // State
            let messages = [];
            let currentHandle = null;
            let isGenerating = false;
            let currentAssistantMessageEl = null;
            let currentAssistantContent = '';

            // DOM Elements
            const promptInput = document.getElementById('prompt');
            const sendButton = document.getElementById('send');
            const stopButton = document.getElementById('stop');
            const regenerateButton = document.getElementById('regenerate');
            const conversationEl = document.getElementById('conversation');
            const conversationContainer = document.getElementById('conversation-container');
            const emptyState = document.getElementById('empty-state');
            const statusAnnouncer = document.getElementById('status-announcer');

            // Utility functions
            function escapeHtml(text) {
                const div = document.createElement('div');
                div.textContent = text;
                return div.innerHTML;
            }

            function announce(message) {
                statusAnnouncer.textContent = '';
                // Force screen reader to re-announce by clearing and setting
                requestAnimationFrame(() => {
                    statusAnnouncer.textContent = message;
                });
            }

            function scrollToBottom() {
                conversationContainer.scrollTop = conversationContainer.scrollHeight;
            }

            function updateButtonStates() {
                const hasContent = promptInput.value.trim().length > 0;
                const hasUserMessage = messages.some(m => m.role === 'user');

                sendButton.disabled = isGenerating || !hasContent;
                stopButton.disabled = !isGenerating;
                regenerateButton.disabled = isGenerating || !hasUserMessage;
                promptInput.disabled = isGenerating;

                if (isGenerating) {
                    sendButton.setAttribute('aria-busy', 'true');
                } else {
                    sendButton.removeAttribute('aria-busy');
                }
            }

            function hideEmptyState() {
                if (emptyState) {
                    emptyState.style.display = 'none';
                }
            }

            function createMessageElement(role, content) {
                const messageDiv = document.createElement('div');
                messageDiv.className = `message message-${role}`;
                messageDiv.setAttribute('role', 'article');
                messageDiv.setAttribute('aria-label', `${role === 'user' ? 'You' : 'Assistant'} said`);

                const labelDiv = document.createElement('div');
                labelDiv.className = 'message-label';
                labelDiv.textContent = role === 'user' ? 'You' : 'Assistant';

                const contentDiv = document.createElement('div');
                contentDiv.className = 'message-content';
                contentDiv.textContent = content;

                messageDiv.appendChild(labelDiv);
                messageDiv.appendChild(contentDiv);

                return messageDiv;
            }

            function createStatusIndicator(type, message) {
                const statusDiv = document.createElement('div');
                statusDiv.className = `status-indicator status-${type}`;
                statusDiv.id = 'current-status';
                statusDiv.setAttribute('role', 'status');

                if (type === 'generating') {
                    const spinner = document.createElement('span');
                    spinner.className = 'spinner';
                    spinner.setAttribute('aria-hidden', 'true');
                    statusDiv.appendChild(spinner);
                }

                const text = document.createElement('span');
                text.textContent = message;
                statusDiv.appendChild(text);

                return statusDiv;
            }

            function removeStatusIndicator() {
                const existing = document.getElementById('current-status');
                if (existing) {
                    existing.remove();
                }
            }

            function showStatus(type, message) {
                removeStatusIndicator();
                const statusEl = createStatusIndicator(type, message);
                conversationEl.appendChild(statusEl);
                scrollToBottom();
            }

            function addUserMessage(content) {
                hideEmptyState();
                const messageEl = createMessageElement('user', content);
                conversationEl.appendChild(messageEl);
                messages.push({ role: 'user', content: content });
                scrollToBottom();
            }

            function startAssistantMessage() {
                hideEmptyState();
                currentAssistantContent = '';
                currentAssistantMessageEl = createMessageElement('assistant', '');
                conversationEl.appendChild(currentAssistantMessageEl);
                scrollToBottom();
            }

            function appendToAssistantMessage(token) {
                currentAssistantContent += token;
                if (currentAssistantMessageEl) {
                    const contentEl = currentAssistantMessageEl.querySelector('.message-content');
                    if (contentEl) {
                        contentEl.textContent = currentAssistantContent;
                    }
                }
                scrollToBottom();
            }

            function finalizeAssistantMessage() {
                if (currentAssistantContent) {
                    messages.push({ role: 'assistant', content: currentAssistantContent });
                }
                currentAssistantMessageEl = null;
                currentAssistantContent = '';
            }

            function removeLastAssistantMessage() {
                // Remove from DOM
                const assistantMessages = conversationEl.querySelectorAll('.message-assistant');
                if (assistantMessages.length > 0) {
                    assistantMessages[assistantMessages.length - 1].remove();
                }
                // Remove from state
                if (messages.length > 0 && messages[messages.length - 1].role === 'assistant') {
                    messages.pop();
                }
            }

            // API interaction
            function streamMessage() {
                if (isGenerating) return;

                isGenerating = true;
                updateButtonStates();
                
                startAssistantMessage();
                showStatus('generating', 'Generating response...');
                announce('Generating response');

                const messagesToSend = [...messages];

                currentHandle = window.ChatSDK.streamChat(messagesToSend, {
                    onToken: function(text) {
                        removeStatusIndicator();
                        appendToAssistantMessage(text);
                    },
                    onDone: function() {
                        isGenerating = false;
                        currentHandle = null;
                        finalizeAssistantMessage();
                        removeStatusIndicator();
                        showStatus('done', 'Response complete');
                        announce('Response complete');
                        updateButtonStates();
                        promptInput.focus();
                        
                        // Remove done status after a delay
                        setTimeout(() => {
                            const status = document.getElementById('current-status');
                            if (status && status.classList.contains('status-done')) {
                                status.remove();
                            }
                        }, 3000);
                    },
                    onError: function(error) {
                        isGenerating = false;
                        currentHandle = null;
                        finalizeAssistantMessage();
                        removeStatusIndicator();
                        const errorMessage = error && error.message ? error.message : 'An error occurred';
                        showStatus('error', 'Error: ' + errorMessage);
                        announce('Error: ' + errorMessage);
                        updateButtonStates();
                        promptInput.focus();
                    }
                });
            }

            function sendMessage() {
                const content = promptInput.value.trim();
                if (!content || isGenerating) return;

                promptInput.value = '';
                addUserMessage(content);
                streamMessage();
            }

            function stopGeneration() {
                if (!isGenerating || !currentHandle) return;

                currentHandle.abort();
                isGenerating = false;
                currentHandle = null;
                
                finalizeAssistantMessage();
                removeStatusIndicator();
                showStatus('stopped', 'Generation stopped');
                announce('Generation stopped');
                updateButtonStates();
                promptInput.focus();
            }

            function regenerateResponse() {
                if (isGenerating) return;
                if (!messages.some(m => m.role === 'user')) return;

                // Remove the last assistant message if it exists
                if (messages.length > 0 && messages[messages.length - 1].role === 'assistant') {
                    removeLastAssistantMessage();
                }

                // Also remove any status indicators
                removeStatusIndicator();

                // Stream a new response
                streamMessage();
                announce('Regenerating response');
            }

            // Event listeners
            sendButton.addEventListener('click', sendMessage);

            stopButton.addEventListener('click', stopGeneration);

            regenerateButton.addEventListener('click', regenerateResponse);

            promptInput.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    if (!sendButton.disabled) {
                        sendMessage();
                    }
                }
            });

            promptInput.addEventListener('input', function() {
                updateButtonStates();
                // Auto-resize textarea
                this.style.height = 'auto';
                this.style.height = Math.min(this.scrollHeight, 200) + 'px';
            });

            // Initialize
            updateButtonStates();
            promptInput.focus();
        })();
    </script>
</body>
