    <script>
        (function() {
            'use strict';

            // State
            let conversationHistory = [];
            let isStreaming = false;
            let currentStreamHandle = null;
            let currentStreamId = 0;
            let currentAssistantMessageEl = null;
            let currentAssistantContent = '';

            // DOM elements
            const chatMessages = document.getElementById('chat-messages');
            const promptInput = document.getElementById('prompt');
            const sendButton = document.getElementById('send');
            const stopButton = document.getElementById('stop');
            const regenerateButton = document.getElementById('regenerate');
            const statusIndicator = document.getElementById('status-indicator');
            const srAnnouncer = document.getElementById('sr-announcer');
            const emptyState = document.getElementById('empty-state');

            // Announce to screen readers without disrupting focus
            function announce(message) {
                srAnnouncer.textContent = '';
                // Force reflow to ensure announcement
                void srAnnouncer.offsetWidth;
                srAnnouncer.textContent = message;
            }

            // Show visual status
            function showStatus(message, type = 'info') {
                statusIndicator.textContent = message;
                statusIndicator.className = 'status-indicator visible';
                if (type === 'error') {
                    statusIndicator.classList.add('error');
                } else if (type === 'success') {
                    statusIndicator.classList.add('success');
                }
            }

            // Hide visual status
            function hideStatus() {
                statusIndicator.classList.remove('visible');
            }

            // Update button states based on streaming status
            function updateButtonStates() {
                const hasHistory = conversationHistory.length > 0;
                const hasUserMessage = conversationHistory.some(m => m.role === 'user');
                const promptHasContent = promptInput.value.trim().length > 0;

                sendButton.disabled = isStreaming || !promptHasContent;
                sendButton.setAttribute('aria-disabled', sendButton.disabled);

                stopButton.disabled = !isStreaming;
                stopButton.setAttribute('aria-disabled', stopButton.disabled);

                regenerateButton.disabled = isStreaming || !hasUserMessage;
                regenerateButton.setAttribute('aria-disabled', regenerateButton.disabled);
            }

            // Hide empty state
            function hideEmptyState() {
                if (emptyState) {
                    emptyState.style.display = 'none';
                }
            }

            // Escape HTML to prevent XSS
            function escapeHtml(text) {
                const div = document.createElement('div');
                div.textContent = text;
                return div.innerHTML;
            }

            // Create a message element
            function createMessageElement(role, content) {
                const messageEl = document.createElement('article');
                messageEl.className = `message message-${role}`;
                messageEl.setAttribute('aria-label', `${role === 'user' ? 'You' : 'Assistant'}`);

                const roleEl = document.createElement('div');
                roleEl.className = 'message-role';
                roleEl.textContent = role === 'user' ? 'You' : 'Assistant';

                const contentEl = document.createElement('div');
                contentEl.className = 'message-content';
                contentEl.textContent = content;

                messageEl.appendChild(roleEl);
                messageEl.appendChild(contentEl);

                return messageEl;
            }

            // Add user message to the chat
            function addUserMessage(content) {
                hideEmptyState();
                const messageEl = createMessageElement('user', content);
                chatMessages.appendChild(messageEl);
                conversationHistory.push({ role: 'user', content: content });
                scrollToBottom();
            }

            // Create assistant message placeholder for streaming
            function createAssistantMessagePlaceholder() {
                hideEmptyState();
                const messageEl = createMessageElement('assistant', '');
                messageEl.classList.add('message-streaming');
                messageEl.setAttribute('aria-busy', 'true');
                chatMessages.appendChild(messageEl);
                return messageEl;
            }

            // Update assistant message content
            function updateAssistantMessage(messageEl, content) {
                const contentEl = messageEl.querySelector('.message-content');
                contentEl.textContent = content;
            }

            // Finalize assistant message (remove streaming indicators)
            function finalizeAssistantMessage(messageEl) {
                messageEl.classList.remove('message-streaming');
                messageEl.removeAttribute('aria-busy');
            }

            // Scroll chat to bottom
            function scrollToBottom() {
                chatMessages.scrollTop = chatMessages.scrollHeight;
            }

            // Start streaming a response
            function startStream(messages) {
                if (isStreaming) return;

                isStreaming = true;
                currentStreamId++;
                const thisStreamId = currentStreamId;
                currentAssistantContent = '';
                currentAssistantMessageEl = createAssistantMessagePlaceholder();

                showStatus('Generating response...');
                announce('Generating response');
                updateButtonStates();
                scrollToBottom();

                currentStreamHandle = window.ChatSDK.streamChat(messages, {
                    onToken: function(text) {
                        // Ignore tokens from aborted streams
                        if (thisStreamId !== currentStreamId) return;
                        if (!isStreaming) return;

                        currentAssistantContent += text;
                        updateAssistantMessage(currentAssistantMessageEl, currentAssistantContent);
                        scrollToBottom();
                    },

                    onDone: function() {
                        // Ignore completion from aborted streams
                        if (thisStreamId !== currentStreamId) return;

                        isStreaming = false;
                        currentStreamHandle = null;

                        if (currentAssistantMessageEl) {
                            finalizeAssistantMessage(currentAssistantMessageEl);
                        }

                        // Add to history only if we have content
                        if (currentAssistantContent.length > 0) {
                            conversationHistory.push({ 
                                role: 'assistant', 
                                content: currentAssistantContent 
                            });
                        }

                        showStatus('Response complete', 'success');
                        announce('Response complete');
                        updateButtonStates();

                        // Clear status after a delay
                        setTimeout(hideStatus, 2000);

                        currentAssistantMessageEl = null;
                        currentAssistantContent = '';
                    },

                    onError: function(error) {
                        // Ignore errors from aborted streams
                        if (thisStreamId !== currentStreamId) return;

                        isStreaming = false;
                        currentStreamHandle = null;

                        if (currentAssistantMessageEl) {
                            finalizeAssistantMessage(currentAssistantMessageEl);
                            // Keep partial content if any
                            if (currentAssistantContent.length > 0) {
                                conversationHistory.push({ 
                                    role: 'assistant', 
                                    content: currentAssistantContent 
                                });
                            } else {
                                // Remove empty message
                                currentAssistantMessageEl.remove();
                            }
                        }

                        const errorMessage = error && error.message ? error.message : 'An error occurred';
                        showStatus('Error: ' + errorMessage, 'error');
                        announce('Error: ' + errorMessage);
                        updateButtonStates();

                        currentAssistantMessageEl = null;
                        currentAssistantContent = '';
                    }
                });
            }

            // Stop current stream
            function stopStream() {
                if (!isStreaming || !currentStreamHandle) return;

                // Increment stream ID to ignore any late tokens
                currentStreamId++;
                
                currentStreamHandle.abort();
                currentStreamHandle = null;
                isStreaming = false;

                if (currentAssistantMessageEl) {
                    finalizeAssistantMessage(currentAssistantMessageEl);
                    // Keep partial content in history
                    if (currentAssistantContent.length > 0) {
                        conversationHistory.push({ 
                            role: 'assistant', 
                            content: currentAssistantContent 
                        });
                    } else {
                        // Remove empty message element
                        currentAssistantMessageEl.remove();
                    }
                }

                showStatus('Generation stopped', 'info');
                announce('Generation stopped');
                updateButtonStates();

                setTimeout(hideStatus, 2000);

                currentAssistantMessageEl = null;
                currentAssistantContent = '';
            }

            // Send message
            function sendMessage() {
                const content = promptInput.value.trim();
                if (!content || isStreaming) return;

                addUserMessage(content);
                promptInput.value = '';
                updateButtonStates();

                // Start streaming with full history
                startStream([...conversationHistory]);
            }

            // Regenerate last assistant response
            function regenerate() {
                if (isStreaming) return;

                // Find last user message index
                let lastUserIndex = -1;
                for (let i = conversationHistory.length - 1; i >= 0; i--) {
                    if (conversationHistory[i].role === 'user') {
                        lastUserIndex = i;
                        break;
                    }
                }

                if (lastUserIndex === -1) return;

                // Remove assistant messages after the last user message from history
                conversationHistory = conversationHistory.slice(0, lastUserIndex + 1);

                // Remove assistant message elements from DOM
                const messages = chatMessages.querySelectorAll('.message-assistant');
                const userMessages = chatMessages.querySelectorAll('.message-user');
                
                // Count how many assistant messages should remain
                let assistantCount = 0;
                for (let i = 0; i < conversationHistory.length; i++) {
                    if (conversationHistory[i].role === 'assistant') {
                        assistantCount++;
                    }
                }

                // Remove excess assistant messages from the end
                for (let i = messages.length - 1; i >= assistantCount; i--) {
                    messages[i].remove();
                }

                announce('Regenerating response');
                
                // Start new stream
                startStream([...conversationHistory]);
            }

            // Auto-resize textarea
            function autoResizeTextarea() {
                promptInput.style.height = 'auto';
                promptInput.style.height = Math.min(promptInput.scrollHeight, 128) + 'px';
            }

            // Event listeners
            sendButton.addEventListener('click', sendMessage);

            stopButton.addEventListener('click', stopStream);

            regenerateButton.addEventListener('click', regenerate);

            promptInput.addEventListener('input', function() {
                updateButtonStates();
                autoResizeTextarea();
            });

            promptInput.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    if (!sendButton.disabled) {
                        sendMessage();
                    }
                }
            });

            // Initialize
            updateButtonStates();
            promptInput.focus();
        })();
    </script>
</body>
</html>
