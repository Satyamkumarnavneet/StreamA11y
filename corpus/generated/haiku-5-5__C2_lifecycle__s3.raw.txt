<script>
(function () {
  'use strict';

  var region = document.getElementById('transcript-region');
  var transcript = document.getElementById('transcript');
  var emptyNote = document.getElementById('empty');
  var prompt = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var statusEl = document.getElementById('status');
  var dot = document.getElementById('dot');
  var form = document.getElementById('composer');

  // Conversation state. Each turn: {role, content, el, bodyEl, noteEl, textNode}
  var turns = [];
  // The generation currently streaming, or null.
  var current = null;

  function sdkAvailable() {
    return !!(window.ChatSDK && typeof window.ChatSDK.streamChat === 'function');
  }

  function announce(message) {
    statusEl.textContent = message;
  }

  function setAria(btn, disabled) {
    btn.setAttribute('aria-disabled', disabled ? 'true' : 'false');
  }

  function isGenerating() {
    return current !== null;
  }

  function hasUserTurn() {
    return turns.some(function (t) { return t.role === 'user'; });
  }

  // Buttons use aria-disabled instead of the disabled attribute so that they
  // stay focusable and keyboard focus is never dropped while state changes.
  function refreshControls() {
    var gen = isGenerating();
    var available = sdkAvailable();
    setAria(sendBtn, gen || !available);
    setAria(stopBtn, !gen);
    setAria(regenBtn, gen || !available || !hasUserTurn());
    dot.classList.toggle('on', gen);
  }

  function atBottom() {
    return region.scrollHeight - region.scrollTop - region.clientHeight < 40;
  }

  function scrollToEnd(force) {
    if (force || atBottom()) {
      region.scrollTop = region.scrollHeight;
    }
  }

  function setNote(turn, text, isError) {
    turn.noteEl.textContent = text || '';
    turn.noteEl.hidden = !text;
    turn.noteEl.classList.toggle('error', !!isError);
  }

  function addTurn(role, content) {
    var li = document.createElement('li');
    li.className = 'msg ' + role;

    var who = document.createElement('p');
    who.className = 'who';
    who.textContent = role === 'user' ? 'You' : 'Assistant';

    var body = document.createElement('div');
    body.className = 'body';

    var note = document.createElement('p');
    note.className = 'note';
    note.hidden = true;

    li.appendChild(who);
    li.appendChild(body);
    li.appendChild(note);
    transcript.appendChild(li);
    emptyNote.hidden = true;

    var turn = { role: role, content: '', el: li, bodyEl: body, noteEl: note, textNode: null };
    if (role === 'user') {
      turn.content = content;
      body.textContent = content;
    }
    turns.push(turn);
    refreshControls();
    return turn;
  }

  function finish(gen, outcome, err) {
    var turn = gen.turn;
    turn.el.removeAttribute('aria-busy');
    if (current === gen) {
      current = null;
    }
    var empty = turn.content === '';
    var message;

    if (outcome === 'done') {
      message = 'Answer complete.';
    } else if (outcome === 'stopped') {
      setNote(turn, empty ? 'Stopped. No text had arrived yet.' : 'Stopped. Partial answer kept.', false);
      message = empty ? 'Answer stopped.' : 'Answer stopped. Partial answer kept.';
    } else {
      var detail = (err && err.message) ? String(err.message) : 'Unknown error.';
      setNote(turn, 'Error: the answer could not be completed. ' + detail, true);
      message = 'Error: the answer could not be completed. ' + detail;
    }

    refreshControls();
    announce(message);
    scrollToEnd(false);
  }

  function startGeneration(label) {
    if (!sdkAvailable()) {
      announce('The chat service is not available.');
      refreshControls();
      return;
    }
    if (isGenerating()) {
      announce('An answer is already being generated. Press Stop to end it.');
      return;
    }

    // Build the request from completed turns (partial answers included) before
    // adding the placeholder for the new answer.
    var messages = turns.map(function (t) {
      return { role: t.role, content: t.content };
    });

    var turn = addTurn('assistant', '');
    turn.el.setAttribute('aria-busy', 'true');
    turn.textNode = document.createTextNode('');
    turn.bodyEl.appendChild(turn.textNode);

    var gen = { turn: turn, closed: false, handle: null };
    current = gen;
    refreshControls();
    announce(label || 'Generating answer.');
    scrollToEnd(true);

    try {
      var handle = window.ChatSDK.streamChat(messages, {
        onToken: function (text) {
          if (gen.closed) return;
          var chunk = (text === null || text === undefined) ? '' : String(text);
          if (!chunk) return;
          var stick = atBottom();
          turn.textNode.appendData(chunk);
          turn.content += chunk;
          if (stick) scrollToEnd(false);
        },
        onDone: function () {
          if (gen.closed) return;
          gen.closed = true;
          finish(gen, 'done');
        },
        onError: function (error) {
          if (gen.closed) return;
          gen.closed = true;
          finish(gen, 'error', error);
        }
      });
      gen.handle = handle;
    } catch (error) {
      if (!gen.closed) {
        gen.closed = true;
        finish(gen, 'error', error);
      }
    }
  }

  function stopGeneration() {
    var gen = current;
    if (!gen || gen.closed) {
      announce('No answer is being generated.');
      return;
    }
    // Mark closed first so any chunk that arrives after this point is ignored.
    gen.closed = true;
    try {
      if (gen.handle && typeof gen.handle.abort === 'function') {
        gen.handle.abort();
      }
    } catch (e) {
      // Ignore abort errors; the answer is stopped either way.
    }
    finish(gen, 'stopped');
  }

  function submit() {
    if (!sdkAvailable()) {
      announce('The chat service is not available.');
      return;
    }
    if (isGenerating()) {
      announce('Wait for the answer to finish, or press Stop.');
      return;
    }
    var text = prompt.value.trim();
    if (!text) {
      announce('Type a message before sending.');
      return;
    }
    addTurn('user', text);
    prompt.value = '';
    startGeneration('Generating answer.');
  }

  function regenerate() {
    if (!sdkAvailable()) {
      announce('The chat service is not available.');
      return;
    }
    if (isGenerating()) {
      announce('Wait for the answer to finish, or press Stop.');
      return;
    }
    if (!hasUserTurn()) {
      announce('There is no message to regenerate.');
      return;
    }
    var last = turns[turns.length - 1];
    if (last && last.role === 'assistant') {
      turns.pop();
      last.el.remove();
    }
    last = turns[turns.length - 1];
    if (!last || last.role !== 'user') {
      announce('There is no message to regenerate.');
      refreshControls();
      return;
    }
    startGeneration('Regenerating answer.');
  }

  sendBtn.addEventListener('click', submit);
  stopBtn.addEventListener('click', function () {
    if (sdkAvailable() && isGenerating()) {
      stopGeneration();
    } else {
      announce('No answer is being generated.');
    }
  });
  regenBtn.addEventListener('click', regenerate);

  prompt.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      submit();
    }
  });

  form.addEventListener('submit', function (e) {
    e.preventDefault();
  });

  refreshControls();
  if (!sdkAvailable()) {
    announce('The chat service is not available.');
  }
})();
</script>
</body>
</html>
