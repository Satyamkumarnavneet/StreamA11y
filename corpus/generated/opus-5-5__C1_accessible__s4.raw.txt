<script>
(function () {
  'use strict';

  var form = document.getElementById('chat-form');
  var promptEl = document.getElementById('prompt');
  var promptError = document.getElementById('prompt-error');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var conversationEl = document.getElementById('conversation');
  var listEl = document.getElementById('messages');
  var emptyEl = document.getElementById('empty');
  var statusBar = document.getElementById('status-bar');
  var statusEl = document.getElementById('status');
  var answerAnnouncer = document.getElementById('answer-announcer');
  var errorAnnouncer = document.getElementById('error-announcer');

  // Conversation state: {id, role, content, status: 'complete'|'streaming'|'stopped'|'error', error}
  var history = [];
  var nextId = 1;
  var active = null; // current request {id, msg, handle, finished}
  var nextReqId = 1;

  var STATE_LABELS = {
    streaming: 'Generating…',
    stopped: 'Stopped',
    error: 'Error',
    complete: 'Finished'
  };

  function isGenerating() { return active !== null; }

  function setDisabled(btn, disabled) {
    btn.setAttribute('aria-disabled', disabled ? 'true' : 'false');
  }
  function isDisabled(btn) { return btn.getAttribute('aria-disabled') === 'true'; }

  function hasUserMessage() {
    for (var i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') return true;
    }
    return false;
  }

  function updateControls() {
    var gen = isGenerating();
    setDisabled(sendBtn, gen);
    setDisabled(stopBtn, !gen);
    setDisabled(regenBtn, !hasUserMessage());
    emptyEl.hidden = history.length > 0;
  }

  function setStatus(state, text) {
    statusBar.setAttribute('data-state', state);
    statusEl.textContent = text;
  }

  function announce(region, text) {
    region.textContent = '';
    window.setTimeout(function () { region.textContent = text; }, 60);
  }

  // ---------- Rendering ----------
  var nodes = {}; // id -> {li, content, state, error}

  function isNearBottom() {
    return conversationEl.scrollHeight - conversationEl.scrollTop - conversationEl.clientHeight < 48;
  }
  function scrollToBottom() { conversationEl.scrollTop = conversationEl.scrollHeight; }

  function renderMessage(msg) {
    var li = document.createElement('li');
    li.className = 'msg ' + msg.role;

    var header = document.createElement('h3');
    header.className = 'msg-header';
    var who = document.createElement('span');
    who.textContent = msg.role === 'user' ? 'You' : 'Assistant';
    header.appendChild(who);

    var state = document.createElement('span');
    state.className = 'msg-state';
    if (msg.role === 'assistant') header.appendChild(state);
    li.appendChild(header);

    var content = document.createElement('p');
    content.className = 'msg-content';
    li.appendChild(content);

    var err = document.createElement('p');
    err.className = 'msg-error';
    err.hidden = true;
    li.appendChild(err);

    listEl.appendChild(li);
    nodes[msg.id] = { li: li, content: content, state: state, error: err };
    updateMessage(msg);
  }

  function updateMessage(msg) {
    var n = nodes[msg.id];
    if (!n) return;
    n.content.textContent = msg.content;
    if (msg.role === 'assistant') {
      n.state.className = 'msg-state ' + msg.status;
      var label = STATE_LABELS[msg.status] || '';
      n.state.textContent = label ? '(' + label + ')' : '';
      if (msg.status === 'streaming') n.li.setAttribute('aria-busy', 'true');
      else n.li.removeAttribute('aria-busy');
      if (msg.status === 'streaming' && !msg.content) n.content.textContent = '…';
      if (msg.status === 'error') {
        n.error.hidden = false;
        n.error.textContent = 'Error: ' + (msg.error || 'Something went wrong.');
      } else {
        n.error.hidden = true;
        n.error.textContent = '';
      }
    }
  }

  function removeMessage(msg) {
    var n = nodes[msg.id];
    if (n && n.li.parentNode) n.li.parentNode.removeChild(n.li);
    delete nodes[msg.id];
  }

  function addMessage(role, content, status) {
    var msg = { id: nextId++, role: role, content: content, status: status || 'complete', error: '' };
    history.push(msg);
    var stick = isNearBottom();
    renderMessage(msg);
    if (stick || role === 'user') scrollToBottom();
    return msg;
  }

  // ---------- Streaming ----------
  function apiMessages() {
    var out = [];
    for (var i = 0; i < history.length; i++) {
      var m = history[i];
      if (m.status === 'streaming') continue;
      if (m.role === 'assistant' && !m.content) continue;
      out.push({ role: m.role, content: m.content });
    }
    return out;
  }

  function errorText(error) {
    if (!error) return 'Unknown error.';
    if (typeof error === 'string') return error;
    if (error.message) return String(error.message);
    try { return String(error); } catch (e) { return 'Unknown error.'; }
  }

  function startStream() {
    var payload = apiMessages();
    var msg = addMessage('assistant', '', 'streaming');
    var req = { id: nextReqId++, msg: msg, handle: null, finished: false };
    active = req;
    setStatus('generating', 'Assistant is generating an answer…');
    updateControls();

    function isCurrent() { return active === req && !req.finished; }

    function finish(status, err) {
      req.finished = true;
      if (active === req) active = null;
      msg.status = status;
      msg.error = err || '';
      updateMessage(msg);
      updateControls();
    }

    var callbacks = {
      onToken: function (text) {
        // Ignores late tokens that arrive after abort or completion.
        if (!isCurrent()) return;
        if (text == null) return;
        var stick = isNearBottom();
        msg.content += String(text);
        updateMessage(msg);
        if (stick) scrollToBottom();
      },
      onDone: function () {
        if (!isCurrent()) return;
        finish('complete');
        setStatus('finished', 'Assistant finished answering.');
        announce(answerAnnouncer, msg.content
          ? 'Assistant answered: ' + msg.content
          : 'Assistant finished with an empty answer.');
      },
      onError: function (error) {
        if (!isCurrent()) return;
        var t = errorText(error);
        finish('error', t);
        setStatus('error', 'Error: ' + t + (msg.content ? ' Partial answer kept.' : '') + ' You can try Regenerate.');
        announce(errorAnnouncer, 'Error while generating answer: ' + t);
      }
    };

    try {
      if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
        throw new Error('Chat service is not available.');
      }
      var handle = window.ChatSDK.streamChat(payload, callbacks);
      req.handle = handle;
      // If a callback already finished the request synchronously, nothing else to do.
    } catch (e) {
      callbacks.onError(e);
    }
  }

  function stopStream(options) {
    var req = active;
    if (!req) return false;
    req.finished = true; // set before abort so any late token is ignored
    active = null;
    try {
      if (req.handle && typeof req.handle.abort === 'function') req.handle.abort();
    } catch (e) { /* ignore abort failures; UI is already stopped */ }
    req.msg.status = 'stopped';
    updateMessage(req.msg);
    updateControls();
    if (!(options && options.silent)) {
      setStatus('stopped', req.msg.content
        ? 'Assistant stopped. Partial answer kept.'
        : 'Assistant stopped before answering.');
    }
    return true;
  }

  // ---------- Actions ----------
  function showPromptError(text) {
    promptError.textContent = text;
    promptError.hidden = false;
    promptEl.setAttribute('aria-invalid', 'true');
  }
  function clearPromptError() {
    promptError.textContent = '';
    promptError.hidden = true;
    promptEl.removeAttribute('aria-invalid');
  }

  function send() {
    if (isGenerating()) {
      setStatus('generating', 'Assistant is still generating. Stop it before sending a new message.');
      return;
    }
    var text = promptEl.value;
    if (!text.trim()) {
      showPromptError('Please type a message before sending.');
      promptEl.focus();
      return;
    }
    clearPromptError();
    addMessage('user', text, 'complete');
    promptEl.value = '';
    startStream();
  }

  function regenerate() {
    if (!hasUserMessage()) return;
    stopStream({ silent: true });
    // Remove assistant answers that follow the last user message.
    while (history.length && history[history.length - 1].role === 'assistant') {
      removeMessage(history.pop());
    }
    announce(answerAnnouncer, 'Regenerating the answer to your last message.');
    startStream();
  }

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    send();
  });

  stopBtn.addEventListener('click', function () {
    if (isDisabled(stopBtn)) return;
    stopStream();
  });

  regenBtn.addEventListener('click', function () {
    if (isDisabled(regenBtn)) return;
    regenerate();
  });

  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing && e.keyCode !== 229) {
      e.preventDefault();
      send();
    }
  });

  promptEl.addEventListener('input', function () {
    if (!promptError.hidden && promptEl.value.trim()) clearPromptError();
  });

  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && isGenerating()) {
      e.preventDefault();
      stopStream();
    }
  });

  updateControls();
})();
</script>
</body>
</html>
