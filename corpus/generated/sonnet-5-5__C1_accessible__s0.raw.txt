<script>
(function () {
  'use strict';

  var $ = function (id) { return document.getElementById(id); };
  var promptEl = $('prompt'), sendBtn = $('send'), stopBtn = $('stop'), regenBtn = $('regenerate');
  var conv = $('conversation'), listEl = $('messages'), emptyEl = $('empty');
  var statusEl = $('status'), statusbar = $('statusbar'), indicator = $('indicator'), errorEl = $('error');
  var form = $('composer');

  var history = [];   // { role, content, status, els }
  var active = null;  // current run: { msg, handle }

  function setDisabled(btn, disabled) {
    if (disabled) btn.setAttribute('aria-disabled', 'true');
    else btn.removeAttribute('aria-disabled');
  }
  function isDisabled(btn) { return btn.getAttribute('aria-disabled') === 'true'; }

  function lastUserIndex() {
    for (var i = history.length - 1; i >= 0; i--) if (history[i].role === 'user') return i;
    return -1;
  }

  function updateButtons() {
    var busy = !!active;
    setDisabled(sendBtn, busy);
    setDisabled(stopBtn, !busy);
    setDisabled(regenBtn, busy || lastUserIndex() < 0);
  }

  function setStatus(kind, text) {
    statusbar.classList.toggle('hidden', !text);
    indicator.textContent = '';
    if (kind === 'generating') {
      var sp = document.createElement('span');
      sp.className = 'spinner';
      indicator.appendChild(sp);
    } else if (kind === 'done') indicator.textContent = '\u2713';
    else if (kind === 'stopped') indicator.textContent = '\u25A0';
    statusEl.textContent = text;
  }

  function nearBottom() {
    return conv.scrollHeight - conv.scrollTop - conv.clientHeight < 80;
  }
  function scrollToBottom() { conv.scrollTop = conv.scrollHeight; }

  var BADGES = {
    streaming: 'Generating\u2026',
    done: 'Finished',
    stopped: 'Stopped (partial answer)',
    error: 'Error'
  };

  function renderBody(m) {
    var b = m.els.body;
    if (m.content) {
      b.classList.remove('placeholder');
      b.textContent = m.content;
    } else if (m.role === 'assistant') {
      b.classList.add('placeholder');
      b.textContent = m.status === 'streaming' ? 'Waiting for the first words\u2026' : 'No text was received.';
    } else {
      b.textContent = '';
    }
  }

  function renderBadge(m) {
    if (m.role !== 'assistant') return;
    m.els.badge.textContent = '(' + BADGES[m.status] + ')';
    m.els.badge.classList.toggle('error', m.status === 'error');
    m.els.li.setAttribute('aria-busy', m.status === 'streaming' ? 'true' : 'false');
  }

  function addMessage(m) {
    var li = document.createElement('li');
    li.className = 'msg ' + m.role;
    var meta = document.createElement('div');
    meta.className = 'meta';
    var who = document.createElement('span');
    who.className = 'who';
    who.textContent = m.role === 'user' ? 'You:' : 'Assistant:';
    var badge = document.createElement('span');
    badge.className = 'badge';
    var body = document.createElement('div');
    body.className = 'body';
    meta.appendChild(who);
    meta.appendChild(badge);
    li.appendChild(meta);
    li.appendChild(body);
    m.els = { li: li, body: body, badge: badge };
    listEl.appendChild(li);
    emptyEl.hidden = true;
    renderBody(m);
    renderBadge(m);
  }

  function finish(run, status, err) {
    if (active !== run) return;
    active = null;
    run.msg.status = status;
    renderBody(run.msg);
    renderBadge(run.msg);
    updateButtons();
    if (status === 'done') {
      setStatus('done', 'Response finished.');
    } else if (status === 'stopped') {
      setStatus('stopped', 'Generation stopped. The partial answer was kept.');
    } else {
      setStatus('error', '');
      var detail = '';
      if (err) detail = typeof err === 'string' ? err : (err.message || '');
      errorEl.textContent = 'Error: the answer could not be completed' +
        (detail ? ' (' + detail + ')' : '') +
        '. Any text already received was kept. Use Regenerate to try again.';
    }
  }

  function startRun() {
    errorEl.textContent = '';
    var payload = history
      .filter(function (m) { return m.content && m.content.length > 0; })
      .map(function (m) { return { role: m.role, content: m.content }; });

    var msg = { role: 'assistant', content: '', status: 'streaming' };
    history.push(msg);
    addMessage(msg);
    var run = { msg: msg, handle: null };
    active = run;
    updateButtons();
    setStatus('generating', 'Assistant is generating a response\u2026');
    scrollToBottom();

    try {
      if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
        throw new Error('The chat service is not available');
      }
      run.handle = window.ChatSDK.streamChat(payload, {
        onToken: function (text) {
          if (active !== run) return;          // ignore late tokens after stop
          var stick = nearBottom();
          msg.content += String(text);
          renderBody(msg);
          if (stick) scrollToBottom();
        },
        onDone: function () { finish(run, 'done'); },
        onError: function (e) { finish(run, 'error', e); }
      });
    } catch (e) {
      finish(run, 'error', e);
    }
  }

  function send() {
    if (active) {
      setStatus('generating', 'Still generating. Press Stop first to cancel the current answer.');
      return;
    }
    var text = promptEl.value.trim();
    if (!text) {
      setStatus('idle', 'Type a message before sending.');
      return;
    }
    promptEl.value = '';
    var um = { role: 'user', content: text, status: 'done' };
    history.push(um);
    addMessage(um);
    startRun();
  }

  function stop() {
    var run = active;
    if (!run) return;
    try { if (run.handle && run.handle.abort) run.handle.abort(); } catch (e) { /* ignore */ }
    finish(run, 'stopped');
  }

  function regenerate() {
    if (active) return;
    var idx = lastUserIndex();
    if (idx < 0) return;
    for (var i = history.length - 1; i > idx; i--) {
      var m = history[i];
      if (m.els && m.els.li.parentNode) m.els.li.parentNode.removeChild(m.els.li);
      history.pop();
    }
    startRun();
  }

  form.addEventListener('submit', function (e) { e.preventDefault(); if (!isDisabled(sendBtn)) send(); else send(); });
  stopBtn.addEventListener('click', function () { if (!isDisabled(stopBtn)) stop(); });
  regenBtn.addEventListener('click', function () { if (!isDisabled(regenBtn)) regenerate(); });

  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.ctrlKey && !e.altKey && !e.metaKey && !e.isComposing) {
      e.preventDefault();
      send();
    } else if (e.key === 'Escape' && active) {
      e.preventDefault();
      stop();
    }
  });

  updateButtons();
  setStatus('idle', '');
})();
</script>
</body>
</html>
