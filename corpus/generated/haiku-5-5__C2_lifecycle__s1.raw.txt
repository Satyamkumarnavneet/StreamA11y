<script>
(function () {
  'use strict';

  var log = document.getElementById('log');
  var transcript = document.getElementById('transcript');
  var emptyNote = document.getElementById('empty');
  var status = document.getElementById('status');
  var form = document.getElementById('composer');
  var prompt = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');

  // Conversation model. Each entry: {role, content, el, body, note, indicator, error}
  var history = [];
  // The stream currently running, or null. Chunks are only applied when active === their session.
  var active = null;

  function setAriaDisabled(btn, flag) {
    btn.setAttribute('aria-disabled', flag ? 'true' : 'false');
  }

  function setStatus(text) {
    // Only the status line is announced; transcript text is never announced per chunk.
    status.textContent = text;
  }

  function lastUserIndex() {
    for (var i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') return i;
    }
    return -1;
  }

  function updateControls() {
    var busy = active !== null;
    setAriaDisabled(sendBtn, busy);
    setAriaDisabled(stopBtn, !busy);
    setAriaDisabled(regenBtn, busy || lastUserIndex() === -1);
    emptyNote.hidden = history.length > 0;
  }

  function scrollToEndIfNear() {
    var nearBottom = log.scrollHeight - log.scrollTop - log.clientHeight < 80;
    if (nearBottom) log.scrollTop = log.scrollHeight;
  }

  function addItem(msg) {
    var li = document.createElement('li');
    li.className = 'msg ' + (msg.role === 'user' ? 'msg-user' : 'msg-assistant');

    var who = document.createElement('p');
    who.className = 'who';
    who.textContent = msg.role === 'user' ? 'You' : 'Assistant';

    var body = document.createElement('div');
    body.className = 'body';
    if (msg.content) body.appendChild(document.createTextNode(msg.content));

    var note = document.createElement('p');
    note.className = 'note';
    note.hidden = true;

    li.appendChild(who);
    li.appendChild(body);
    li.appendChild(note);
    transcript.appendChild(li);

    msg.el = li;
    msg.body = body;
    msg.note = note;
    msg.indicator = null;
    history.push(msg);
    return msg;
  }

  function showNote(msg, text, isError) {
    msg.note.textContent = text;
    msg.note.className = 'note' + (isError ? ' error' : '');
    msg.note.hidden = false;
  }

  function describeError(err) {
    if (err && typeof err.message === 'string' && err.message) return err.message;
    if (typeof err === 'string' && err) return err;
    return 'The answer could not be completed.';
  }

  // Clears busy state for a stream. Called on done, stop, and error.
  function endStream(s) {
    if (active === s) active = null;
    s.msg.el.removeAttribute('aria-busy');
    if (s.msg.indicator) {
      s.msg.indicator.remove();
      s.msg.indicator = null;
    }
    updateControls();
  }

  function failStream(s, message) {
    if (active !== s) return;
    s.msg.error = message;
    showNote(s.msg, 'Error: ' + message + (s.msg.content ? ' Partial answer kept.' : ''), true);
    endStream(s);
    setStatus('Error: ' + message);
  }

  function stopStream() {
    var s = active;
    if (!s) return;
    // Mark the stream inactive first so any chunk still in flight is ignored.
    active = null;
    try {
      if (s.handle && typeof s.handle.abort === 'function') s.handle.abort();
    } catch (e) {
      // Ignore abort failures; the stream is already dead from the UI's point of view.
    }
    showNote(s.msg, 'Stopped. Partial answer kept.', false);
    endStream(s);
    setStatus('Answer stopped. Partial answer kept.');
  }

  function startAnswer(label) {
    var payload = history.map(function (m) {
      return { role: m.role, content: m.content };
    });

    var msg = addItem({ role: 'assistant', content: '', error: null });
    msg.el.setAttribute('aria-busy', 'true');

    var ind = document.createElement('span');
    ind.className = 'indicator';
    ind.setAttribute('aria-hidden', 'true');
    msg.el.appendChild(ind);
    msg.indicator = ind;

    var s = { msg: msg, handle: null };
    active = s;
    updateControls();
    setStatus(label);
    scrollToEndIfNear();

    var api = window.ChatSDK;
    if (!api || typeof api.streamChat !== 'function') {
      failStream(s, 'The chat service is not available.');
      return;
    }

    try {
      s.handle = api.streamChat(payload, {
        onToken: function (text) {
          if (active !== s) return; // stopped or replaced: never touch the transcript
          var t = text == null ? '' : String(text);
          if (!t) return;
          msg.content += t;
          msg.body.appendChild(document.createTextNode(t));
          scrollToEndIfNear();
        },
        onDone: function () {
          if (active !== s) return;
          endStream(s);
          setStatus('Answer finished.');
        },
        onError: function (err) {
          if (active !== s) return;
          failStream(s, describeError(err));
        }
      });
    } catch (err) {
      failStream(s, describeError(err));
    }
  }

  function sendMessage() {
    if (active) {
      setStatus('An answer is still generating. Wait for it to finish, or press Stop.');
      return;
    }
    var text = prompt.value.trim();
    if (!text) {
      setStatus('Type a message before sending.');
      return;
    }
    prompt.value = '';
    addItem({ role: 'user', content: text });
    updateControls();
    scrollToEndIfNear();
    startAnswer('Generating answer…');
  }

  function regenerate() {
    if (active) {
      setStatus('An answer is still generating. Wait for it to finish, or press Stop.');
      return;
    }
    var idx = lastUserIndex();
    if (idx === -1) {
      setStatus('There is no message to regenerate.');
      return;
    }
    // Remove the answer(s) that follow the last user message, then ask again.
    while (history.length > idx + 1) {
      var old = history.pop();
      old.el.remove();
    }
    updateControls();
    startAnswer('Regenerating answer…');
  }

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    sendMessage();
  });

  prompt.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      sendMessage();
    }
  });

  stopBtn.addEventListener('click', stopStream);
  regenBtn.addEventListener('click', regenerate);

  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && active) {
      stopStream();
    }
  });

  updateControls();
  setStatus('Ready.');
})();
</script>
</body>
</html>
