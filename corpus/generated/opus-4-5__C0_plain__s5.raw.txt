<body>
    <div class="chat-container">
        <div class="chat-header">Chat Assistant</div>
        
        <div class="messages-container" id="messages"></div>
        
        <div class="status-indicator" id="status"></div>
        
        <div class="input-container">
            <div class="input-row">
                <textarea id="prompt" placeholder="Type your message..." rows="1"></textarea>
            </div>

    <script>
        (function() {
            const messagesContainer = document.getElementById('messages');
            const promptInput = document.getElementById('prompt');
            const sendButton = document.getElementById('send');
            const stopButton = document.getElementById('stop');
            const regenerateButton = document.getElementById('regenerate');
            const statusIndicator = document.getElementById('status');

            let conversationHistory = [];
            let currentHandle = null;
            let isStreaming = false;
            let currentAssistantMessageEl = null;
            let currentAssistantContent = '';

            function setStatus(text, type) {
                statusIndicator.textContent = text;
                statusIndicator.className = 'status-indicator';
                if (text) {
                    statusIndicator.classList.add('visible', type);
                }
            }

            function clearStatus() {
                statusIndicator.className = 'status-indicator';
                statusIndicator.textContent = '';
            }

            function updateButtonStates() {
                sendButton.disabled = isStreaming || !promptInput.value.trim();
                stopButton.disabled = !isStreaming;
                regenerateButton.disabled = isStreaming || conversationHistory.length === 0 || conversationHistory[conversationHistory.length - 1].role !== 'assistant';
            }

            function createMessageElement(role, content) {
                const messageDiv = document.createElement('div');
                messageDiv.className = `message ${role}`;

                const roleLabel = document.createElement('div');
                roleLabel.className = 'message-role';
                roleLabel.textContent = role === 'user' ? 'You' : 'Assistant';

                const contentDiv = document.createElement('div');
                contentDiv.className = 'message-content';
                contentDiv.textContent = content;

                messageDiv.appendChild(roleLabel);
                messageDiv.appendChild(contentDiv);

                return messageDiv;
            }

            function scrollToBottom() {
                messagesContainer.scrollTop = messagesContainer.scrollHeight;
            }

            function addUserMessage(content) {
                const messageEl = createMessageElement('user', content);
                messagesContainer.appendChild(messageEl);
                conversationHistory.push({ role: 'user', content });
                scrollToBottom();
            }

            function startAssistantMessage() {
                currentAssistantContent = '';
                currentAssistantMessageEl = createMessageElement('assistant', '');
                const contentEl = currentAssistantMessageEl.querySelector('.message-content');
                contentEl.innerHTML = '<span class="typing-indicator"></span>';
                messagesContainer.appendChild(currentAssistantMessageEl);
                scrollToBottom();
            }

            function updateAssistantMessage(text) {
                currentAssistantContent += text;
                if (currentAssistantMessageEl) {
                    const contentEl = currentAssistantMessageEl.querySelector('.message-content');
                    contentEl.innerHTML = escapeHtml(currentAssistantContent) + '<span class="typing-indicator"></span>';
                    scrollToBottom();
                }
            }

            function finalizeAssistantMessage() {
                if (currentAssistantMessageEl) {
                    const contentEl = currentAssistantMessageEl.querySelector('.message-content');
                    contentEl.textContent = currentAssistantContent || '(empty response)';
                }
                if (currentAssistantContent) {
                    conversationHistory.push({ role: 'assistant', content: currentAssistantContent });
                }
                currentAssistantMessageEl = null;
            }

            function escapeHtml(text) {
                const div = document.createElement('div');
                div.textContent = text;
                return div.innerHTML;
            }

            function streamChat(messages) {
                isStreaming = true;
                updateButtonStates();
                setStatus('Generating...', 'generating');
                startAssistantMessage();

                currentHandle = window.ChatSDK.streamChat(messages, {
                    onToken: function(text) {
                        updateAssistantMessage(text);
                    },
                    onDone: function() {
                        isStreaming = false;
                        currentHandle = null;
                        finalizeAssistantMessage();
                        setStatus('Finished', 'finished');
                        updateButtonStates();
                        setTimeout(clearStatus, 2000);
                    },
                    onError: function(error) {
                        isStreaming = false;
                        currentHandle = null;
                        finalizeAssistantMessage();
                        setStatus('Error: ' + (error.message || error), 'error');
                        updateButtonStates();
                    }
                });
            }

            function sendMessage() {
                const content = promptInput.value.trim();
                if (!content || isStreaming) return;

                promptInput.value = '';
                addUserMessage(content);
                updateButtonStates();

                streamChat([...conversationHistory]);
            }

            function stopGeneration() {
                if (currentHandle && isStreaming) {
                    currentHandle.abort();
                    isStreaming = false;
                    currentHandle = null;
                    finalizeAssistantMessage();
                    setStatus('Stopped', 'stopped');
                    updateButtonStates();
                    setTimeout(clearStatus, 2000);
                }
            }

            function regenerate() {
                if (isStreaming || conversationHistory.length === 0) return;

                // Remove the last assistant message from history and DOM
                if (conversationHistory[conversationHistory.length - 1].role === 'assistant') {
                    conversationHistory.pop();
                    const lastMessageEl = messagesContainer.querySelector('.message.assistant:last-child');
                    if (lastMessageEl) {
                        lastMessageEl.remove();
                    }
                }

                // If there's still a user message, regenerate from it
                if (conversationHistory.length > 0 && conversationHistory[conversationHistory.length - 1].role === 'user') {
                    streamChat([...conversationHistory]);
                }
            }

            // Event listeners
            sendButton.addEventListener('click', sendMessage);
            stopButton.addEventListener('click', stopGeneration);
            regenerateButton.addEventListener('click', regenerate);

            promptInput.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    sendMessage();
                }
            });

            promptInput.addEventListener('input', function() {
                updateButtonStates();
                // Auto-resize textarea
                this.style.height = 'auto';
                this.style.height = Math.min(this.scrollHeight, 120) + 'px';
            });

            // Initial state
            updateButtonStates();
        })();
    </script>
</body>
</html>
