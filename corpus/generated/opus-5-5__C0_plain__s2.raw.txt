<script>
(function () {
  'use strict';

  // ---- State ----
  // Each message: { role, content, status } where status for assistant is
  // 'streaming' | 'finished' | 'stopped' | 'error'
  var messages = [];
  var current = null; // active request: { msg, handle, closed }

  // ---- Elements ----
  var convEl = document.getElementById('conversation');
  var emptyEl = document.getElementById('empty');
  var promptEl = document.getElementById('prompt');
  var sendBtn = document.getElementById('send');
  var stopBtn = document.getElementById('stop');
  var regenBtn = document.getElementById('regenerate');
  var statusEl = document.getElementById('status');
  var formEl = document.getElementById('composer');

  // ---- Helpers ----
  function isStreaming() { return current !== null; }

  function lastUserIndex() {
    for (var i = messages.length - 1; i >= 0; i--) {
      if (messages[i].role === 'user') return i;
    }
    return -1;
  }

  function setStatus(kind, text) {
    statusEl.className = kind || '';
    statusEl.textContent = text;
  }

  function updateControls() {
    var streaming = isStreaming();
    sendBtn.disabled = streaming || promptEl.value.trim() === '';
    stopBtn.disabled = !streaming;
    regenBtn.disabled = streaming || lastUserIndex() === -1;
  }

  function nearBottom() {
    return convEl.scrollHeight - convEl.scrollTop - convEl.clientHeight < 80;
  }

  function scrollToBottom() { convEl.scrollTop = convEl.scrollHeight; }

  function autoResize() {
    promptEl.style.height = 'auto';
    promptEl.style.height = Math.min(promptEl.scrollHeight, 200) + 'px';
  }

  // ---- Rendering ----
  function createMessageEl(msg) {
    var wrap = document.createElement('div');
    wrap.className = 'msg ' + msg.role;

    var role = document.createElement('div');
    role.className = 'role';
    role.textContent = msg.role === 'user' ? 'You' : 'Assistant';

    var bubble = document.createElement('div');
    bubble.className = 'bubble';

    var note = document.createElement('div');
    note.className = 'note';

    wrap.appendChild(role);
    wrap.appendChild(bubble);
    wrap.appendChild(note);

    msg._el = wrap;
    msg._bubble = bubble;
    msg._note = note;
    updateMessageEl(msg);
    return wrap;
  }

  function updateMessageEl(msg) {
    if (!msg._el) return;
    var bubble = msg._bubble;
    var note = msg._note;

    if (msg.role === 'assistant') {
      msg._el.classList.toggle('streaming', msg.status === 'streaming');
      if (msg.content === '') {
        bubble.classList.add('placeholder');
        if (msg.status === 'streaming') bubble.textContent = 'Thinking…';
        else if (msg.status === 'stopped') bubble.textContent = '(stopped before any text was generated)';
        else if (msg.status === 'error') bubble.textContent = '(no response)';
        else bubble.textContent = '(empty response)';
      } else {
        bubble.classList.remove('placeholder');
        bubble.textContent = msg.content;
      }

      note.className = 'note';
      if (msg.status === 'stopped') {
        note.classList.add('stopped');
        note.textContent = 'Stopped by user';
      } else if (msg.status === 'error') {
        note.classList.add('error');
        note.textContent = 'Error: ' + (msg.error || 'unknown error');
      } else {
        note.textContent = '';
      }
    } else {
      bubble.textContent = msg.content;
      note.textContent = '';
    }
  }

  function renderAll() {
    while (convEl.firstChild) convEl.removeChild(convEl.firstChild);
    if (messages.length === 0) {
      convEl.appendChild(emptyEl);
      return;
    }
    for (var i = 0; i < messages.length; i++) {
      convEl.appendChild(messages[i]._el || createMessageEl(messages[i]));
    }
    scrollToBottom();
  }

  function appendMessage(msg) {
    messages.push(msg);
    if (emptyEl.parentNode) emptyEl.parentNode.removeChild(emptyEl);
    convEl.appendChild(createMessageEl(msg));
    scrollToBottom();
  }

  function errorText(err) {
    if (!err) return 'unknown error';
    if (typeof err === 'string') return err;
    if (err.message) return String(err.message);
    try { return JSON.stringify(err); } catch (e) { return String(err); }
  }

  // ---- Streaming ----
  function buildHistory() {
    // Send user messages and assistant answers that contain text.
    var out = [];
    for (var i = 0; i < messages.length; i++) {
      var m = messages[i];
      if (m.role === 'user') {
        out.push({ role: 'user', content: m.content });
      } else if (m.content !== '' && m.status !== 'streaming') {
        out.push({ role: 'assistant', content: m.content });
      }
    }
    return out;
  }

  function closeRequest(req, status, err) {
    if (req.closed) return;
    req.closed = true;
    if (current === req) current = null;
    req.msg.status = status;
    if (err !== undefined) req.msg.error = errorText(err);
    updateMessageEl(req.msg);

    if (status === 'finished') setStatus('finished', 'Finished');
    else if (status === 'stopped') setStatus('stopped', 'Stopped');
    else if (status === 'error') setStatus('error', 'Error: ' + req.msg.error);

    updateControls();
  }

  function startAssistant() {
    var history = buildHistory();
    var msg = { role: 'assistant', content: '', status: 'streaming' };
    appendMessage(msg);

    var req = { msg: msg, handle: null, closed: false };
    current = req;
    setStatus('generating', 'Generating…');
    updateControls();

    var callbacks = {
      onToken: function (text) {
        // Ignore tokens arriving after stop/finish/error (e.g. one in flight after abort()).
        if (req.closed || current !== req) return;
        if (text == null) return;
        var stick = nearBottom();
        msg.content += String(text);
        updateMessageEl(msg);
        if (stick) scrollToBottom();
      },
      onDone: function () {
        if (req.closed) return;
        closeRequest(req, 'finished');
      },
      onError: function (error) {
        if (req.closed) return;
        closeRequest(req, 'error', error);
      }
    };

    try {
      if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
        throw new Error('Chat service is unavailable.');
      }
      var handle = window.ChatSDK.streamChat(history, callbacks);
      req.handle = handle;
      // If the stream already ended synchronously, nothing else to do.
    } catch (e) {
      closeRequest(req, 'error', e);
    }
  }

  function stopStreaming() {
    var req = current;
    if (!req) return;
    // Mark closed first so any late onToken is ignored; partial text is kept.
    closeRequest(req, 'stopped');
    try {
      if (req.handle && typeof req.handle.abort === 'function') req.handle.abort();
    } catch (e) {
      // Ignore abort failures; the answer is already marked as stopped locally.
    }
  }

  function send() {
    if (isStreaming()) return;
    var text = promptEl.value;
    if (text.trim() === '') return;
    appendMessage({ role: 'user', content: text });
    promptEl.value = '';
    autoResize();
    startAssistant();
    promptEl.focus();
  }

  function regenerate() {
    if (isStreaming()) return;
    var idx = lastUserIndex();
    if (idx === -1) return;
    // Remove any assistant answers after the last user message.
    messages = messages.slice(0, idx + 1);
    renderAll();
    startAssistant();
  }

  // ---- Events ----
  formEl.addEventListener('submit', function (e) {
    e.preventDefault();
    send();
  });

  promptEl.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });

  promptEl.addEventListener('input', function () {
    autoResize();
    updateControls();
  });

  stopBtn.addEventListener('click', stopStreaming);
  regenBtn.addEventListener('click', regenerate);

  // ---- Init ----
  setStatus('', 'Ready');
  updateControls();
  promptEl.focus();
})();
</script>
</body>
</html>
