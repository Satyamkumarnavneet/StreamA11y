    <div class="container">
        <div class="header">Chat Assistant</div>

            <div class="empty-state">

    <script>
        const messagesContainer = document.getElementById('messages');
        const promptInput = document.getElementById('prompt');
        const sendButton = document.getElementById('send');
        const stopButton = document.getElementById('stop');
        const regenerateButton = document.getElementById('regenerate');

        let messages = [];
        let isStreaming = false;
        let currentHandle = null;

        function clearEmptyState() {
            const emptyState = messagesContainer.querySelector('.empty-state');
            if (emptyState) {
                emptyState.remove();
            }
        }

        function addMessage(role, content) {
            clearEmptyState();
            const messageEl = document.createElement('div');
            messageEl.className = `message ${role}`;
            
            const contentEl = document.createElement('div');
            contentEl.className = 'message-content';
            contentEl.textContent = content;
            
            messageEl.appendChild(contentEl);
            messagesContainer.appendChild(messageEl);
            messagesContainer.scrollTop = messagesContainer.scrollHeight;
            
            return messageEl;
        }

        function addStatusMessage(text, type) {
            const statusEl = document.createElement('div');
            statusEl.className = `status ${type}`;
            statusEl.textContent = text;
            statusEl.id = 'status-message';
            messagesContainer.appendChild(statusEl);
            messagesContainer.scrollTop = messagesContainer.scrollHeight;
            return statusEl;
        }

        function removeStatusMessage() {
            const status = document.getElementById('status-message');
            if (status) {
                status.remove();
            }
        }

        function updateLastAssistantMessage(text) {
            const messages = messagesContainer.querySelectorAll('.message.assistant');
            if (messages.length > 0) {
                const lastMessage = messages[messages.length - 1];
                const contentEl = lastMessage.querySelector('.message-content');
                if (contentEl) {
                    contentEl.textContent = text;
                }
            }
            messagesContainer.scrollTop = messagesContainer.scrollHeight;
        }

        function updateButtonStates() {
            sendButton.disabled = isStreaming || promptInput.value.trim() === '';
            stopButton.disabled = !isStreaming;
            regenerateButton.disabled = isStreaming || messages.length < 2;
        }

        function autoResizeTextarea() {
            promptInput.style.height = 'auto';
            promptInput.style.height = Math.min(promptInput.scrollHeight, 120) + 'px';
        }

        async function sendMessage() {
            const content = promptInput.value.trim();
            if (!content || isStreaming) return;

            promptInput.value = '';
            autoResizeTextarea();

            messages.push({ role: 'user', content });
            addMessage('user', content);

            await streamResponse();
        }

        async function streamResponse() {
            isStreaming = true;
            updateButtonStates();

            clearEmptyState();
            addMessage('assistant', '');
            addStatusMessage('Generating...', 'generating');

            let assistantContent = '';
            const messages_copy = [...messages];

            currentHandle = window.ChatSDK.streamChat(messages_copy, {
                onToken: (text) => {
                    assistantContent += text;
                    updateLastAssistantMessage(assistantContent);
                },
                onDone: () => {
                    removeStatusMessage();
                    messages.push({ role: 'assistant', content: assistantContent });
                    isStreaming = false;
                    currentHandle = null;
                    updateButtonStates();
                },
                onError: (error) => {
                    removeStatusMessage();
                    addStatusMessage(`Error: ${error.message}`, 'error');
                    messages.push({ role: 'assistant', content: assistantContent });
                    isStreaming = false;
                    currentHandle = null;
                    updateButtonStates();
                }
            });
        }

        function stopStreaming() {
            if (currentHandle && isStreaming) {
                currentHandle.abort();
                removeStatusMessage();
                addStatusMessage('Stopped', 'stopped');
                messages.push({ role: 'assistant', content: document.querySelectorAll('.message.assistant')[-1]?.querySelector('.message-content')?.textContent || '' });
                isStreaming = false;
                currentHandle = null;
                updateButtonStates();
            }
        }

        async function regenerate() {
            if (messages.length < 2 || isStreaming) return;

            const lastUserMessageIndex = messages.length - 2;
            messages = messages.slice(0, lastUserMessageIndex + 1);

            messagesContainer.innerHTML = '';
            for (const msg of messages) {
                addMessage(msg.role, msg.content);
            }

            await streamResponse();
        }

        sendButton.addEventListener('click', sendMessage);
        stopButton.addEventListener('click', stopStreaming);
        regenerateButton.addEventListener('click', regenerate);

        promptInput.addEventListener('keydown', (e) => {
            if (e.key === 'Enter' && !e.shiftKey) {
                e.preventDefault();
                sendMessage();
            }
        });

        promptInput.addEventListener('input', () => {
            autoResizeTextarea();
            updateButtonStates();
        });

        updateButtonStates();
    </script>
</body>
