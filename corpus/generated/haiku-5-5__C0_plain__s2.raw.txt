<script>
(function () {
  'use strict';

  var els = {
    conversation: document.getElementById('conversation'),
    status: document.getElementById('status'),
    prompt: document.getElementById('prompt'),
    send: document.getElementById('send'),
    stop: document.getElementById('stop'),
    regenerate: document.getElementById('regenerate')
  };

  // Conversation history: [{role: 'user' | 'assistant', content: string}]
  var history = [];

  // The stream currently in progress, or null.
  // { id, handle, bubble, index }
  var active = null;
  var nextRequestId = 1;

  function setStatus(kind, text) {
    els.status.className = 'status' + (kind ? ' ' + kind : '');
    els.status.textContent = text;
  }

  function hasUserMessage() {
    for (var i = 0; i < history.length; i++) {
      if (history[i].role === 'user') return true;
    }
    return false;
  }

  function updateButtons() {
    var streaming = !!active;
    els.send.disabled = streaming;
    els.stop.disabled = !streaming;
    els.regenerate.disabled = streaming || !hasUserMessage();
  }

  function scrollToBottom() {
    els.conversation.scrollTop = els.conversation.scrollHeight;
  }

  // Rebuilds the whole conversation from history. Uses textContent only,
  // so model output is never interpreted as HTML.
  function renderHistory() {
    els.conversation.innerHTML = '';

    if (history.length === 0) {
      var empty = document.createElement('div');
      empty.className = 'empty';
      empty.textContent = 'Send a message to start the conversation.';
      els.conversation.appendChild(empty);
      return null;
    }

    var lastBubble = null;
    for (var i = 0; i < history.length; i++) {
      var m = history[i];
      var wrap = document.createElement('div');
      wrap.className = 'msg ' + m.role;

      var label = document.createElement('div');
      label.className = 'role';
      label.textContent = m.role === 'user' ? 'You' : 'Assistant';

      var content = document.createElement('div');
      content.className = 'content';
      content.textContent = m.content;

      wrap.appendChild(label);
      wrap.appendChild(content);
      els.conversation.appendChild(wrap);
      lastBubble = { wrap: wrap, content: content };
    }
    scrollToBottom();
    return lastBubble;
  }

  // Removes a trailing empty assistant placeholder, if any.
  function dropEmptyAssistant() {
    var last = history[history.length - 1];
    if (last && last.role === 'assistant' && last.content === '') {
      history.pop();
    }
  }

  function finishStreamUi(bubbleWrap) {
    if (bubbleWrap) bubbleWrap.classList.remove('streaming');
  }

  function startStream() {
    var requestId = nextRequestId++;

    // Snapshot the messages to send (history as it stands, no placeholder).
    var messages = history.map(function (m) {
      return { role: m.role, content: m.content };
    });

    // Placeholder assistant entry that receives streamed tokens.
    history.push({ role: 'assistant', content: '' });
    var index = history.length - 1;

    var rendered = renderHistory();
    var req = { id: requestId, handle: null, bubble: rendered ? rendered.content : null, wrap: rendered ? rendered.wrap : null, index: index };
    if (req.wrap) req.wrap.classList.add('streaming');

    active = req;
    setStatus('generating', 'Generating…');
    updateButtons();

    function isCurrent() {
      return active === req;
    }

    function completeNormally(kind, text) {
      active = null;
      finishStreamUi(req.wrap);
      dropEmptyAssistant();
      setStatus(kind, text);
      renderHistory();
      updateButtons();
      els.prompt.focus();
    }

    var callbacks = {
      onToken: function (text) {
        // Ignore anything that arrives after stop/abort or from an old request.
        if (!isCurrent()) return;
        var entry = history[req.index];
        if (!entry) return;
        entry.content += String(text);
        if (req.bubble) req.bubble.textContent = entry.content;
        scrollToBottom();
      },
      onDone: function () {
        if (!isCurrent()) return;
        completeNormally('done', 'Finished');
      },
      onError: function (error) {
        if (!isCurrent()) return;
        var msg = (error && error.message) ? error.message : String(error || 'Unknown error');
        completeNormally('error', 'Error: ' + msg);
      }
    };

    try {
      req.handle = window.ChatSDK.streamChat(messages, callbacks);
    } catch (err) {
      if (isCurrent()) {
        var msg = (err && err.message) ? err.message : String(err);
        completeNormally('error', 'Error: ' + msg);
      }
    }
  }

  function send() {
    if (active) return;
    var text = els.prompt.value.trim();
    if (!text) return;

    history.push({ role: 'user', content: text });
    els.prompt.value = '';
    renderHistory();
    startStream();
  }

  function stop() {
    if (!active) return;
    var req = active;

    // Mark as inactive first so any late onToken is ignored.
    active = null;
    finishStreamUi(req.wrap);

    try {
      if (req.handle && typeof req.handle.abort === 'function') {
        req.handle.abort();
      }
    } catch (e) {
      // Ignore abort failures; the stream is already marked stopped locally.
    }

    // Keep the partial answer. Drop it only if nothing was shown at all.
    dropEmptyAssistant();
    setStatus('stopped', 'Stopped');
    renderHistory();
    updateButtons();
    els.prompt.focus();
  }

  function regenerate() {
    if (active) return;

    var lastUser = -1;
    for (var i = history.length - 1; i >= 0; i--) {
      if (history[i].role === 'user') { lastUser = i; break; }
    }
    if (lastUser === -1) return;

    // Discard everything after the last user message (old answer(s)).
    history.length = lastUser + 1;
    renderHistory();
    startStream();
  }

  // ---- Wiring ----

  els.send.addEventListener('click', send);
  els.stop.addEventListener('click', stop);
  els.regenerate.addEventListener('click', regenerate);

  els.prompt.addEventListener('keydown', function (e) {
    if (e.key === 'Enter' && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      send();
    }
  });

  // ---- Init ----

  if (!window.ChatSDK || typeof window.ChatSDK.streamChat !== 'function') {
    setStatus('error', 'Error: chat service is not available');
    els.send.disabled = true;
    els.prompt.disabled = true;
  } else {
    setStatus('', 'Ready');
  }

  renderHistory();
  updateButtons();
})();
</script>
</body>
</html>
